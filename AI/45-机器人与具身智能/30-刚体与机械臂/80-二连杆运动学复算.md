# 二连杆运动学复算

这组有限计算对应[[AI/45-机器人与具身智能/30-刚体与机械臂/01-二连杆正逆运动学与Jacobian|二连杆主解释]]的同一条固定基座平面2R机械臂。它测的是理想FK/IK/导数/功率是否一致，**没有**积分机械臂动力学、运行反馈、观测真实传感器或驱动物理执行器。单位杆的近奇异反例不是一张真实测速图。

## 输入与复跑入口

固定输入为同目录的 [inputs.json](inputs.json)，共5个姿态、8个位置IK目标、3组力与速度、4个近奇异角度和5个有限核方向曲线点。姿态/IK项各自带杆长；力/速度/有限曲线三组固定为 $l_1=l_2=1$ m。角度为rad、逆时针为正；所有末端点、力和线速度在世界系W表示。没有种子筛选、训练或后验参数搜索。输入里的 `.5235987755982988` 等是 $\pi/6$ 的float64近似，不是精确实数。

规范源码在 `code-sync/`，下方逐字镜像供阅读。`arm_kinematics.py`只读导入现有 `AI/45-机器人与具身智能/robotics_lab.py` 的 `finite/positive/se2/rotation`，不复制旧仿真或使用其点质量 `model()`。来源中的列向量SE(2)合同保持唯一。正常落库后默认按相对路径找到该文件；隔离候选用 `--base <已发表robotics_lab.py>`，测试用环境变量 `WWL_ROBOTICS_BASE` 指向相同文件，避免偷偷写死作者机器路径。任意数据适配应显式提供新输入和单位，不能把真实关节日志的“度”当作rad。

浮点接口要求指定shape、有限数、严格正杆长；超大/极小float及极端病态IK不作精度保证。IK只容许 $|c_2|-1\le10^{-12}$ 的舍入越界被夹回端点，明显越界返回 `unreachable`；这是余弦域上的无量纲容差，不是所有长度尺度上一律 $10^{-12}$ m的位置保证。返回 `boundary` 是数值边界判断，足够近的不同实数输入可能在浮点中合并。精确等长且输入点恰为原点才返回连续族，不能把小球内目标都粗暴当作原点。`speed_request`在 $\sigma_{\min}\le10^{-12}$ m/rad 时拒绝普通求逆；本轮杆长为1 m，不能不改阈值就把这个绝对门槛当作跨尺度条件数判据。

## 读保存结果，改变一个条件

`results.json.configurations`保存每个姿态的 $T_{WE}$、肘点/末端点、$J_p$、$J_{pose}$、奇异值及明确 $10^{-12}$ 秩阈值；`targets`保存解类型、原始/夹取余弦、每支关节和回代误差。连续族不是给一组代表角假称全体，所以 `branches` 为空而有 `family` 字段。`power_cases`保存原始力/关节速度、输出速度/力矩和两端功率；`near_singular`保存逆需求、uniform scale和逐项clip后的真实 $J_p\dot q$，不把“已剪裁”当作“任务仍完成”。`null_curve`保存每点关节、肘和末端；两幅SVG由这些同一记录生成。

先尝试：把近奇异请求从 $(-.1,0)$ 换为 $(0,.1)$ m/s，会不会仍按 $1/\epsilon$ 发散？答案：对 $q=(0,\epsilon)$、$\sin\epsilon\ne0$，第一行给 $\dot q_1+\dot q_2=0$，第二行给 $\dot q_1=.1$、$\dot q_2=-.1$，不发散；奇异敏感性与所请求方向有关。当前 `run()`固定取这组单位杆，改变输入中的速度后可重跑并比较，但应保存为新结果，不覆盖本轮教学基准。

再尝试：把所有长度乘1000并把目标/线速度也乘1000，关节逆速度需求应保持，$J_p$的奇异值乘1000。这说明用统一单位与相对条件比裸绝对阈值重要；单位从m换成mm时，力仍为N，功率/力矩输出也必须同步换算回SI，不可把N·mm错标为N·m。

"""Fixed-base planar 2R position kinematics; m, rad, s, N. No dynamics.
Reuse the published robotics_lab.py column-vector SE(2), never its point-mass model.
"""
import argparse
import hashlib
import importlib.util
import json
from pathlib import Path
import numpy as np

_BASE = None
_BASE_PATH = None


def set_base(path=None):
    global _BASE, _BASE_PATH
    path = Path(path) if path is not None else Path(__file__).resolve().parents[2] / 'robotics_lab.py'
    spec = importlib.util.spec_from_file_location('wwl_planar_base', path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    _BASE, _BASE_PATH = module, path.resolve()


def base():
    if _BASE is None:
        set_base()
    return _BASE


def inputs(q, lengths):
    q = base().finite(q, (2,), 'q radians')
    lengths = base().finite(lengths, (2,), 'link metres')
    if np.any(lengths <= 0):
        raise ValueError('link lengths must be strictly positive')
    return q, lengths


def forward(q, lengths=(1., 1.)):
    """T_WE (3,3), elbow_W (2,), endpoint_W (2,); right factor acts first."""
    q, lengths = inputs(q, lengths)
    first = base().se2([0., 0.], q[0])
    second = base().se2([lengths[0], 0.], q[1])
    tool = base().se2([lengths[1], 0.], 0.)
    transform = first @ second @ tool
    elbow = (first @ np.array([lengths[0], 0., 1.]))[:2]
    return transform, elbow, transform[:2, 2].copy()


def jacobian(q, lengths=(1., 1.)):
    """Position derivative d(x_W,y_W)/d(q1,q2), 2x2, m/rad."""
    q, (a, b) = inputs(q, lengths)
    s1, c1 = np.sin(q[0]), np.cos(q[0])
    s12, c12 = np.sin(q.sum()), np.cos(q.sum())
    return np.array([[-a*s1-b*s12, -b*s12], [a*c1+b*c12, b*c12]])


def pose_jacobian(q, lengths=(1., 1.)):
    """Local coordinate derivative (x_W,y_W,phi)/q; not a spatial twist J."""
    return np.vstack([jacobian(q, lengths), [1., 1.]])


def inverse(target, lengths=(1., 1.), cosine_tolerance=1e-12):
    """Unconstrained position IK modulo 2pi. Exact zero/equal lengths => continuum.
    Only |c2|-1 <= tolerance is clamped; boundary answers are numerical.
    """
    _, (a, b) = inputs([0., 0.], lengths)
    target = base().finite(target, (2,), 'target metres')
    tolerance = base().positive(cosine_tolerance, 'cosine tolerance', allow_zero=True)
    if np.all(target == 0.) and a == b:
        return {'kind': 'continuum', 'branches': [], 'family': 'q1 arbitrary; q2 = pi modulo 2pi'}
    with np.errstate(over='raise', invalid='raise', divide='raise'):
        c = float((target @ target - a*a - b*b) / (2*a*b))
    if not np.isfinite(c):
        raise ValueError('cosine not representable')
    if abs(c) > 1. + tolerance:
        return {'kind': 'unreachable', 'branches': [], 'cosine_raw': c}
    clipped = float(np.clip(c, -1., 1.))
    sine = np.sqrt(max(0., 1.-clipped*clipped))
    signs = [1.] if sine == 0. else [1., -1.]
    branches = []
    for sign in signs:
        q2 = np.arctan2(sign*sine, clipped)
        q1 = np.arctan2(target[1], target[0]) - np.arctan2(b*np.sin(q2), a+b*np.cos(q2))
        q = (np.array([q1, q2]) + np.pi) % (2*np.pi) - np.pi
        branches.append(q.tolist())
    return {'kind': 'boundary' if sine == 0. else 'two_branches', 'branches': branches,
            'cosine_raw': c, 'cosine_clamped': clipped}


def power(q, qdot, force, lengths=(1., 1.)):
    """F is desired robot-on-environment endpoint force, both F and v in W.
    tau = J.T F is its ideal port-equivalent joint torque, not inverse dynamics.
    """
    j = jacobian(q, lengths)
    qdot = base().finite(qdot, (2,), 'joint rad/s')
    force = base().finite(force, (2,), 'endpoint force N')
    velocity, torque = j @ qdot, j.T @ force
    return {'velocity_m_s': velocity.tolist(), 'torque_N_m': torque.tolist(),
            'endpoint_power_W': float(force @ velocity), 'joint_power_W': float(torque @ qdot)}


def speed_request(q, velocity, limit, lengths=(1., 1.)):
    j = jacobian(q, lengths)
    velocity = base().finite(velocity, (2,), 'velocity m/s')
    limit = base().positive(limit, 'joint speed rad/s')
    # This API reports inverse demand only at numerically nonsingular configurations.
    if np.linalg.svd(j, compute_uv=False)[-1] <= 1e-12:
        raise ValueError('position Jacobian numerically singular; no ordinary inverse')
    demand = np.linalg.solve(j, velocity)
    scale = min(1., limit / max(float(np.max(np.abs(demand))), np.finfo(float).tiny))
    scaled = scale * demand
    clipped = np.clip(demand, -limit, limit)
    return {'demand_rad_s': demand.tolist(), 'scale': scale,
            'scaled_rad_s': scaled.tolist(), 'scaled_velocity_m_s': (j @ scaled).tolist(),
            'clipped_rad_s': clipped.tolist(), 'clipped_velocity_m_s': (j @ clipped).tolist()}


def run(config):
    cases = []
    for item in config['configurations']:
        q, lengths = item['q_rad'], item['lengths_m']
        transform, elbow, endpoint = forward(q, lengths)
        j = jacobian(q, lengths)
        cases.append({**item, 'T_WE': transform.tolist(), 'elbow_W_m': elbow.tolist(),
                      'endpoint_W_m': endpoint.tolist(), 'J_position_m_rad': j.tolist(),
                      'J_pose': pose_jacobian(q, lengths).tolist(),
                      'singular_values_m_rad': np.linalg.svd(j, compute_uv=False).tolist(),
                      'position_rank_at_1e-12': int(np.linalg.matrix_rank(j, tol=1e-12)),
                      'pose_rank_at_1e-12': int(np.linalg.matrix_rank(pose_jacobian(q, lengths), tol=1e-12))})
    ik = []
    for item in config['targets']:
        answer = inverse(item['target_W_m'], item['lengths_m'])
        checked = []
        for q in answer['branches']:
            _, _, endpoint = forward(q, item['lengths_m'])
            checked.append({'q_rad': q, 'endpoint_W_m': endpoint.tolist(),
                            'residual_m': float(np.linalg.norm(endpoint - item['target_W_m']))})
        ik.append({**item, **answer, 'forward_checks': checked})
    near = []
    for epsilon in config['near_singular']['epsilon_rad']:
        q = [0., epsilon]
        near.append({'epsilon_rad': epsilon,
                     'sigma_min_m_rad': float(np.linalg.svd(jacobian(q), compute_uv=False)[-1]),
                     **speed_request(q, config['near_singular']['velocity_W_m_s'],
                                     config['near_singular']['joint_limit_rad_s'])})
    curve = []
    for epsilon in config['null_curve_epsilon_rad']:
        q = [epsilon, -2*epsilon]
        _, elbow, endpoint = forward(q)
        curve.append({'epsilon_rad': epsilon, 'q_rad': q, 'elbow_W_m': elbow.tolist(),
                      'endpoint_W_m': endpoint.tolist(), 'inward_displacement_m': float(2.-endpoint[0])})
    return {'model': config['model'], 'units': config['units'], 'configurations': cases,
            'targets': ik, 'power_cases': [{**item, **power(item['q_rad'], item['qdot_rad_s'], item['force_W_N'])}
                                         for item in config['power_cases']],
            'near_singular': near, 'null_curve': curve,
            'base_sha256': hashlib.sha256(_BASE_PATH.read_bytes()).hexdigest(),
            'source_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--base', type=Path, help='published robotics_lab.py; needed in isolated candidate')
    parser.add_argument('--input', type=Path, default=Path(__file__).resolve().parents[1] / 'inputs.json')
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args()
    set_base(args.base)
    config = json.loads(args.input.read_text())
    result = run(config)
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(result, ensure_ascii=False, indent=2, allow_nan=False) + '\n')
    print(json.dumps({'configurations': len(result['configurations']), 'targets': len(result['targets']),
                      'power_cases': len(result['power_cases']), 'near_singular': len(result['near_singular'])}))


if __name__ == '__main__':
    main()

"""Finite analytic expectations and derivative checks, not hardware acceptance."""
import importlib.util
import os
from pathlib import Path
import unittest
import numpy as np

SPEC = importlib.util.spec_from_file_location('arm', Path(__file__).with_name('arm_kinematics.py'))
arm = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(arm)


class ArmTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        arm.set_base(os.environ.get('WWL_ROBOTICS_BASE'))

    def test_chain_and_regular_closed_form(self):
        t, elbow, endpoint = arm.forward([np.pi/6, -np.pi/3])
        np.testing.assert_allclose(elbow, [np.sqrt(3)/2, .5], atol=1e-14)
        np.testing.assert_allclose(endpoint, [np.sqrt(3), 0], atol=1e-14)
        np.testing.assert_allclose(t[:2, :2], arm.base().rotation(-np.pi/6), atol=1e-14)
        np.testing.assert_allclose(arm.jacobian([np.pi/6, -np.pi/3]),
                                   [[0., .5], [np.sqrt(3), np.sqrt(3)/2]], atol=1e-14)
        # Translation/rotation order already shares se2; this detects reversed chain factors.
        np.testing.assert_allclose(arm.forward([np.pi/2, 0])[2], [0, 2], atol=1e-14)

    def test_position_jacobian_differences_and_frame_change(self):
        h, worst = 1e-6, 0.
        for lengths in ([1., 1.], [1.3, .8]):
            for q in ([0., 0.], [.2, -1.1], [1., np.pi], [-.7, 1.2]):
                q = np.array(q)
                fd = np.column_stack([(arm.forward(q+h*np.eye(2)[i], lengths)[2]
                                        -arm.forward(q-h*np.eye(2)[i], lengths)[2])/(2*h)
                                       for i in range(2)])
                j = arm.jacobian(q, lengths)
                worst = max(worst, float(np.max(np.abs(fd-j))))
                np.testing.assert_allclose(fd, j, atol=5e-9, rtol=0)
                # A fixed observer rotation maps endpoint velocities, not qdot.
                r = arm.base().rotation(.4)
                force, qdot = np.array([2., -3.]), np.array([.3, .2])
                np.testing.assert_allclose((r@j).T@(r@force), j.T@force, atol=1e-14)
                self.assertAlmostEqual(float((r@force)@((r@j)@qdot)), float(force@(j@qdot)), places=13)
        print('author central-difference max_abs_error', worst)

    def test_inverse_branches_boundaries_family_and_hole(self):
        answer = arm.inverse([1., 1.])
        self.assertEqual(answer['kind'], 'two_branches')
        np.testing.assert_allclose(answer['branches'], [[0, np.pi/2], [np.pi/2, -np.pi/2]], atol=1e-14)
        for lengths, target in [([1.,1.],[1.,1.]), ([1.5,1.],[.5,0.]),
                                ([1.,1.5],[.5,0.]), ([1.,1.],[2.,0.]),
                                ([1.3,.8],[-.5,1.])]:
            result = arm.inverse(target, lengths)
            for q in result['branches']:
                np.testing.assert_allclose(arm.forward(q, lengths)[2], target, atol=1e-12)
        self.assertEqual(arm.inverse([0.,0.])['kind'], 'continuum')
        for q1 in [-2., -.3, 0., .7, 2.]:
            np.testing.assert_allclose(arm.forward([q1, np.pi])[2], [0.,0.], atol=1e-14)
        self.assertEqual(arm.inverse([.25,0.], [1.5,1.])['kind'], 'unreachable')
        self.assertEqual(arm.inverse([2.1,0.])['kind'], 'unreachable')
        self.assertEqual(arm.inverse([2.+1e-14,0.])['kind'], 'boundary')

    def test_singular_task_and_finite_null_curve(self):
        j = arm.jacobian([0.,0.])
        np.testing.assert_array_equal(j, [[0.,0.], [2.,1.]])
        self.assertEqual(np.linalg.matrix_rank(j), 1)
        self.assertEqual(np.linalg.matrix_rank(arm.pose_jacobian([0.,0.])), 2)
        np.testing.assert_array_equal(j@np.array([1.,-2.]), [0.,0.])
        for epsilon in [.005,.02,.1]:
            endpoint = arm.forward([epsilon,-2*epsilon])[2]
            np.testing.assert_allclose(endpoint, [2*np.cos(epsilon), 0.], atol=1e-14)
            self.assertGreater(2.-endpoint[0], 0.)
        with self.assertRaises(ValueError):
            arm.speed_request([0.,0.], [-.1,0.], 1.)

    def test_power_and_near_singular_limit(self):
        p = arm.power([0.,0.], [.3,-.2], [3.,-2.])
        np.testing.assert_allclose(p['velocity_m_s'], [0.,.4], atol=1e-14)
        np.testing.assert_allclose(p['torque_N_m'], [-4.,-2.], atol=1e-14)
        self.assertAlmostEqual(p['joint_power_W'], -.8)
        self.assertAlmostEqual(p['endpoint_power_W'], -.8)
        for epsilon in [.5,.1,.02,.005]:
            request = arm.speed_request([0.,epsilon], [-.1,0.], 1.)
            expected = [-.1/np.tan(epsilon), .1*(1+np.cos(epsilon))/np.sin(epsilon)]
            np.testing.assert_allclose(request['demand_rad_s'], expected, atol=1e-10)
            self.assertLessEqual(max(abs(np.array(request['scaled_rad_s']))), 1.+1e-14)
            np.testing.assert_allclose(request['scaled_velocity_m_s'], [-.1*request['scale'],0.], atol=1e-14)

    def test_reject_wrong_shapes_and_nonfinite(self):
        for q, lengths in [([0.,0.],[0.,1.]), ([0.,0.],[-1.,1.]), ([float('nan'),0.],[1.,1.]),
                           ([0.,0.,0.],[1.,1.]), ([0.,0.],[1.,float('inf')])]:
            with self.assertRaises(ValueError):
                arm.forward(q, lengths)
        with self.assertRaises(ValueError):
            arm.power([0.,0.], [.2,.3], [1.,2.,3.])

    unittest.main(verbosity=2)
