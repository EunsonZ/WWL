cmake_minimum_required(VERSION 3.20)
project(WWLABICConsumer LANGUAGES C)
include(CTest)
find_package(WWLMeasurementABI 1 CONFIG REQUIRED)
add_executable(abi_consumer consumer.c)
target_compile_features(abi_consumer PRIVATE c_std_11)
set_target_properties(abi_consumer PROPERTIES C_EXTENSIONS OFF)
target_link_libraries(abi_consumer PRIVATE WWL::measurement_abi)
if(MSVC)
  target_compile_options(abi_consumer PRIVATE /W4)
else()
  target_compile_options(abi_consumer PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME installed_c_abi COMMAND abi_consumer)
