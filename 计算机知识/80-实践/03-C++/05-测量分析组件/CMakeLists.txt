cmake_minimum_required(VERSION 3.20)
project(WWLMeasurement VERSION 1.2.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
set(WWL_CPP_LABS "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Existing C++ practice source root")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(measurement STATIC src/measurement.cpp src/report.cpp src/report_queue.cpp)
add_library(WWL::measurement ALIAS measurement)
target_compile_features(measurement PUBLIC cxx_std_20)
set_target_properties(measurement PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(measurement PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(measurement PRIVATE
  "${WWL_CPP_LABS}/02-RAII与张量缓冲区" "${WWL_CPP_LABS}/04-布局缓存与并行归约")
target_link_libraries(measurement PUBLIC Threads::Threads)
function(options target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
  endif()
endfunction()
options(measurement)
add_executable(measurement_cli src/cli.cpp)
target_link_libraries(measurement_cli PRIVATE measurement)
options(measurement_cli)
add_executable(measurement_report src/report_cli.cpp)
target_link_libraries(measurement_report PRIVATE measurement)
options(measurement_report)
add_executable(measurement_batch src/batch_cli.cpp)
target_link_libraries(measurement_batch PRIVATE measurement)
options(measurement_batch)
add_executable(report_benchmark src/report_benchmark.cpp)
target_link_libraries(report_benchmark PRIVATE measurement)
options(report_benchmark)
add_executable(first_measurement stage0/first_measurement.cpp)
target_compile_features(first_measurement PRIVATE cxx_std_20)
options(first_measurement)
if(BUILD_TESTING)
  add_executable(measurement_tests tests/tests.cpp)
  target_link_libraries(measurement_tests PRIVATE measurement)
  target_include_directories(measurement_tests PRIVATE "${WWL_CPP_LABS}/04-布局缓存与并行归约")
  options(measurement_tests)
  add_test(NAME contracts COMMAND measurement_tests)
  add_executable(queue_tests tests/queue_tests.cpp)
  target_link_libraries(queue_tests PRIVATE measurement)
  options(queue_tests)
  add_test(NAME bounded_queue_contracts COMMAND queue_tests)
  set_tests_properties(bounded_queue_contracts PROPERTIES TIMEOUT 20)
  add_test(NAME batch_cli COMMAND ${CMAKE_COMMAND}
    -DCLI=$<TARGET_FILE:measurement_batch> -DDATA=${CMAKE_CURRENT_SOURCE_DIR}/data
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/batch_cli.cmake)
  add_executable(report_tests tests/report_tests.cpp)
  target_link_libraries(report_tests PRIVATE measurement)
  options(report_tests)
  add_test(NAME window_report_contracts COMMAND report_tests)
  add_test(NAME report_cli COMMAND ${CMAKE_COMMAND}
    -DCLI=$<TARGET_FILE:measurement_report> -DDATA=${CMAKE_CURRENT_SOURCE_DIR}/data
    -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/report-test.txt -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/report_cli.cmake)
  add_test(NAME cli_valid COMMAND measurement_cli "${CMAKE_CURRENT_SOURCE_DIR}/data/codes.txt")
  set_tests_properties(cli_valid PROPERTIES PASS_REGULAR_EXPRESSION "count=4 sum_mV=20 mean_mV=5 population_variance_mV2=5 rms_squared_mV2=30")
  add_test(NAME cli_empty COMMAND measurement_cli "${CMAKE_CURRENT_SOURCE_DIR}/data/empty.txt")
  set_tests_properties(cli_empty PROPERTIES PASS_REGULAR_EXPRESSION "count=0; no statistics")
  add_test(NAME cli_errors_and_first_program COMMAND ${CMAKE_COMMAND}
    -DCLI=$<TARGET_FILE:measurement_cli> -DFIRST=$<TARGET_FILE:first_measurement>
    -DDATA=${CMAKE_CURRENT_SOURCE_DIR}/data -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_errors.cmake)
endif()
install(TARGETS measurement EXPORT WWLMeasurementTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} PATTERN "detail" EXCLUDE)
install(EXPORT WWLMeasurementTargets FILE WWLMeasurementTargets.cmake NAMESPACE WWL::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WWLMeasurement)
configure_package_config_file(cmake/WWLMeasurementConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/WWLMeasurementConfig.cmake"
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WWLMeasurement)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/WWLMeasurementConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/WWLMeasurementConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/WWLMeasurementConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WWLMeasurement)
