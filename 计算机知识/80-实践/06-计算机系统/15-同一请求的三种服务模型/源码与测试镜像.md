"""Three deliberately small loopback line servers with one shared contract.

This is teaching code, not a framework.  It never listens outside 127.0.0.1,
uses no external service, and each runner has a deadline and a finite accept
budget.  ``serve_*`` returns after that budget or deadline and closes every
socket it owns.
"""

from __future__ import annotations

import queue
import selectors
import socket
import threading
import time
from collections import deque
from dataclasses import dataclass, field
from typing import Callable


HOST = "127.0.0.1"
MAX_LINE_BYTES = 64  # Includes the terminating LF; the payload is at most 63 B.
READ_CHUNK_BYTES = 32
MAX_INFLIGHT_LINES = 3
MAX_OUTPUT_BYTES = MAX_INFLIGHT_LINES * (MAX_LINE_BYTES + 3)
BUSY = b"ERR busy\n"


def decode_line(wire: bytes) -> bytes:
    """Return the one response for a complete, LF-terminated wire line.

    A line is valid only when its complete UTF-8 wire representation is at
    most 64 bytes including LF.  Newlines delimit requests; ``CR`` remains an
    ordinary payload byte.  The response is ASCII, so a bad input never makes
    the server's reply undecodable.
    """
    assert wire.endswith(b"\n")
    if len(wire) > MAX_LINE_BYTES:
        return b"ERR line-too-long\n"
    try:
        text = wire[:-1].decode("utf-8", "strict")
    except UnicodeDecodeError:
        return b"ERR utf8\n"
    return b"OK " + text.encode("utf-8") + b"\n"


@dataclass
class LineState:
    pending: bytearray = field(default_factory=bytearray)
    output: bytearray = field(default_factory=bytearray)
    frame_remaining: deque[int] = field(default_factory=deque)
    read_eof: bool = False
    rejected: bool = False

    @property
    def read_paused(self) -> bool:
        return len(self.frame_remaining) >= MAX_INFLIGHT_LINES

    def _queue(self, response: bytes) -> None:
        assert len(self.output) + len(response) <= MAX_OUTPUT_BYTES
        self.output.extend(response)
        self.frame_remaining.append(len(response))

    def written(self, count: int) -> None:
        """Record a prefix sent by the socket and reopen reads after a frame."""
        del self.output[:count]
        while count:
            remaining = self.frame_remaining[0]
            used = min(count, remaining)
            count -= used
            if used == remaining:
                self.frame_remaining.popleft()
            else:
                self.frame_remaining[0] -= used

    def receive(self, data: bytes) -> None:
        """Consume bytes incrementally; all complete lines become responses."""
        if self.read_eof or self.rejected:
            return
        if len(data) > READ_CHUNK_BYTES:
            raise ValueError("caller must retain unread bytes when the read budget is exhausted")
        if self.read_paused:
            raise RuntimeError("read attempted while this connection is backpressured")
        self.pending.extend(data)
        while True:
            if self.read_paused:
                return
            newline = self.pending.find(b"\n")
            if newline < 0:
                if len(self.pending) >= MAX_LINE_BYTES:
                    self._queue(b"ERR line-too-long\n")
                    self.pending.clear()
                    self.rejected = True
                return
            wire = bytes(self.pending[: newline + 1])
            del self.pending[: newline + 1]
            self._queue(decode_line(wire))

    def eof(self) -> None:
        self.read_eof = True
        if self.pending and not self.rejected:
            self._queue(b"ERR eof-partial\n")
            self.pending.clear()


def _drain(conn: socket.socket, state: LineState, deadline: float) -> None:
    """Blocking connection owner used by the iterative and worker models."""
    conn.settimeout(max(0.01, deadline - time.monotonic()))
    try:
        while not state.read_eof and time.monotonic() < deadline:
            if state.output:
                sent = conn.send(state.output)
                state.written(sent)
                if not state.output and not state.read_paused:
                    state.receive(b"")
                continue
            if state.pending and not state.read_paused:
                state.receive(b"")
                if state.output:
                    continue
            try:
                data = conn.recv(READ_CHUNK_BYTES)
            except socket.timeout:
                break
            if not data:
                state.eof()
                break
            state.receive(data)
        while state.output and time.monotonic() < deadline:
            sent = conn.send(state.output)
            state.written(sent)
            if not state.output and not state.read_paused:
                state.receive(b"")
    finally:
        try:
            conn.shutdown(socket.SHUT_WR)
        except OSError:
            pass
        conn.close()


def _listener() -> socket.socket:
    listener = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    listener.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    listener.bind((HOST, 0))
    listener.listen()
    listener.settimeout(0.02)
    return listener


def serve_iterative(
    publish: Callable[[int], None], *, max_connections: int, timeout: float = 2.0
) -> None:
    """One connection owner at a time.  A slow connection delays the next one."""
    deadline = time.monotonic() + timeout
    listener = _listener()
    publish(listener.getsockname()[1])
    accepted = 0
    try:
        while accepted < max_connections and time.monotonic() < deadline:
            try:
                conn, _ = listener.accept()
            except socket.timeout:
                continue
            accepted += 1
            _drain(conn, LineState(), deadline)
    finally:
        listener.close()


def serve_prethreaded(
    publish: Callable[[int], None], *, max_connections: int, workers: int = 2,
    capacity: int = 2, timeout: float = 2.0
) -> None:
    """A finite admission queue.  Full means a visible `ERR busy`, never growth."""
    deadline = time.monotonic() + timeout
    listener = _listener()
    jobs: queue.Queue[socket.socket | None] = queue.Queue(maxsize=capacity)
    done = threading.Event()

    def worker() -> None:
        while True:
            item = jobs.get()
            try:
                if item is None:
                    return
                _drain(item, LineState(), deadline)
            finally:
                jobs.task_done()

    threads = [threading.Thread(target=worker, daemon=True) for _ in range(workers)]
    for thread in threads:
        thread.start()
    publish(listener.getsockname()[1])
    accepted = 0
    try:
        while accepted < max_connections and time.monotonic() < deadline:
            try:
                conn, _ = listener.accept()
            except socket.timeout:
                continue
            accepted += 1
            try:
                jobs.put_nowait(conn)
            except queue.Full:
                try:
                    conn.sendall(BUSY)
                    conn.shutdown(socket.SHUT_WR)
                except OSError:
                    pass
                conn.close()
    finally:
        listener.close()
        # Workers have a finite socket timeout, so each sentinel is eventually read.
        for _ in threads:
            while time.monotonic() < deadline:
                try:
                    jobs.put(None, timeout=0.02)
                    break
                except queue.Full:
                    continue
        done.set()
        for thread in threads:
            thread.join(timeout=max(0.0, deadline - time.monotonic()) + 0.1)
        if any(thread.is_alive() for thread in threads):
            raise RuntimeError("worker did not stop before the service deadline")


@dataclass
class EventConnection:
    sock: socket.socket
    generation: int
    state: LineState = field(default_factory=LineState)


class GenerationTable:
    """A deterministic ownership model for stale readiness after FD-number reuse."""

    def __init__(self) -> None:
        self._next = 0
        self._live: dict[int, int] = {}

    def open(self, fd: int) -> int:
        self._next += 1
        self._live[fd] = self._next
        return self._next

    def owns(self, fd: int, generation: int) -> bool:
        return self._live.get(fd) == generation

    def close(self, fd: int, generation: int) -> bool:
        if not self.owns(fd, generation):
            return False
        del self._live[fd]
        return True


def serve_selectors(
    publish: Callable[[int], None], *, max_connections: int, timeout: float = 2.0
) -> None:
    """One-thread nonblocking model with per-connection state and a read budget.

    ``selectors`` chooses the OS mechanism (kqueue on many macOS builds,
    epoll on Linux, etc.).  It is not a claim that every readiness API shares
    identical edge-trigger or cancellation semantics.
    """
    deadline = time.monotonic() + timeout
    selector = selectors.DefaultSelector()
    listener = _listener()
    listener.setblocking(False)
    generations = GenerationTable()
    accepted = 0

    def close(event: EventConnection) -> None:
        fd = event.sock.fileno()
        if not generations.close(fd, event.generation):
            return
        try:
            selector.unregister(event.sock)
        except Exception:
            pass
        event.sock.close()

    selector.register(listener, selectors.EVENT_READ, data=None)
    publish(listener.getsockname()[1])
    try:
        while time.monotonic() < deadline:
            if accepted >= max_connections and len(selector.get_map()) == 1:
                return
            for key, mask in selector.select(timeout=0.02):
                if key.data is None:
                    if accepted >= max_connections:
                        continue
                    conn, _ = listener.accept()
                    conn.setblocking(False)
                    event = EventConnection(conn, generations.open(conn.fileno()))
                    selector.register(conn, selectors.EVENT_READ, data=event)
                    accepted += 1
                    continue
                event: EventConnection = key.data
                if not generations.owns(event.sock.fileno(), event.generation):
                    continue
                if mask & selectors.EVENT_READ:
                    # One recv per readiness turn is an explicit fairness budget.
                    try:
                        data = event.sock.recv(READ_CHUNK_BYTES)
                    except BlockingIOError:
                        data = None
                    if data == b"":
                        event.state.eof()
                    elif data:
                        event.state.receive(data)
                    if event.state.output:
                        selector.modify(event.sock, selectors.EVENT_WRITE, event)
                if mask & selectors.EVENT_WRITE and event.state.output:
                    try:
                        sent = event.sock.send(event.state.output)
                    except BlockingIOError:
                        sent = 0
                    event.state.written(sent)
                    if not event.state.output and not event.state.read_eof:
                        event.state.receive(b"")
                        if event.state.output:
                            selector.modify(event.sock, selectors.EVENT_WRITE, event)
                        else:
                            selector.modify(event.sock, selectors.EVENT_READ, event)
                if event.state.read_eof and not event.state.output:
                    close(event)
    finally:
        for key in list(selector.get_map().values()):
            try:
                selector.unregister(key.fileobj)
            except Exception:
                pass
            try:
                key.fileobj.close()
            except OSError:
                pass
        selector.close()

import socket
import threading
import unittest
import re
from pathlib import Path

from line_servers import (
    BUSY,
    GenerationTable,
    LineState,
    MAX_INFLIGHT_LINES,
    MAX_OUTPUT_BYTES,
    READ_CHUNK_BYTES,
    serve_iterative,
    serve_prethreaded,
    serve_selectors,
)


def start(target, **kwargs):
    ready = threading.Event()
    port = []

    def publish(value):
        port.append(value)
        ready.set()

    thread = threading.Thread(target=target, args=(publish,), kwargs=kwargs)
    thread.start()
    if not ready.wait(1):
        raise AssertionError("server never published its loopback port")
    return port[0], thread


def exchange(port, parts):
    with socket.create_connection(("127.0.0.1", port), timeout=1) as client:
        for part in parts:
            client.sendall(part)
        client.shutdown(socket.SHUT_WR)
        result = bytearray()
        while True:
            chunk = client.recv(1024)
            if not chunk:
                return bytes(result)
            result.extend(chunk)


class ProtocolTests(unittest.TestCase):
    def test_fragmented_and_coalesced_lines_share_one_contract(self):
        state = LineState()
        state.receive(b"one")
        state.receive(b"\ntwo\n")
        state.eof()
        self.assertEqual(state.output, b"OK one\nOK two\n")

    def test_half_close_flushes_response_and_partial_line_is_an_error(self):
        state = LineState()
        state.receive(b"answer\nleft")
        state.eof()
        self.assertEqual(state.output, b"OK answer\nERR eof-partial\n")

    def test_overlimit_and_bad_utf8_are_separate_errors(self):
        exact = LineState()
        exact.receive(b"x" * READ_CHUNK_BYTES)
        exact.receive(b"x" * (63 - READ_CHUNK_BYTES) + b"\n")
        self.assertEqual(exact.output, b"OK " + b"x" * 63 + b"\n")
        over = LineState()
        over.receive(b"x" * READ_CHUNK_BYTES)
        over.receive(b"x" * READ_CHUNK_BYTES)
        self.assertEqual(over.output, b"ERR line-too-long\n")
        bad = LineState()
        bad.receive(bytes([0xff, 0x0a]))
        self.assertEqual(bad.output, b"ERR utf8\n")

    def test_slow_reader_backpressures_after_three_inflight_frames(self):
        state = LineState()
        state.receive(b"x\n" * (MAX_INFLIGHT_LINES + 1))
        self.assertTrue(state.read_paused)
        self.assertEqual(len(state.frame_remaining), MAX_INFLIGHT_LINES)
        self.assertLessEqual(len(state.output), MAX_OUTPUT_BYTES)
        with self.assertRaises(RuntimeError):
            state.receive(b"next\n")
        state.written(len(b"OK x\n"))
        self.assertFalse(state.read_paused)
        state.receive(b"")
        self.assertTrue(state.read_paused)
        self.assertLessEqual(len(state.output), MAX_OUTPUT_BYTES)

    def test_generation_model_rejects_a_stale_event_after_fd_reuse(self):
        table = GenerationTable()
        old = table.open(9)
        self.assertTrue(table.close(9, old))
        new = table.open(9)
        self.assertNotEqual(old, new)
        self.assertFalse(table.owns(9, old))
        self.assertTrue(table.owns(9, new))

    def test_source_mirror_matches_both_executable_sources(self):
        directory = Path(__file__).resolve().parent
        mirror = (directory / "源码与测试镜像.md").read_text(encoding="utf-8")
        for name in ("line_servers.py", "test_line_servers.py"):
            match = re.search(
                rf"<!-- code-sync: \./{re.escape(name)} -->\n"
                rf"~~~python\n(.*?)~~~\n",
                mirror,
                flags=re.DOTALL,
            )
            self.assertIsNotNone(match, name)
            self.assertEqual(match.group(1), (directory / name).read_text(encoding="utf-8"))


class ServerTests(unittest.TestCase):
    def _assert_common_io(self, target):
        port, thread = start(target, max_connections=1)
        self.assertEqual(exchange(port, [b"a", b"\nb\n"]), b"OK a\nOK b\n")
        thread.join(2)
        self.assertFalse(thread.is_alive())

    def _assert_packed_half_close_resumes_pending_line(self, target):
        port, thread = start(target, max_connections=1)
        self.assertEqual(exchange(port, [b"x\n" * 4]), b"OK x\n" * 4)
        thread.join(2)
        self.assertFalse(thread.is_alive())

    def test_iterative_common_contract(self):
        self._assert_common_io(serve_iterative)

    def test_iterative_packed_half_close(self):
        self._assert_packed_half_close_resumes_pending_line(serve_iterative)

    def test_prethreaded_common_contract(self):
        self._assert_common_io(serve_prethreaded)

    def test_prethreaded_packed_half_close(self):
        self._assert_packed_half_close_resumes_pending_line(serve_prethreaded)

    def test_selectors_common_contract(self):
        self._assert_common_io(serve_selectors)

    def test_selectors_packed_half_close(self):
        self._assert_packed_half_close_resumes_pending_line(serve_selectors)

    def test_prethreaded_queue_pressure_is_explicit_rejection(self):
        port, thread = start(serve_prethreaded, max_connections=3, workers=1, capacity=1)
        first = socket.create_connection(("127.0.0.1", port), timeout=1)
        second = socket.create_connection(("127.0.0.1", port), timeout=1)
        third = socket.create_connection(("127.0.0.1", port), timeout=1)
        self.assertEqual(third.recv(1024), BUSY)
        third.close()
        first.shutdown(socket.SHUT_WR)
        self.assertEqual(first.recv(1024), b"")
        first.close()
        second.shutdown(socket.SHUT_WR)
        self.assertEqual(second.recv(1024), b"")
        second.close()
        thread.join(3)
        self.assertFalse(thread.is_alive())


if __name__ == "__main__":
    unittest.main()
