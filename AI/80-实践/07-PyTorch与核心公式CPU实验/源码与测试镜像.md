"""离线 CPU 横线/竖线分类教学。数据全为合成，不下载图像或预训练权重。"""
from __future__ import annotations
import argparse
import copy
import json
import math
import tempfile
from pathlib import Path
import torch
from torch import nn
from torch.utils.data import DataLoader, TensorDataset


def validate_data_config(noise: float, position_min: int, position_max: int) -> None:
    if not math.isfinite(noise) or noise < 0:
        raise ValueError('noise 必须是非负有限数')
    if type(position_min) is not int or type(position_max) is not int or not (0 <= position_min <= position_max < 16):
        raise ValueError('位置须满足整数 0 <= min <= max < 16，端点均包含')


def make_data(count: int, seed: int, noise: float = 0.15,
              position_min: int = 3, position_max: int = 12) -> TensorDataset:
    validate_data_config(noise, position_min, position_max)
    if type(count) is not int or count < 1:
        raise ValueError('count 必须是正整数')
    g = torch.Generator().manual_seed(seed)
    labels = torch.arange(count) % 2
    x = torch.randn(count, 1, 16, 16, generator=g)*noise
    positions = torch.randint(position_min, position_max + 1, (count,), generator=g)
    for i, (label, position) in enumerate(zip(labels.tolist(), positions.tolist())):
        if label == 0:
            x[i, 0, position, 2:14] += 1.
        else:
            x[i, 0, 2:14, position] += 1.
    if not torch.isfinite(x).all():
        raise ValueError('生成的 FP32 像素非有限；减小noise')
    return TensorDataset(x, labels)


def build_model() -> nn.Module:
    return nn.Sequential(nn.Conv2d(1, 8, 3, padding=1), nn.ReLU(),
                         nn.Conv2d(8, 16, 3, padding=1), nn.ReLU(),
                         nn.AdaptiveAvgPool2d(1), nn.Flatten(), nn.Linear(16, 2))


@torch.no_grad()
def evaluate(model: nn.Module, dataset: TensorDataset) -> dict[str, float]:
    model.eval()
    logits = model(dataset.tensors[0])
    labels = dataset.tensors[1]
    return {'loss': float(nn.functional.cross_entropy(logits, labels)),
            'accuracy': float((logits.argmax(-1) == labels).float().mean())}


def write_visuals(report: dict, output: Path) -> None:
    history = report['history']
    maximum = max(max(row['train_loss'], row['validation']['loss']) for row in history)
    scale = maximum if maximum > 0 else 1.
    curves = []
    for role, color in [('train_loss', '#2255aa'), ('validation', '#aa4422')]:
        points = []
        for index, row in enumerate(history):
            loss = row[role] if role == 'train_loss' else row[role]['loss']
            x = 50 + 540 * index / max(1, len(history) - 1)
            y = 270 - 220 * loss / scale
            points.append(f'{x:.2f},{y:.2f}')
        curves.append(f'<polyline fill="none" stroke="{color}" stroke-width="2" points="{" ".join(points)}"/>')
        for point in points:
            px, py = point.split(',')
            curves.append(f'<circle cx="{px}" cy="{py}" r="2" fill="{color}"/>')
    svg = ('<svg xmlns="http://www.w3.org/2000/svg" width="640" height="340" viewBox="0 0 640 340">'
           '<rect width="640" height="340" fill="white"/><g font-family="sans-serif" font-size="14">'
           '<text x="40" y="25">Synthetic CNN: loss (blue=train, red=validation)</text>'
           f'<text x="10" y="55">{scale:.3g}</text><text x="20" y="275">0</text>'
           '<path d="M50 50 V270 H590" fill="none" stroke="black"/>'
           '<text x="50" y="300">epoch 1</text>'
           f'<text x="520" y="300">epoch {len(history)}</text>' + ''.join(curves) + '</g></svg>')
    (output / 'history.svg').write_text(svg, encoding='utf-8')
    tiles = []
    samples = report['samples'] + report['error_samples']
    for index, sample in enumerate(samples):
        left, top = (index % 4) * 180 + 10, (index // 4) * 195 + 30
        tiles.append(f'<text x="{left}" y="{top-8}">i={sample["index"]} y={sample["label"]} p={sample["prediction"]}</text>')
        for row, pixels in enumerate(sample['pixels']):
            for col, value in enumerate(pixels):
                # Display mapping only; raw, unclipped values stay in metrics.json.
                gray = round(255 * min(1., max(0., value)))
                tiles.append(f'<rect x="{left+col*10}" y="{top+row*10}" width="10" height="10" fill="rgb({gray},{gray},{gray})"/>')
    height = 30 + 195 * ((len(samples) + 3) // 4)
    (output / 'samples.svg').write_text(
        f'<svg xmlns="http://www.w3.org/2000/svg" width="730" height="{height}">'
        f'<rect width="730" height="{height}" fill="white"/><g font-family="sans-serif" font-size="13">'
        + ''.join(tiles) + '</g></svg>', encoding='utf-8')


def run(epochs: int, output: Path, *, noise: float = 0.15,
        position_min: int = 3, position_max: int = 12, seed: int = 2026,
        test_noise: float | None = None, test_position_min: int | None = None,
        test_position_max: int | None = None) -> dict:
    if epochs < 1:
        raise ValueError('epochs 必须 >= 1')
    validate_data_config(noise, position_min, position_max)
    if type(seed) is not int or not (2015 <= seed < 2**63):
        raise ValueError('seed 必须是 [2015, 2**63) 中的整数')
    test_config = (noise if test_noise is None else test_noise,
                   position_min if test_position_min is None else test_position_min,
                   position_max if test_position_max is None else test_position_max)
    validate_data_config(*test_config)
    torch.manual_seed(seed)
    torch.set_num_threads(1)
    # 默认仍为 train=11、valid=22、test=33、shuffle=44。
    split_seeds = {'train': seed - 2015, 'validation': seed - 2004, 'test': seed - 1993}
    train = make_data(256, split_seeds['train'], noise, position_min, position_max)
    valid = make_data(128, split_seeds['validation'], noise, position_min, position_max)
    test = make_data(128, split_seeds['test'], *test_config)
    loader = DataLoader(train, batch_size=32, shuffle=True,
                        generator=torch.Generator().manual_seed(seed - 1982), num_workers=0)
    model = build_model()
    optimizer = torch.optim.AdamW(model.parameters(), lr=0.01, weight_decay=0.001)
    history = []; best_loss = float('inf'); best = None; best_epoch = None
    for epoch in range(epochs):
        model.train(); loss_sum = 0.
        for x, y in loader:
            optimizer.zero_grad(set_to_none=True)
            loss = nn.functional.cross_entropy(model(x), y)
            loss.backward(); optimizer.step()
            loss_sum += float(loss.detach())*len(y)
        metrics = evaluate(model, valid)
        history.append({'epoch': epoch+1, 'train_loss': loss_sum/len(train), 'validation': metrics})
        if metrics['loss'] < best_loss:
            best_loss = metrics['loss']; best = copy.deepcopy(model.state_dict()); best_epoch = epoch+1
    assert best is not None
    model.load_state_dict(best)
    output.mkdir(parents=True, exist_ok=True)
    checkpoint = output/'model.pt'
    torch.save(best, checkpoint)
    restored = build_model()
    restored.load_state_dict(torch.load(checkpoint, map_location='cpu', weights_only=True))
    model.eval(); restored.eval()
    with torch.no_grad():
        torch.testing.assert_close(model(test.tensors[0]), restored(test.tensors[0]), atol=0, rtol=0)
    report = {'task': 'synthetic_horizontal_vertical_lines', 'device': 'cpu',
              'torch': torch.__version__, 'seed': seed, 'epochs': epochs, 'selected_epoch': best_epoch,
              'split_sizes': {'train': 256, 'validation': 128, 'test': 128},
              'test': evaluate(restored, test), 'checkpoint_roundtrip_exact': True, 'history': history,
              'limits': '仅代表此合成分布与固定种子；不证明真实图像泛化或 GPU 性能。'}
    report['config'] = {'image_shape': [1, 16, 16], 'noise': noise,
                        'position_inclusive': [position_min, position_max],
                        'test_noise': test_config[0], 'test_position_inclusive': list(test_config[1:]),
                        'split_seeds': split_seeds, 'shuffle_seed': seed - 1982}
    with torch.no_grad():
        logits = restored(test.tensors[0])
        predictions = logits.argmax(-1)
        labels = test.tensors[1]
        errors = (predictions != labels).nonzero(as_tuple=False).flatten().tolist()
    def inspect(index: int) -> dict:
        return {'index': index, 'label': int(labels[index]),
                'prediction': int(predictions[index]), 'logits': logits[index].tolist(),
                'pixels': test.tensors[0][index, 0].tolist()}
    # 报告均在验证选模完成后生成，不能反过来用于选epoch。
    report['predictions'] = predictions.tolist()
    report['error_indices'] = errors
    report['samples'] = [inspect(i) for i in range(min(8, len(test)))]
    report['error_samples'] = [inspect(i) for i in errors[:8]]
    report['parameters'] = sum(p.numel() for p in restored.parameters())
    serialized = json.dumps(report, ensure_ascii=False, indent=2, allow_nan=False)
    (output/'metrics.json').write_text(serialized, encoding='utf-8')
    write_visuals(report, output)
    return report


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--epochs', type=int, default=10)
    parser.add_argument('--output', type=Path, help='显式保存目录；不指定时使用运行结束自动清理的临时目录')
    parser.add_argument('--noise', type=float, default=0.15)
    parser.add_argument('--position-min', type=int, default=3)
    parser.add_argument('--position-max', type=int, default=12)
    parser.add_argument('--seed', type=int, default=2026)
    parser.add_argument('--test-noise', type=float)
    parser.add_argument('--test-position-min', type=int)
    parser.add_argument('--test-position-max', type=int)
    args = parser.parse_args()
    options = dict(noise=args.noise, position_min=args.position_min,
                   position_max=args.position_max, seed=args.seed,
                   test_noise=args.test_noise, test_position_min=args.test_position_min,
                   test_position_max=args.test_position_max)
    if args.output is None:
        with tempfile.TemporaryDirectory(prefix='ai-cnn-') as temporary:
            result = run(args.epochs, Path(temporary), **options)
    else:
        result = run(args.epochs, args.output, **options)
    print(json.dumps(result, ensure_ascii=False, indent=2))

if __name__ == '__main__':
    main()

"""运行：python -m unittest discover -s . -p 'test_*.py' -v。"""
import unittest
import torch
from torch import nn
from torch.nn import functional as F
from core_checks import online_attention, cached_causal_mask, ctc_collapse

class CoreTests(unittest.TestCase):
    def setUp(self):
        torch.manual_seed(17)
        torch.set_num_threads(1)

    def test_broadcasting_trap(self):
        prediction = torch.arange(4.).reshape(4, 1)
        target = torch.arange(4.)
        self.assertEqual((prediction-target).shape, (4, 4))
        self.assertEqual((prediction-target[:, None]).square().mean().item(), 0.)

    def test_view_and_copy(self):
        x = torch.arange(6.).reshape(2, 3)
        x.T[0, 1] = 99
        self.assertEqual(x[1, 0].item(), 99.)
        y = x.clone(); y[0, 0] = -7
        self.assertEqual(x[0, 0].item(), 0.)

    def test_scalar_autograd(self):
        x, w, b = [torch.tensor(v, requires_grad=True) for v in (2., 3., 1.)]
        loss = ((w*x+b)-5).square()/2
        loss.backward()
        self.assertEqual([loss.item(), w.grad.item(), b.grad.item(), x.grad.item()],
                         [2., 4., 2., 6.])

    def test_gradcheck(self):
        x = torch.randn(3, dtype=torch.double, requires_grad=True)
        self.assertTrue(torch.autograd.gradcheck(lambda z: (z.sin()*z).sum(), (x,)))

    def test_parameter_registration(self):
        class Net(nn.Module):
            def __init__(self):
                super().__init__()
                self.layers = nn.ModuleList([nn.Linear(2, 3), nn.Linear(3, 1)])
                self.register_buffer('scale', torch.ones(1))
        model = Net()
        self.assertEqual(sum(p.numel() for p in model.parameters()), 13)
        self.assertIn('scale', model.state_dict())
        self.assertNotIn('scale', dict(model.named_parameters()))

    def test_convolution_example(self):
        x = torch.arange(1., 10.).reshape(1, 1, 3, 3)
        kernel = torch.tensor([[1., 0.], [0., -1.]]).reshape(1, 1, 2, 2)
        torch.testing.assert_close(F.conv2d(x, kernel), torch.full((1, 1, 2, 2), -4.))

    def test_convolution_shape_parameters(self):
        layer = nn.Conv2d(3, 16, kernel_size=3, stride=2, padding=1)
        self.assertEqual(layer(torch.randn(2, 3, 32, 32)).shape, (2, 16, 16, 16))
        self.assertEqual(sum(p.numel() for p in layer.parameters()), 448)

    def test_receptive_field(self):
        radius, jump = 1, 1
        values = []
        for kernel, stride in [(3, 2), (3, 2), (3, 1)]:
            radius += (kernel-1)*jump
            jump *= stride
            values.append((radius, jump))
        self.assertEqual(values, [(3, 2), (7, 4), (15, 4)])

    def test_cnn_data_config_and_seed(self):
        from train_cnn import make_data
        a, b = make_data(8, 11), make_data(8, 11)
        self.assertTrue(torch.equal(a.tensors[0], b.tensors[0]))
        exact = make_data(2, 11, noise=0, position_min=4, position_max=4)
        self.assertEqual(exact.tensors[0][0, 0, 4, 2:14].tolist(), [1.] * 12)
        self.assertEqual(exact.tensors[0][1, 0, 2:14, 4].tolist(), [1.] * 12)
        for kwargs in [dict(noise=float('nan')), dict(noise=float('inf')),
                       dict(noise=-1), dict(position_min=-1), dict(position_max=16),
                       dict(position_min=10, position_max=9)]:
            with self.subTest(kwargs=kwargs), self.assertRaises(ValueError):
                make_data(2, 11, **kwargs)

    def test_cnn_shapes_and_parameter_ledger(self):
        from train_cnn import build_model
        model = build_model()
        self.assertEqual([sum(p.numel() for p in model[i].parameters()) for i in [0, 2, 6]],
                         [80, 1168, 34])
        x = torch.zeros(1, 1, 16, 16)
        shapes, byte_counts = [list(x.shape)], [x.numel() * x.element_size()]
        for i, layer in enumerate(model):
            x = layer(x)
            if i in [0, 2, 4, 6]:
                shapes.append(list(x.shape)); byte_counts.append(x.numel() * x.element_size())
        self.assertEqual(shapes, [[1, 1, 16, 16], [1, 8, 16, 16], [1, 16, 16, 16],
                                  [1, 16, 1, 1], [1, 2]])
        self.assertEqual(byte_counts, [1024, 8192, 16384, 64, 8])

    def test_cnn_visual_report_interface(self):
        import tempfile
        from pathlib import Path
        import xml.etree.ElementTree as ET
        from train_cnn import write_visuals
        sample = {'index': 0, 'label': 0, 'prediction': 1, 'pixels': [[0.] * 16 for _ in range(16)]}
        report = {'history': [{'epoch': 1, 'train_loss': .7, 'validation': {'loss': .8}},
                              {'epoch': 2, 'train_loss': .4, 'validation': {'loss': .6}}],
                  'samples': [sample], 'error_samples': [sample]}
        with tempfile.TemporaryDirectory(prefix='cnn-visual-fixture-') as directory:
            output = Path(directory)
            write_visuals(report, output)
            for name in ['history.svg', 'samples.svg']:
                root = ET.fromstring((output / name).read_text())
                self.assertEqual(root.tag, '{http://www.w3.org/2000/svg}svg')
            history = (output / 'history.svg').read_text()
            self.assertIn('blue=train, red=validation', history)
            self.assertIn('epoch 2', history)
            self.assertIn('y=0 p=1', (output / 'samples.svg').read_text())

    def test_residual_jacobian(self):
        x = torch.tensor(2., requires_grad=True)
        y = x+x*x
        y.backward()
        self.assertEqual(x.grad.item(), 5.)

    def test_online_attention(self):
        q, k, v = (torch.randn(*shape, dtype=torch.double) for shape in
                    [(4, 5), (11, 5), (11, 7)])
        expected = (q @ k.T/5**0.5).softmax(-1) @ v
        for block in [1, 3, 20]:
            torch.testing.assert_close(online_attention(q, k, v, block), expected,
                                       rtol=1e-11, atol=1e-11)

    def test_online_attention_large_logits(self):
        q = torch.tensor([[1000., -1000.]], dtype=torch.double)
        k = torch.tensor([[10., 1.], [9., -1.], [-10., 2.]], dtype=torch.double)
        v = torch.randn(3, 4, dtype=torch.double)
        result = online_attention(q, k, v, 2)
        self.assertTrue(torch.isfinite(result).all())
        torch.testing.assert_close(result, (q @ k.T/2**0.5).softmax(-1) @ v)

    def test_online_attention_invalid(self):
        with self.assertRaises(ValueError):
            online_attention(torch.ones(2, 3), torch.ones(4, 3), torch.ones(4, 1), 0)

    def test_offset_causal_mask(self):
        expected = torch.tensor([[1, 1, 1, 1, 1, 1, 0], [1, 1, 1, 1, 1, 1, 1]], dtype=torch.bool)
        self.assertTrue(torch.equal(cached_causal_mask(5, 2), expected))

    def test_cached_attention_matches_full(self):
        q, k, v = [torch.randn(1, 2, 7, 4, dtype=torch.double) for _ in range(3)]
        full = F.scaled_dot_product_attention(q, k, v, dropout_p=0., is_causal=True)
        cached = F.scaled_dot_product_attention(q[:, :, 5:], k, v,
                     attn_mask=cached_causal_mask(5, 2), dropout_p=0., is_causal=False)
        torch.testing.assert_close(cached, full[:, :, 5:], atol=1e-12, rtol=1e-12)

    def test_mla_absorption(self):
        c, up_k, up_v, q = [torch.randn(*shape, dtype=torch.double) for shape in
                            [(7, 3), (5, 3), (4, 3), (2, 5)]]
        weights = (q @ (c @ up_k.T).T/5**0.5).softmax(-1)
        expanded = weights @ (c @ up_v.T)
        absorbed_weights = ((q @ up_k) @ c.T/5**0.5).softmax(-1)
        absorbed = (absorbed_weights @ c) @ up_v.T
        torch.testing.assert_close(expanded, absorbed, atol=1e-12, rtol=1e-12)

    def test_global_token_average(self):
        sums, counts = [8., 6.], [8, 2]
        global_mean = sum(sums)/sum(counts)
        mean_of_means = sum(s/n for s, n in zip(sums, counts))/2
        scaled_ddp = sum(2*s/sum(counts) for s in sums)/2
        self.assertAlmostEqual(global_mean, 1.4)
        self.assertAlmostEqual(mean_of_means, 2.)
        self.assertAlmostEqual(scaled_ddp, global_mean)

    def test_ctc_order(self):
        self.assertEqual(ctc_collapse([1, 1, 0, 1]), [1, 1])
        self.assertEqual(ctc_collapse([0, 0, 1, 2, 2, 0]), [1, 2])

    def test_diffusion_identity(self):
        x0 = torch.randn(8, dtype=torch.double)
        eps = torch.randn_like(x0)
        alpha_bar = torch.tensor(0.64, dtype=torch.double)
        xt = alpha_bar.sqrt()*x0 + (1-alpha_bar).sqrt()*eps
        reconstructed = (xt-(1-alpha_bar).sqrt()*eps)/alpha_bar.sqrt()
        torch.testing.assert_close(reconstructed, x0)

    def test_flow_path(self):
        noise, data = torch.randn(3), torch.randn(3)
        velocity = data-noise
        torch.testing.assert_close(noise+velocity, data)
        # 这里只验证配对条件路径，不代表学到的边缘速度场是一条直线。

    def test_cross_entropy_logits(self):
        logits = torch.tensor([[2., 1., -1.]], dtype=torch.double)
        target = torch.tensor([0])
        torch.testing.assert_close(F.cross_entropy(logits, target),
                                   -logits.log_softmax(-1)[0, 0])

if __name__ == '__main__':
    unittest.main()
