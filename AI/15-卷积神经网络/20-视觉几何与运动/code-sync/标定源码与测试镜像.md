"""Planar multi-view initialization and joint square-pixel k1 calibration.

Board-to-camera P_C = R_CB q_B + t_CB. Lengths metres, image residuals pixels,
rotvecs radians, radial coefficients in normalized camera coordinates. Uses the
published pinhole and Hartley conditioning interfaces; synthetic points only.
"""
from pathlib import Path
import argparse
import json
import numpy as np
from scipy.optimize import least_squares
from scipy.spatial.transform import Rotation
import scipy
import stereo_lab as stereo
from pose_estimation_lab import hartley


def board_points(columns, rows, spacing_m):
    spacing=stereo.positive(spacing_m,'spacing_m')
    if any(isinstance(v,bool) or not isinstance(v,int) or v<2 for v in [columns,rows]):
        raise ValueError('board grid needs at least 2 rows and 2 columns')
    return np.array([[(i-(columns-1)/2)*spacing,(j-(rows-1)/2)*spacing,0.]
                     for j in range(rows) for i in range(columns)])


def board(value):
    points=stereo.batch(value,3)
    if len(points)<4 or not np.all(points[:,2]==0):
        raise ValueError('expected at least four points in board Z=0 plane')
    if np.linalg.matrix_rank(np.column_stack((points[:,:2],np.ones(len(points)))))<3:
        raise ValueError('board points are collinear')
    return points


def camera_points(points,rvec,tvec):
    rv,t=np.asarray(rvec,dtype=float),np.asarray(tvec,dtype=float)
    if rv.shape!=(3,) or t.shape!=(3,) or not np.isfinite(rv).all() or not np.isfinite(t).all():
        raise ValueError('rotvec and translation must be finite (3,) arrays')
    return stereo.batch(points,3)@Rotation.from_rotvec(rv).as_matrix().T+t


def radial_project(points,rvec,tvec,f,c,k1,k2=0.):
    if not np.isfinite([k1,k2]).all():raise ValueError('nonfinite radial coefficients')
    xyz=camera_points(points,rvec,tvec)
    normal=stereo.project(xyz,1.,[0.,0.])
    radius2=np.sum(normal**2,axis=1)
    distorted=normal*(1+k1*radius2+k2*radius2**2)[:,None]
    return stereo.batch(stereo.positive(f,'f_px')*distorted+stereo.principal(c),2)


def homography(points,uv):
    q=board(points); p=stereo.batch(uv,2)
    if len(q)!=len(p):raise ValueError('point count mismatch')
    if np.linalg.matrix_rank(np.column_stack((p,np.ones(len(p)))))<3:
        raise ValueError('target image points are collinear; no invertible plane homography')
    source,ts=hartley(np.column_stack((q[:,:2],np.ones(len(q)))))
    target,ti=hartley(np.column_stack((p,np.ones(len(p)))))
    rows=[]
    for a,b in zip(source,target):
        rows.extend([np.r_[a,np.zeros(3),-b[0]*a],np.r_[np.zeros(3),a,-b[1]*a]])
    matrix=np.asarray(rows)
    _,sv,vh=np.linalg.svd(matrix,full_matrices=True)
    if sv[7]/sv[0]<1e-10:raise ValueError('homography design lacks rank eight')
    h=np.linalg.solve(ti,vh[-1].reshape(3,3))@ts
    if abs(h[2,2])<1e-12:raise ValueError('homography normalization chart invalid')
    return h/h[2,2]


def intrinsic_from_homographies(homographies):
    if len(homographies)<3:raise ValueError('this general-K initializer needs at least 3 views')
    rows=[]
    for h in homographies:
        def vij(i,j):
            a,b=h[:,i],h[:,j]
            return np.array([a[0]*b[0],a[0]*b[1]+a[1]*b[0],a[1]*b[1],
                             a[2]*b[0]+a[0]*b[2],a[2]*b[1]+a[1]*b[2],a[2]*b[2]])
        rows.extend([vij(0,1),vij(0,0)-vij(1,1)])
    v=np.asarray(rows)
    norms=np.linalg.norm(v,axis=1)
    if np.any(norms<=1e-15):raise ValueError('zero calibration constraint')
    v=v/norms[:,None]
    scale=np.linalg.norm(v,axis=0)
    if np.any(scale<=1e-15):raise ValueError('unobserved calibration constraint column')
    _,sv,vh=np.linalg.svd(v/scale,full_matrices=True)
    ratio=sv[4]/sv[0]
    if ratio<1e-8:raise ValueError('view constraints lack rank five or are poorly conditioned')
    b=vh[-1]/scale
    metric=np.array([[b[0],b[1],b[3]],[b[1],b[2],b[4]],[b[3],b[4],b[5]]])
    k,metric,eigen=intrinsic_from_metric(metric)
    return k,{'constraint_singular_values':sv.tolist(),'sigma5_over_sigma1':float(ratio),
              'B_up_to_scale':metric.tolist(),'B_eigenvalues':eigen.tolist()}


def intrinsic_from_metric(value):
    metric=np.asarray(value,dtype=float)
    if metric.shape!=(3,3) or not np.isfinite(metric).all() or not np.allclose(metric,metric.T,atol=1e-12,rtol=0):
        raise ValueError('B must be finite symmetric (3,3)')
    metric=metric if metric[2,2]>0 else -metric
    eigen=np.linalg.eigvalsh(metric)
    if eigen[0]<=1e-12*eigen[-1]:raise ValueError('estimated B is not safely SPD; no eigenvalue repair')
    lower=np.linalg.cholesky(metric)
    k=np.linalg.inv(lower.T); k/=k[2,2]
    return k,metric,eigen


def pose_from_homography(h,k):
    columns=np.linalg.solve(k,h)
    scale=2/(np.linalg.norm(columns[:,0])+np.linalg.norm(columns[:,1]))
    if columns[2,2]<0:scale=-scale
    r1,r2=scale*columns[:,0],scale*columns[:,1]
    raw=np.column_stack((r1,r2,np.cross(r1,r2)))
    u,_,vh=np.linalg.svd(raw)
    rotation=u@np.diag([1.,1.,np.linalg.det(u@vh)])@vh
    return Rotation.from_matrix(rotation).as_rotvec(),scale*columns[:,2]


def initialize(points,images):
    q=board(points)
    pixels=np.asarray(images,dtype=float)
    if pixels.ndim!=3 or pixels.shape[1:]!=(len(q),2) or not np.isfinite(pixels).all():
        raise ValueError('images must be finite (V,N,2)')
    hs=[homography(q,image) for image in pixels]
    general,diagnostics=intrinsic_from_homographies(hs)
    f=(general[0,0]+general[1,1])/2
    c=general[:2,2]
    square=np.array([[f,0.,c[0]],[0.,f,c[1]],[0.,0.,1.]])
    poses=[pose_from_homography(h,square) for h in hs]
    return f,c,poses,{'general_K':general.tolist(),'square_initial_K':square.tolist(),
                    'homographies':[h.tolist() for h in hs],**diagnostics}


def unpack(parameters,views):
    f=np.exp(parameters[0]); c=parameters[1:3]; k1=parameters[3]
    poses=[]
    for i in range(views):
        start=4+6*i
        poses.append((parameters[start:start+3],np.r_[parameters[start+3:start+5],np.exp(parameters[start+5])]))
    return f,c,k1,poses


def calibrate(points,images):
    q=board(points)
    observed=np.asarray(images,dtype=float)
    f,c,poses,initial=initialize(q,observed)
    count=len(poses)
    parameters=[np.log(f),*c,0.]
    for rv,t in poses:
        if t[2]<=0:raise ValueError('nonforward initialization')
        parameters.extend([*rv,*t[:2],np.log(t[2])])
    parameters=np.asarray(parameters)
    radius=np.max(np.linalg.norm(q,axis=1))
    lower=[np.log(100.),-2000.,-2000.,-.5]
    upper=[np.log(4000.),2000.,2000.,.5]
    for _ in poses:
        lower.extend([-np.pi]*3+[-100*radius]*2+[np.log(1.1*radius)])
        upper.extend([np.pi]*3+[100*radius]*2+[np.log(100*radius)])
    if np.any(parameters<=lower) or np.any(parameters>=upper):raise ValueError('initialization outside declared solver box')
    def residual(parameters):
        focal,center,k1,pose_list=unpack(parameters,count)
        return np.asarray([radial_project(q,rv,t,focal,center,k1)-uv
                           for (rv,t),uv in zip(pose_list,observed)]).ravel()
    fit=least_squares(residual,parameters,method='trf',jac='3-point',x_scale='jac',loss='linear',
                      bounds=(lower,upper),ftol=1e-12,xtol=1e-12,gtol=1e-10,max_nfev=150)
    if not fit.success:raise ValueError('joint fit did not converge within finite budget')
    focal,center,k1,pose_list=unpack(fit.x,count)
    norms=np.linalg.norm(fit.jac,axis=0)
    if np.any(norms<=1e-12):raise ValueError('joint residual has an unobserved parameter column')
    sv=np.linalg.svd(fit.jac/norms,compute_uv=False)
    focal_direction=fit.jac[:,0]
    all_depth_direction=sum(fit.jac[:,4+6*i+5] for i in range(count))
    cosine=(focal_direction@all_depth_direction)/(np.linalg.norm(focal_direction)*np.linalg.norm(all_depth_direction))
    predicted=np.asarray([radial_project(q,rv,t,focal,center,k1) for rv,t in pose_list])
    errors=np.linalg.norm(predicted-observed,axis=2)
    return {'f_px':float(focal),'principal_px':center.tolist(),'k1':float(k1),
      'rotvec_rad':[rv.tolist() for rv,t in pose_list],'translation_m':[t.tolist() for rv,t in pose_list],
      'predicted_px':predicted.tolist(),'residual_px':(predicted-observed).tolist(),
      'rms_point_error_px':float(np.sqrt(np.mean(errors**2))),'max_point_error_px':float(errors.max()),
      'initialization':initial,'solver':{'method':'trf','loss':'linear','jac':'3-point','x_scale':'jac',
        'nfev':fit.nfev,'status':fit.status,'optimality':float(fit.optimality),
        'column_normalized_jacobian_singular_values':sv.tolist(),
        'column_normalized_condition':float(sv[0]/sv[-1]),'cosine_logf_vs_joint_logtz':float(cosine)}}


def undistort(uv,f,c,k1,radius_max=.75):
    image=stereo.batch(uv,2)
    f,c=stereo.positive(f,'f_px'),stereo.principal(c)
    bound=stereo.positive(radius_max,'radius_max')
    if not np.isfinite(k1) or min(1.,1+3*k1*bound**2)<=0:
        raise ValueError('radial inverse is not monotone on declared radius domain')
    distorted=(image-c)/f
    rd=np.linalg.norm(distorted,axis=1)
    maximum=bound*(1+k1*bound**2)
    if np.any(rd>maximum+1e-12):raise ValueError('distorted radius outside inverse branch')
    lo=np.zeros_like(rd); hi=np.full_like(rd,bound)
    for _ in range(70):
        mid=(lo+hi)/2
        lower=mid*(1+k1*mid**2)<rd
        lo=np.where(lower,mid,lo); hi=np.where(lower,hi,mid)
    radius=(lo+hi)/2
    factor=np.divide(radius,rd,out=np.ones_like(rd),where=rd>0)
    return stereo.batch(f*distorted*factor[:,None]+c,2)


def make_fixture(spec):
    q=board_points(spec['grid_columns'],spec['grid_rows'],spec['spacing_m'])
    rvs=np.asarray(spec['training_rotvec_rad']); translations=np.asarray(spec['training_translation_m'])
    if rvs.ndim!=2 or rvs.shape[1]!=3 or rvs.shape!=translations.shape or len(rvs)<3:
        raise ValueError('fixture needs at least three equally sized rotvec/translation triples')
    f,c,k=spec['f_px'],spec['principal_px'],spec['k1']
    ideal=np.asarray([radial_project(q,rv,t,f,c,0.) for rv,t in zip(rvs,translations)])
    radial=np.asarray([radial_project(q,rv,t,f,c,k) for rv,t in zip(rvs,translations)])
    noise=np.array([[[(i*7+j*3)%11*.004-.020,(i*5+j*7)%13*.003-.018]
                    for j in range(len(q))] for i in range(len(rvs))])
    outside=np.asarray([radial_project(q,rv,t,f,c,k,spec['unmodeled_k2']) for rv,t in zip(rvs,translations)])
    hold=radial_project(q,spec['heldout_rotvec_rad'],spec['heldout_translation_m'],f,c,k)
    hold_ideal=radial_project(q,spec['heldout_rotvec_rad'],spec['heldout_translation_m'],f,c,0.)
    hold_outside=radial_project(q,spec['heldout_rotvec_rad'],spec['heldout_translation_m'],f,c,k,spec['unmodeled_k2'])
    return {'synthetic_spec':spec,'board_points_m':q.tolist(),'undistorted_px':ideal.tolist(),
      'radial_clean_px':radial.tolist(),'radial_noisy_px':(radial+noise).tolist(),
      'unmodeled_k2_px':outside.tolist(),'corner_bias_px':(radial+spec['corner_bias_px']).tolist(),
      'fixed_noise_px':noise.tolist(),'heldout_radial_px':hold.tolist(),'heldout_undistorted_px':hold_ideal.tolist(),
      'heldout_k2_px':hold_outside.tolist()}

def run(fixture):
    q=board(fixture['board_points_m']); spec=fixture['synthetic_spec']
    f,c,poses,linear=initialize(q,fixture['undistorted_px'])
    cases={}
    for label,key,holdkey in [('clean','radial_clean_px','heldout_radial_px'),
                              ('noisy','radial_noisy_px','heldout_radial_px'),
                              ('k2_mismatch','unmodeled_k2_px','heldout_k2_px'),
                              ('corner_bias','corner_bias_px','heldout_radial_px')]:
        fit=calibrate(q,fixture[key])
        predicted=radial_project(q,spec['heldout_rotvec_rad'],spec['heldout_translation_m'],fit['f_px'],fit['principal_px'],fit['k1'])
        error=np.linalg.norm(predicted-np.asarray(fixture[holdkey]),axis=1)
        fit['heldout_with_known_pose']={'predicted_px':predicted.tolist(),'rms_point_error_px':float(np.sqrt(np.mean(error**2))),
                                      'max_point_error_px':float(error.max())}
        cases[label]=fit
    frontal=[radial_project(q,[0,0,0],[i*.01,-i*.01,.6+i*.05],spec['f_px'],spec['principal_px'],0) for i in range(5)]
    try:initialize(q,frontal)
    except ValueError as error:frontal_failure=str(error)
    else:raise AssertionError('expected frontal-view rank rejection')
    unit_probe=[]
    for rv,t in zip(spec['training_rotvec_rad'],spec['training_translation_m']):
        a=radial_project(q,rv,t,spec['f_px'],spec['principal_px'],spec['k1'])
        b=radial_project(2*q,rv,2*np.asarray(t),spec['f_px'],spec['principal_px'],spec['k1'])
        unit_probe.append(float(np.max(np.abs(a-b))))
    # Inverse is checked against ideal geometry, not against itself alone.
    clean=cases['clean']; inverse=[]
    for uv,ideal in zip(fixture['radial_clean_px'],fixture['undistorted_px']):
        inverse.append(float(np.max(np.abs(undistort(uv,clean['f_px'],clean['principal_px'],clean['k1'],spec['inverse_radius_max'])-ideal))))
    return {'model':'synthetic planar board; square pixels/zero skew + k1 only',
      'versions':{'numpy':np.__version__,'scipy':scipy.__version__},'linear_ideal':linear,'cases':cases,
      'frontal_degenerate_rejection':frontal_failure,'double_board_and_translation_error_px':unit_probe,
      'clean_inverse_vs_ideal_max_component_px':inverse}


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--input',type=Path,default=Path(__file__).parent/'inputs/calibration-observations.json')
    parser.add_argument('--make-fixture',type=Path)
    parser.add_argument('--output-dir',type=Path,required=True)
    args=parser.parse_args(); args.output_dir.mkdir(parents=True,exist_ok=True)
    if args.make_fixture:
        fixture=make_fixture(json.loads(args.make_fixture.read_text(encoding='utf-8')))
        (args.output_dir/'calibration-observations.json').write_text(json.dumps(fixture,indent=2,allow_nan=False)+'\n',encoding='utf-8')
        print('fixed observations generated');return
    result=run(json.loads(args.input.read_text(encoding='utf-8')))
    (args.output_dir/'calibration-results.json').write_text(json.dumps(result,indent=2,allow_nan=False)+'\n',encoding='utf-8')
    figures(result,json.loads(args.input.read_text(encoding='utf-8')),args.output_dir)
    print(json.dumps({k:{n:v[n] for n in ['f_px','principal_px','k1','rms_point_error_px']} for k,v in result['cases'].items()}))


if __name__=='__main__':main()

"""New calibration behaviors; old stereo/two-view/pose suites stay frozen."""
from pathlib import Path
import json
import unittest
import numpy as np
import calibration_lab as c


class CalibrationChecks(unittest.TestCase):
    def setUp(self):
        self.data=json.loads((Path(__file__).parent/'inputs/calibration-observations.json').read_text())
        self.spec=self.data['synthetic_spec']; self.q=np.asarray(self.data['board_points_m'])

    def test_linear_multiview_and_spd_sign(self):
        f,center,poses,info=c.initialize(self.q,self.data['undistorted_px'])
        expected=np.array([[800.,0,320.],[0,800.,240.],[0,0,1.]])
        np.testing.assert_allclose(info['general_K'],expected,atol=1e-7,rtol=0)
        # Independent scalar Appendix-B extraction is performed by the reviewer;
        # this check also covers signed B and nonsquare/skew input support.

        inv=np.linalg.inv(k); b=inv.T@inv

        for (rv,t),true_r,true_t in zip(poses,self.spec['training_rotvec_rad'],self.spec['training_translation_m']):

            np.testing.assert_allclose(t,true_t,atol=1e-9,rtol=0)

    def test_joint_clean_truth_and_objective(self):
