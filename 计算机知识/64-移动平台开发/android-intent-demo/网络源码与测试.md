package org.wwl.networkdemo

import org.json.JSONObject
import org.wwl.intentdemo.Pending
import org.wwl.intentdemo.Protocol
import java.io.IOException
import java.net.URL
import java.net.URLEncoder
import javax.net.ssl.HttpsURLConnection

class HttpFailure(val code:Int):IOException("HTTP $code; no new business decision inferred")

/** Default platform TLS/hostname verification; app's DEBUG resource adds only its lab CA. */
class HttpsTransport(private val actor:String) {
    private val endpoint="https://127.0.0.1:18751"
    private fun token(mode:String):String = when(mode) {
        "self" -> "wwl175-public-$actor"
        "other" -> "wwl175-public-${if(actor=="D")"E" else "D"}"
        "invalid" -> "deliberately-invalid-public-fixture"
        else -> throw IllegalArgumentException("credential mode")
    }
    private fun request(path:String,mode:String,body:JSONObject?=null,drop:Boolean=false):JSONObject {
        val connection=URL(endpoint+path).openConnection() as HttpsURLConnection
        try {
            connection.instanceFollowRedirects=false
            connection.connectTimeout=1500;connection.readTimeout=2000
            connection.requestMethod=if(body==null)"GET" else "POST"
            connection.setRequestProperty("Authorization","Bearer ${token(mode)}")
            connection.setRequestProperty("Accept","application/json")
            connection.setRequestProperty("Connection","close")
            if(drop) { require(BuildConfig.DEBUG);connection.setRequestProperty("X-WWL-Lab-Drop-Reply","true") }
            if(body!=null) {
                val bytes=body.toString().toByteArray(Charsets.UTF_8)
                connection.doOutput=true
                connection.setRequestProperty("Content-Type","application/json; charset=utf-8")
                connection.setFixedLengthStreamingMode(bytes.size)
                connection.outputStream.use { it.write(bytes) }
            }
            val code=connection.responseCode
            if(code!=200)throw HttpFailure(code) // Includes 3xx; never forward Bearer to redirects.
            if(connection.contentType?.substringBefore(';')?.trim()!="application/json")throw IOException("content type")
            val bytes=connection.inputStream.use { input ->
                val output=java.io.ByteArrayOutputStream();val buffer=ByteArray(4096)
                while(output.size()<=65536) {
                    val n=input.read(buffer,0,minOf(buffer.size,65537-output.size()))
                    if(n<0)break
                    output.write(buffer,0,n)
                }
                output.toByteArray()
            }
            if(bytes.size>65536)throw IOException("response too large")
            return JSONObject(String(bytes,Charsets.UTF_8))
        } finally { connection.disconnect() }
    }
    fun snapshot(mode:String)=request("/v2/snapshot",mode)
    fun query(p:Pending,mode:String):JSONObject {
        fun encoded(s:String)=URLEncoder.encode(s,"UTF-8")
        return request("/v2/intent?intent_id=${encoded(p.id)}&fingerprint=${encoded(p.fp)}",mode)
    }
    fun reserve(p:Pending,mode:String,drop:Boolean=false):JSONObject {
        require(p.actor==actor)
        val b=JSONObject().put("protocol",Protocol.NAME).put("actor",p.actor).put("intent_id",p.id)
            .put("start",p.start).put("end",p.end).put("fingerprint",p.fp).put("policy_version",2)
        return request("/v2/reserve",mode,b,drop)
    }
}

package org.wwl.networkdemo

import android.app.Application
import android.os.Handler
import android.os.Looper
import android.os.Process
import android.util.AtomicFile
import org.json.JSONObject
import org.wwl.intentdemo.ClientStore
import org.wwl.intentdemo.Pending
import org.wwl.intentdemo.Protocol
import java.io.File
import java.util.concurrent.ScheduledThreadPoolExecutor
import java.util.concurrent.TimeUnit

class NetworkApp:Application() {
    lateinit var repository:NetworkRepository
        private set
    override fun onCreate() { super.onCreate();repository=NetworkRepository(this) }
}

/** One process worker; service authority is outside both app sandboxes. No MockAuthority instance. */
class NetworkRepository(private val app:Application) {
    val worker=ScheduledThreadPoolExecutor(1)
    val client=ClientStore(app)
    private val main=Handler(Looper.getMainLooper())
    private val transport=HttpsTransport(BuildConfig.ACTOR)
    private val projection=AtomicFile(File(app.filesDir,"projection.json"))
    private val credential=AtomicFile(File(app.filesDir,"credential-mode.txt"))
    private var hint="尚未联网。缓存不是容量承诺。"
    private fun atomic(file:AtomicFile,bytes:ByteArray) {
        val out=file.startWrite()
        try { out.write(bytes);file.finishWrite(out) } catch(e:Exception) { file.failWrite(out);throw e }
    }
    private fun mode():String = if(credential.baseFile.exists())String(credential.readFully(),Charsets.UTF_8).also {
        require(it in setOf("self","other","invalid"))
    } else "self"
    init { worker.execute { client.recoverInflight();exportState() } }
    fun audit(kind:String,detail:String) { worker.execute { client.audit(kind,detail);exportState() } }
    fun exportState():JSONObject {
        val cached=if(projection.baseFile.exists())JSONObject(String(projection.readFully(),Charsets.UTF_8)) else JSONObject.NULL
        val state=JSONObject().put("pid",Process.myPid()).put("uid",Process.myUid()).put("actor",BuildConfig.ACTOR)
            .put("client",client.snapshot()).put("projection",cached).put("credential_mode",mode()).put("transport_hint",hint)
        atomic(AtomicFile(File(app.filesDir,"network-state.json")),state.toString(2).toByteArray(Charsets.UTF_8))
        return state
    }
    fun recordUi(value:JSONObject) { worker.execute {
        atomic(AtomicFile(File(app.filesDir,"network-ui.json")),value.toString(2).toByteArray(Charsets.UTF_8));exportState()
    } }
    fun refresh(callback:(JSONObject)->Unit) { worker.execute { val s=exportState();main.post { callback(s) } } }
    private fun work(callback:(Result<JSONObject>)->Unit,block:()->Unit) { worker.execute {
        val r=runCatching { block();exportState() };main.post { callback(r) }
    } }
    fun queue(start:Int,end:Int,id:String,callback:(Result<JSONObject>)->Unit)=work(callback) {
        client.queue(BuildConfig.ACTOR,id,start,end);hint="只在本机落盘；未访问服务。"
    }
    fun saveDraft(start:String,end:String,callback:(Result<JSONObject>)->Unit)=work(callback) { client.saveDraft(start,end) }
    fun credential(mode:String,callback:(Result<JSONObject>)->Unit)=work(callback) {
        require(mode in setOf("self","other","invalid"))
        atomic(credential,mode.toByteArray(Charsets.UTF_8));hint="切换公开测试凭据：$mode；未改变业务actor/id。"
    }
    fun fetch(callback:(Result<JSONObject>)->Unit)=work(callback) {
        try {
            val raw=transport.snapshot(mode());require(raw.getString("protocol")==Protocol.NAME)
            val rows=raw.getJSONArray("bookings")
            for(i in 0 until rows.length()) {
                val b=rows.getJSONObject(i)
                for(k in listOf("id","start","end","policy_version"))require(b.get(k) is Int||b.get(k) is Long)
                require(b.getLong("id")>0);Protocol.text(b.get("owner") as String)
                val s=b.getLong("start");val e=b.getLong("end")
                require(s in 540..1020&&e-s in 30..120&&e<=1020)
                require(b.getLong("policy_version") in 1..2)
            }
            // Only one in-process worker; receive time is local metadata, NOT a service version.
            atomic(projection,JSONObject().put("snapshot",raw).put("received_at_ms",System.currentTimeMillis()).toString().toByteArray(Charsets.UTF_8))
            hint="已取服务投影；它可能随后过时，仍不预留容量。"
        } catch(e:Exception) { hint=describe(e);client.audit("projection_failed",hint) }
    }
    private fun describe(e:Exception):String = if(e is HttpFailure)"HTTP ${e.code}：未据此改判业务；保留原id。" else "传输/校验失败 ${e.javaClass.simpleName}：结果可能未知，保留原id。"
    private fun unknown(p:Pending,r:JSONObject):Boolean {
        require(r.get("protocol")==Protocol.NAME&&r.get("actor")==p.actor&&r.get("intent_id")==p.id)
        if(r.optString("status")!="unknown")return false
        require(r.keys().asSequence().toSet()==setOf("protocol","actor","intent_id","status","reason"))
        require(r.get("reason")=="not_registered")
        return true
    }
    fun attempt(actor:String,id:String,operation:String,callback:(String,Result<JSONObject>)->Unit) {
        require(operation in setOf("send","drop","recover","query","delay"))
        worker.execute {
            val key="$actor\u0000$id"
            val result=runCatching {
                require(actor==BuildConfig.ACTOR)
                val p=client.get(actor,id)?:error("no persisted intent")
                client.phase(p,"sending");client.audit("https_attempt",p.id+":"+operation);exportState()
                var response:JSONObject?=null
                try {
                    val m=mode()
                    response=when(operation) {
                        "query","recover" -> transport.query(p,m).let { q ->
                            if(unknown(p,q)) {
                                if(operation=="recover")transport.reserve(p,m) else null
                            } else q
                        }
                        else -> transport.reserve(p,m,operation=="drop")
                    }
                    if(response==null) { client.phase(p,"unknown");hint="查询尚未登记；不是终局拒绝。" }
                } catch(e:Exception) {
                    client.phase(p,if(e is HttpFailure&&e.code==409)"protocol_error" else "unknown")
                    hint=describe(e);client.audit("https_uncertain",hint)
                }
                val reply=response
                if(reply!=null) {
                    // Persist an authenticated response only into its original row.
                    if(operation=="delay") {
                        client.audit("local_reply_held",p.id);exportState()
                        worker.schedule({
                            val delayed=runCatching { client.accept(p,reply);hint="原回复已入原行；请核UI发布门。";exportState() }
                            if(delayed.isFailure) { client.phase(p,"protocol_error");exportState() }
                            main.post { callback(key,delayed) }
                        },3,TimeUnit.SECONDS)
                        return@execute
                    }
                    try { client.accept(p,reply);hint="服务终局已核身份并保存。" }
                    catch(e:Exception) { client.phase(p,"protocol_error");hint="业务响应校验失败；原记录保留。";throw e }
                }
                exportState()
            }
            main.post { callback(key,result) }
        }
    }
}

"""Loopback HTTPS adapter. All booking decisions come from the given SE82 source.
TLS keys are generated locally for this lab; bearer strings are deliberately PUBLIC.
No OAuth issuance, account management, production hosting or TLS implementation here.
"""
import argparse
import hashlib
import importlib.util
import json
from pathlib import Path
import socket
import socketserver
import sqlite3
import ssl
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import parse_qs, urlsplit

PROTOCOL = 'wwl-reservation-v2/1'
TOKENS = {'wwl175-public-D': 'D', 'wwl175-public-E': 'E'}
SEEDS = [('A', 540, 600), ('B', 600, 630), ('C', 650, 680)]


def load_contract(path):
    spec = importlib.util.spec_from_file_location('se82_network_contract', path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def strict_json(raw):
    def pairs(items):
        result = {}
        for key, value in items:
            if key in result:
                raise ValueError('duplicate key')
            result[key] = value
        return result
    def constant(_):
        raise ValueError('nonfinite JSON')
    value = json.loads(raw.decode('utf-8'), object_pairs_hook=pairs, parse_constant=constant)
    if not isinstance(value, dict):
        raise ValueError('object required')
    return value


class LabServer(ThreadingHTTPServer):
    daemon_threads = True
    def __init__(self, address, contract, database, events, *, allow_drop=False):
        self.contract, self.database = contract, Path(database)
        self.allow_drop, self.events = allow_drop, Path(events)
        self.lock = threading.Lock()
        self.sequence = 0
        existed = self.database.exists()
        self.database.parent.mkdir(parents=True, exist_ok=True)
        self.events.parent.mkdir(parents=True, exist_ok=True)
        with contract.Authority(str(self.database)) as authority:
            if not existed:
                authority.seed_legacy(SEEDS)
            if authority.writer_epoch != 2:
                raise ValueError('this adapter supports fixed epoch 2 only')
        super().__init__(address, Handler)

    def server_bind(self):
        # Fixed loopback fixture: do not block startup on machine reverse DNS.
        socketserver.TCPServer.server_bind(self)
        self.server_name = 'localhost'
        self.server_port = self.socket.getsockname()[1]

    def event(self, **fields):
        with self.lock:
            self.sequence += 1
            row = dict(sequence=self.sequence, **fields)
            with self.events.open('a', encoding='utf-8') as f:
                f.write(json.dumps(row, ensure_ascii=False, sort_keys=True) + '\n')


class Handler(BaseHTTPRequestHandler):
    protocol_version = 'HTTP/1.1'
    server_version = 'WWLTeachingAdapter/1'
    def log_message(self, *_):
        pass  # Never log Authorization or query strings via default access log.

    def reply(self, code, value):
        raw = json.dumps(value, ensure_ascii=False, separators=(',', ':')).encode('utf-8')
        self.send_response(code)
        self.send_header('Content-Type', 'application/json; charset=utf-8')
        self.send_header('Content-Length', str(len(raw)))
        self.send_header('Connection', 'close')
        self.send_header('Cache-Control', 'no-store')
        if code == 401:
            self.send_header('WWW-Authenticate', 'Bearer realm="wwl-public-lab", error="invalid_token"')
        self.end_headers()
        self.wfile.write(raw)
        self.close_connection = True

    def handle_request(self, method):
        subject = None
        identity = None
        path = urlsplit(self.path)
        try:
            auth = self.headers.get_all('Authorization', [])
            if len(auth) != 1 or not auth[0].startswith('Bearer '):
                code, value = 401, {'error': 'invalid_token'}
            else:
                subject = TOKENS.get(auth[0][7:])
                if subject is None:
                    code, value = 401, {'error': 'invalid_token'}
                else:
                    code, value, identity = self.authorized(method, path, subject)
            self.server.event(method=method, path=path.path, subject=subject,
                              intent_id=identity, http_status=code,
                              outcome=value.get('status'), error=value.get('error'),
                              tls=self.connection.version())
            if code == 200 and path.path == '/v2/reserve' and self.headers.get('X-WWL-Lab-Drop-Reply') == 'true':
                # reserve() has returned after COMMIT. Close the actual TLS/TCP
                # connection BEFORE HTTP headers, including every automatic retry.
                self.server.event(kind='committed_then_wire_closed', subject=subject,
                                  intent_id=identity, outcome=value['status'])
                self.close_connection = True
                try:
                    self.connection.shutdown(socket.SHUT_RDWR)
                except OSError:
                    pass
                self.connection.close()
                return
            self.reply(code, value)
        except self.server.contract.IntentConflict:
            self.server.event(method=method, path=path.path, subject=subject, http_status=409,
                              error='intent_conflict')
            self.reply(409, {'error': 'intent_conflict'})
        except (ValueError, UnicodeError, self.server.contract.InvalidInput) as exc:
            self.server.event(method=method, path=path.path, subject=subject, http_status=400,
                              error=type(exc).__name__)
            self.reply(400, {'error': 'invalid_request'})
        except (sqlite3.OperationalError, self.server.contract.WriterFenced):
            self.server.event(method=method, path=path.path, subject=subject, http_status=503,
                              error='authority_unavailable')
            self.reply(503, {'error': 'authority_unavailable'})

    def authorized(self, method, path, subject):
        c = self.server.contract
        if method == 'GET' and path.path == '/v2/snapshot' and not path.query:
            with c.Authority(str(self.server.database)) as a:
                return 200, {'protocol': PROTOCOL, 'bookings': a.snapshot()}, None
        if method == 'GET' and path.path == '/v2/intent':
            params = parse_qs(path.query, keep_blank_values=True, strict_parsing=True)
            if set(params) != {'intent_id', 'fingerprint'} or any(len(v) != 1 for v in params.values()):
                raise ValueError('query shape')
            identity, fp = params['intent_id'][0], params['fingerprint'][0]
            c.text(identity, 'intent_id')
            if len(fp) != 64 or any(x not in '0123456789abcdef' for x in fp):
                raise ValueError('fingerprint')
            with c.Authority(str(self.server.database)) as a:
                return 200, a.query(subject, identity, fp), identity
        if method != 'POST' or path.path != '/v2/reserve' or path.query:
            return 404, {'error': 'not_found'}, None
        if self.headers.get('X-WWL-Lab-Drop-Reply') is not None and (
                not self.server.allow_drop or self.headers['X-WWL-Lab-Drop-Reply'] != 'true'):
            return 400, {'error': 'lab_fault_disabled'}, None
        lengths = self.headers.get_all('Content-Length', [])
        if self.headers.get('Transfer-Encoding') or len(lengths) != 1 or not lengths[0].isascii() or not lengths[0].isdecimal():
            raise ValueError('length required')
        size = int(lengths[0])
        if not 0 < size <= 4096 or self.headers.get_content_type() != 'application/json':
            raise ValueError('body contract')
        value = strict_json(self.rfile.read(size))
        if set(value) != {'protocol', 'actor', 'intent_id', 'start', 'end', 'fingerprint', 'policy_version'}:
            raise ValueError('request fields')
        if value['actor'] != subject:
            return 403, {'error': 'actor_mismatch'}, None
        if value['protocol'] != PROTOCOL or type(value['policy_version']) is not int or value['policy_version'] != 2:
            raise ValueError('version')
        identity = c.text(value['intent_id'], 'intent_id')
        for k in ('start', 'end'):
            if type(value[k]) is not int or not -(2**31) <= value[k] < 2**31:
                raise ValueError('Android Int range')
        fp = c.fingerprint(subject, value['start'], value['end'])
        if value['fingerprint'] != fp:
            raise ValueError('fingerprint mismatch')
        with c.Authority(str(self.server.database)) as a:
            return 200, a.reserve(subject, identity, value['start'], value['end'], writer_epoch=2), identity

    def do_GET(self):
        self.handle_request('GET')

    def do_POST(self):
        self.handle_request('POST')
