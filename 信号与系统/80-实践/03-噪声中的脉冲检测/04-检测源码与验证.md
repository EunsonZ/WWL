```python
"""U6 finite real-vector detection; known sigma and finite search grid.
All simulation streams split by SeedSequence; no output is written on import.
"""
import argparse
import json
from pathlib import Path
from statistics import NormalDist
import numpy as np

NORMAL = NormalDist()


def real_array(x, name):
    """Reject complex values before float conversion, including object arrays."""
    try:
        raw = np.asarray(x)
        if np.iscomplexobj(raw) or (raw.dtype == object and any(
                isinstance(v, (complex, np.complexfloating)) or np.iscomplexobj(v)
                for v in raw.flat)):
            raise ValueError(name + ' must be real')
        a = np.asarray(raw, dtype=float)
    except (TypeError, ValueError, OverflowError) as e:
        raise ValueError(name + ' must contain real finite numeric values') from e
    if not np.all(np.isfinite(a)):
        raise ValueError(name + ' must contain real finite numeric values')
    return a


def vector(x, name='vector'):
    a = real_array(x, name)
    if a.ndim != 1 or a.size == 0:
        raise ValueError(name + ' must be a nonempty finite real vector')
    return a


def positive(x, name):
    a = real_array(x, name)
    if a.ndim != 0 or a <= 0:
        raise ValueError(name + ' must be a positive finite real scalar')
    return float(a)


def template(s):
    s = vector(s, 'template')
    with np.errstate(over='ignore', under='ignore', invalid='ignore'):
        energy = float(np.dot(s, s))
    if not np.isfinite(energy) or energy <= 0:
        raise ValueError('template energy must be positive and numerically finite')
    return s


def threshold(alpha):
    a = real_array(alpha, 'alpha')
    if a.ndim != 0 or not 0 < a < 1:
        raise ValueError('alpha must lie strictly between zero and one')
    return NORMAL.inv_cdf(1-float(a))


def known_score(x, s, sigma=1.0):
    x, s = vector(x), template(s)
    if x.shape != s.shape:
        raise ValueError('x and template shapes differ')
    return float(x @ s / (positive(sigma, 'sigma') * np.linalg.norm(s)))


def white_scores(x, s, sigma=1.0):
    x, s = vector(x), template(s)
    if x.size < s.size:
        raise ValueError('record shorter than template')
    return np.correlate(x, s, mode='valid') / (positive(sigma,'sigma') * np.linalg.norm(s))


def colored_score(x, s, covariance):
    """Real SPD C; symmetry tolerance is 1e-12 of max(abs(C)), unit invariant.
    Roundoff within that relative tolerance is explicitly symmetrized before
    Cholesky; inconsistent triangles are never silently read as another model.
    """
    x, s = vector(x), template(s)
    c = real_array(covariance, 'covariance')
    if x.shape != s.shape or c.shape != (s.size,s.size):
        raise ValueError('covariance shape and vectors must match')
    scale = float(np.max(np.abs(c)))
    if scale <= 0:
        raise ValueError('covariance must be positive definite')
    cn = c / scale
    if np.max(np.abs(cn-cn.T)) > 1e-12:
        raise ValueError('covariance must be symmetric within relative tolerance 1e-12')
    cn = 0.5 * (cn + cn.T)
    try:
        l = np.linalg.cholesky(cn)
        wx = np.linalg.solve(l, x / np.sqrt(scale))
        ws = np.linalg.solve(l, s / np.sqrt(scale))
    except np.linalg.LinAlgError as e:
        raise ValueError('covariance must be positive definite and solvable') from e
    norm = float(np.linalg.norm(ws))
    if not np.isfinite(norm) or norm <= 0 or not np.all(np.isfinite(wx)):
        raise ValueError('whitened vectors must have finite values and nonzero template norm')
    score = float(wx @ (ws / norm))
    if not np.isfinite(score):
        raise ValueError('score must be numerically finite')
    return score


def periodogram(x, fs, demean=False):
    """Rectangular, one-sided density: unit(x)^2/Hz; exact bin-area identity."""
    x = vector(x)
    fs = positive(fs,'fs')
    if demean:
        x = x - np.mean(x)
    p = abs(np.fft.rfft(x))**2 / (fs*x.size)
    if x.size % 2 == 0:
        p[1:-1] *= 2
    else:
        p[1:] *= 2
    return np.fft.rfftfreq(x.size,1/fs), p


def sample_variance(x):
    x = vector(x)
    if x.size < 2:
        raise ValueError('unbiased iid variance needs at least two samples')
    return float(np.sum((x-x.mean())**2)/(x.size-1))


def evaluate(labels, decisions):
    """FPR/TPR conditional denominators; absent class is unknown (None)."""
    y, d = real_array(labels, 'labels'), real_array(decisions, 'decisions')
    if y.ndim != 1 or y.size == 0 or d.shape != y.shape or not np.all(np.isin(y,[0,1])) or not np.all(np.isin(d,[0,1])):
        raise ValueError('matching nonempty binary labels and decisions required')
    n0,n1 = int(np.sum(y==0)), int(np.sum(y==1))
    fp,tp = int(np.sum((y==0)&(d==1))),int(np.sum((y==1)&(d==1)))
    return {'n0':n0,'n1':n1,'fp':fp,'tp':tp,'fpr':fp/n0 if n0 else None,'tpr':tp/n1 if n1 else None}


def calibrate_maxima(maxima, alpha):
    """Noise-only training quantile; empirical, no population guarantee."""
    m=vector(maxima,'maxima')
    threshold(alpha)
    return float(np.quantile(m,1-alpha,method='higher'))


def simulation(seed=62026, ntrain=6000, ntest=12000):
    if not isinstance(seed,(int,np.integer)) or seed<0 or min(ntrain,ntest)<2:
        raise ValueError('nonnegative integer seed and at least two train/test records')
    s=np.array([0.0,1.0,2.0,1.0,0.0]); n=24; amplitude=1.25; alpha=.01
    streams=[np.random.default_rng(v) for v in np.random.SeedSequence(seed).spawn(3)]
    def bank(rng,count):
        x=rng.normal(size=(count,n))
        return x, np.array([white_scores(row,s) for row in x])
    _, train=bank(streams[0],ntrain)
    x0,z0=bank(streams[1],ntest)
    x1=streams[2].normal(size=(ntest,n)); starts=streams[2].integers(0,n-s.size+1,size=ntest)
    for i,k in enumerate(starts): x1[i,k:k+s.size]+=amplitude*s
    z1=np.array([white_scores(row,s) for row in x1])
    g=calibrate_maxima(train.max(axis=1),alpha)
    single=threshold(alpha); bonf=threshold(alpha/z0.shape[1])
    estimate=z1.argmax(axis=1); detected=z1.max(axis=1)>g
    result={'seed':seed,'ntrain':ntrain,'ntest_per_class':ntest,'sigma':1.0,'amplitude':amplitude,'template':s.tolist(),'record_length':n,'search_positions':z0.shape[1],'alpha':alpha,'threshold_single':single,'threshold_bonferroni':bonf,'threshold_calibrated':g,
      'fpr_at_one_position':float(np.mean(z0[:,0]>single)),
      'fpr_global_using_single':float(np.mean(z0.max(axis=1)>single)),
      'fpr_global_bonferroni':float(np.mean(z0.max(axis=1)>bonf)),
      'fpr_global_calibrated':float(np.mean(z0.max(axis=1)>g)),
      'tpr_global_calibrated':float(detected.mean()),
      'arrival_mae_all_h1_samples':float(np.mean(abs(estimate-starts))),
      'arrival_mae_detected_only_samples':float(np.mean(abs(estimate[detected]-starts[detected]))) if detected.any() else None,
      'known_arrival_theory_tpr':1-NORMAL.cdf(single-amplitude*np.linalg.norm(s)),
      'evidence':'finite simulated records, not confidence interval or NP proof'}
    return result


def main():
    p=argparse.ArgumentParser(description=__doc__); p.add_argument('--output',required=True,type=Path); a=p.parse_args()
    a.output.mkdir(parents=True,exist_ok=True)
    (a.output/'results.json').write_text(json.dumps(simulation(),indent=2)+'\n')

if __name__=='__main__': main()
```

```python
import unittest
import math
import warnings
import numpy as np
from detection_lab import (known_score, white_scores, colored_score, periodogram,
    threshold, sample_variance, evaluate, calibrate_maxima, simulation)

class DetectionTests(unittest.TestCase):
    def test_closed_form(self):
        self.assertAlmostEqual(known_score([2,1],[1,2]),4/math.sqrt(5))
        self.assertAlmostEqual(colored_score([2,1],[1,2],[[4,0],[0,1]]),2.5/math.sqrt(4.25))
        np.testing.assert_allclose(white_scores([0,1,2,1,0],[1,2,1]),[4,6,4]/np.sqrt(6))
        self.assertAlmostEqual(threshold(.5),0)

    def test_psd_dc_and_parseval(self):
        for x in ([2,2,2,2],[1,-1,1,-1],[1,2,3,4,5]):
            f,p=periodogram(x,8)
            self.assertAlmostEqual(sum(p)*8/len(x),np.mean(np.square(x)))
            _,q=periodogram(x,8,True)
            self.assertAlmostEqual(sum(q)*8/len(x),np.var(x,ddof=0))
            self.assertAlmostEqual(q[0],0)
        _,p=periodogram([2,2,2,2],8)
        self.assertEqual(p[0],2)

    def test_variance_bias_correction_independent_enumeration(self):
        # All iid Bernoulli(.5) pairs: unbiased mean is population variance .25.
        pairs=[[0,0],[0,1],[1,0],[1,1]]
        self.assertEqual(sum(sample_variance(x) for x in pairs)/4,.25)
        self.assertEqual(sum(np.var(x) for x in pairs)/4,.125)

    def test_multiple_independent_positions(self):
        # Template [1] makes the positions genuinely independent, exact FPR 1-(1-a)^K.
        rng=np.random.default_rng(71); z=rng.normal(size=(100000,8)); t=threshold(.01)
        observed=np.mean(z.max(axis=1)>t); exact=1-.99**8
        self.assertLess(abs(observed-exact),.004)
        self.assertLess(np.mean(z.max(axis=1)>threshold(.01/8)),.014)
        self.assertGreater(observed,.05)

    def test_labels_and_unknown(self):
        self.assertEqual(evaluate([0,0,1,1],[0,1,0,1])['fpr'],.5)
        self.assertIsNone(evaluate([1],[1])['fpr'])
        for y,d in [([0,2],[0,1]),([0],[float('nan')]),([] ,[]),([0,1],[1])]:
            with self.assertRaises(ValueError): evaluate(y,d)

    def test_invalid_inputs(self):
        for x,s in [([], [1]),([1],[0]),([float('nan')],[1]),([1],[1,2])]:
            with self.assertRaises(ValueError): known_score(x,s)
        for a in (0,1,-.1,float('nan')):
            with self.assertRaises(ValueError): threshold(a)
        for c in ([[1,2],[2,1]],[[1,1],[0,1]]):
            with self.assertRaises(ValueError): colored_score([1,2],[1,1],c)
        with self.assertRaises(ValueError): sample_variance([1])
        with self.assertRaises(ValueError): white_scores([1],[1,1])
        with self.assertRaises(ValueError): periodogram([1],0)

    def test_real_only_numpy_and_object_inputs(self):
        for c in (np.array([[4+3j,0],[0,1]]), np.array([[4+0j,0],[0,1]]),
                  np.array([[4+3j,0],[0,1]],dtype=object)):
            with self.assertRaises(ValueError): colored_score([2,1],[1,2],c)
        for complex_vector in (np.array([2+3j,1]),np.array([2+0j,1]),
                               np.array([2+3j,1],dtype=object)):
            with self.assertRaises(ValueError): known_score(complex_vector,[1,2])
            with self.assertRaises(ValueError): known_score([2,1],complex_vector)
            with self.assertRaises(ValueError): white_scores(complex_vector,[1])
            with self.assertRaises(ValueError): colored_score(complex_vector,[1,2],np.eye(2))
            with self.assertRaises(ValueError): colored_score([2,1],complex_vector,np.eye(2))
            with self.assertRaises(ValueError): periodogram(complex_vector,8)
        for value in (1+0j,np.array(1+2j),np.array(1+0j,dtype=object)):
            with self.assertRaises(ValueError): known_score([2,1],[1,2],value)
            with self.assertRaises(ValueError): periodogram([2,1],value)
            with self.assertRaises(ValueError): threshold(value)
        with self.assertRaises(ValueError): evaluate(np.array([0,1],dtype=complex),[0,1])
        for bad in ([object(),1],['nonnumeric',1],[float('inf'),1]):
            with self.assertRaises(ValueError): known_score(bad,[1,2])
        for c in ([[object(),0],[0,1]],[[float('nan'),0],[0,1]]):
            with self.assertRaises(ValueError): colored_score([2,1],[1,2],c)

    def test_numpy_complex_scalars_inside_object_arrays(self):
        # 2 scalar types x 2 imaginary cases x 6 entries = 24 rejections.
        for dtype in (np.complex64, np.complex128):
            for imag in (0.0, 2.0):
                cov=np.array([[dtype(4+imag*1j),0],[0,1]],dtype=object)
                vec=np.array([dtype(2+imag*1j),1],dtype=object)
                scalar=np.array(dtype(1+imag*1j),dtype=object)
                alpha=np.array(dtype(.1+imag*1j),dtype=object)
                labels=np.array([0,dtype(1+imag*1j)],dtype=object)
                cases=(('covariance',lambda:colored_score([2,1],[1,2],cov)),
                       ('vector',lambda:known_score(vec,[1,2])),
                       ('template',lambda:known_score([2,1],vec)),
                       ('sigma',lambda:known_score([2,1],[1,2],scalar)),
                       ('alpha',lambda:threshold(alpha)),
                       ('labels',lambda:evaluate(labels,[0,1])))
                for entry,fn in cases:
                    with self.subTest(dtype=dtype.__name__,imag=imag,entry=entry):
                        with warnings.catch_warnings(record=True) as seen:
                            warnings.simplefilter('always')
                            with self.assertRaises(ValueError): fn()
                        self.assertEqual(seen,[], 'reject before any lossy float conversion')

    def test_covariance_symmetry_is_unit_invariant(self):
        asymmetric=np.array([[4.,0.],[.9,1.]])
        for unit in (1.,1e-6,1e6):
            with self.assertRaises(ValueError):
                colored_score(unit*np.array([2.,1.]),unit*np.array([1.,2.]),unit**2*asymmetric)
        # Separate two-scale small off-diagonal counterexample avoids absolute tolerance.
        other=np.array([[1.,0.],[1e-4,2.]])
        for unit in (1.,1e-6):
            with self.assertRaises(ValueError):
                colored_score(unit*np.array([2.,1.]),unit*np.array([1.,2.]),unit**2*other)
        c=np.array([[4.,.5],[.5,1.]])
        baseline=colored_score([2,1],[1,2],c)
        for unit in (1e-6,1e6):
            self.assertAlmostEqual(colored_score(unit*np.array([2.,1.]),unit*np.array([1.,2.]),unit**2*c),baseline)
        for c in (np.zeros((2,2)),np.diag([1.,0.]),np.diag([1.,-1.])):
            with self.assertRaises(ValueError): colored_score([2,1],[1,2],c)
        with self.assertRaises(ValueError): colored_score([2,1],[0,0],np.eye(2))

    def test_calibration_and_reproducibility(self):
        self.assertEqual(calibrate_maxima([1,2,3,4],.25),4)
        self.assertEqual(simulation(11,20,40),simulation(11,20,40))

if __name__=='__main__': unittest.main()
```
