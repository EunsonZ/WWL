# 路径提升与群字复算

这个标准库实验核对四类有限计算：自由字的相邻逆对消去、指数和、$n$ 片圆周覆盖中的提升终点、环面的格点终点与 $\mathbb{RP}^2$ 的奇偶类。它不证明路径/同伦提升、van Kampen 定理、平面脊柱形变收缩，也不凭有限样本认证群表示。

## 从发布后的库根运行

```bash
python3 -B 数学/36-拓扑学/10-覆盖空间与van-Kampen/covering_vankampen_lab.py --word abBAcC --winding 7 --sheets 5
python3 -B -m unittest discover -s 数学/36-拓扑学/10-覆盖空间与van-Kampen -p 'test_covering_vankampen_lab.py' -v
```

候选区首条命令的输出为：

```text
free reduction: 'abBAcC' -> ''
exponent sums: (0, 0, 0)
5-sheet lift of winding 7: endpoint=2, closed=False
torus commutator endpoint: (0, 0)
RP2 classes of a^0..a^5: (0, 1, 0, 1, 0, 1)
```

注意 `abAB` 的指数和是 $(0,0,0)$，但自由约化后仍是 `abAB`。指数和只看阿贝尔化，不能判定自由群中的单位元。环面已经加入交换关系，所以同一词的格点终点为 $(0,0)$；这是两个不同群中的计算口径。

```python
"""Finite checks for covering endpoints and group-word examples.

These functions verify particular symbolic calculations.  Finite computation
does not prove path lifting, van Kampen's theorem, or a group presentation.
"""
from __future__ import annotations

import argparse


INVERSE = {"a": "A", "A": "a", "b": "B", "B": "b", "c": "C", "C": "c"}


def _validate_word(word: str) -> None:
    if not isinstance(word, str):
        raise TypeError("word must be a string")
    invalid = sorted(set(word) - set(INVERSE))
    if invalid:
        raise ValueError(f"invalid letters: {''.join(invalid)}")


def reduce_free_word(word: str) -> str:
    """Cancel adjacent inverse pairs in a word on a,b,c."""
    _validate_word(word)
    stack: list[str] = []
    for letter in word:
        if stack and INVERSE[letter] == stack[-1]:
            stack.pop()
        else:
            stack.append(letter)
    return "".join(stack)


def exponent_sums(word: str) -> tuple[int, int, int]:
    """Return exponent sums of a,b,c, as used after abelianization."""
    _validate_word(word)
    return (
        word.count("a") - word.count("A"),
        word.count("b") - word.count("B"),
        word.count("c") - word.count("C"),
    )


def circle_lift_endpoint(winding: int, sheets: int, start: int = 0) -> tuple[int, bool]:
    """Endpoint sheet and closedness for z -> z**sheets."""
    values = (winding, sheets, start)
    if any(isinstance(value, bool) or not isinstance(value, int) for value in values):
        raise TypeError("winding, sheets, and start must be integers")
    if sheets <= 0:
        raise ValueError("sheets must be positive")
    endpoint = (start + winding) % sheets
    return endpoint, endpoint == start % sheets


def torus_lift_endpoint(word: str) -> tuple[int, int]:
    """Integer-lattice endpoint of a torus word lifted from (0,0)."""
    a_sum, b_sum, _ = exponent_sums(word)
    if "c" in word or "C" in word:
        raise ValueError("torus words may only use a,A,b,B")
    return a_sum, b_sum


def rp2_class(exponent: int) -> int:
    """Class of a**exponent in <a | a**2=1>."""
    if isinstance(exponent, bool) or not isinstance(exponent, int):
        raise TypeError("exponent must be an integer")
    return exponent % 2


def report(word: str, winding: int, sheets: int) -> None:
    endpoint, closed = circle_lift_endpoint(winding, sheets)
    print(f"free reduction: {word!r} -> {reduce_free_word(word)!r}")
    print(f"exponent sums: {exponent_sums(word)}")
    print(f"{sheets}-sheet lift of winding {winding}: endpoint={endpoint}, closed={closed}")
    print(f"torus commutator endpoint: {torus_lift_endpoint('abAB')}")
    print(f"RP2 classes of a^0..a^5: {tuple(rp2_class(k) for k in range(6))}")


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--word", default="abBAcC")
    parser.add_argument("--winding", type=int, default=7)
    parser.add_argument("--sheets", type=int, default=5)
    arguments = parser.parse_args()
    report(arguments.word, arguments.winding, arguments.sheets)
```

```python
"""Tests for the finite covering-space and word calculations."""
import unittest

import covering_vankampen_lab as lab


class CoveringVanKampenLabTests(unittest.TestCase):
    def test_free_reduction(self):
        self.assertEqual(lab.reduce_free_word("abBAcC"), "")
        self.assertEqual(lab.reduce_free_word("abAB"), "abAB")
        self.assertEqual(lab.reduce_free_word("abcCBA"), "")

    def test_reduction_is_idempotent(self):
        for word in ("", "aA", "abAB", "aaBBccCCA"):
            reduced = lab.reduce_free_word(word)
            self.assertEqual(lab.reduce_free_word(reduced), reduced)

    def test_exponent_sums_do_not_detect_free_identity(self):
        self.assertEqual(lab.exponent_sums("abAB"), (0, 0, 0))
        self.assertEqual(lab.reduce_free_word("abAB"), "abAB")

    def test_circle_lift_endpoint_and_closedness(self):
        self.assertEqual(lab.circle_lift_endpoint(7, 5, 1), (3, False))
        self.assertEqual(lab.circle_lift_endpoint(15, 5, 1), (1, True))
        self.assertEqual(lab.circle_lift_endpoint(-2, 5, 1), (4, False))

    def test_torus_endpoints(self):
        self.assertEqual(lab.torus_lift_endpoint("aaaBB"), (3, -2))
        self.assertEqual(lab.torus_lift_endpoint("abAB"), (0, 0))

    def test_rp2_parity(self):
        self.assertEqual(tuple(lab.rp2_class(k) for k in range(-2, 4)), (0, 1, 0, 1, 0, 1))

    def test_invalid_inputs(self):
        cases = [
            lambda: lab.reduce_free_word("ad"),
            lambda: lab.reduce_free_word(3),
            lambda: lab.circle_lift_endpoint(1, 0),
            lambda: lab.circle_lift_endpoint(True, 2),
            lambda: lab.torus_lift_endpoint("ac"),
            lambda: lab.rp2_class(2.0),
        ]
        for case in cases:
            with self.assertRaises((TypeError, ValueError)):
                case()


if __name__ == "__main__":
    unittest.main()
```

## 两个读数怎样回到正文

- `circle_lift_endpoint(7, 5, 1) == (3, False)` 对应 01 的迁移题：从第1片出发，绕数7使片编号加7模5，终点第3片。
- `torus_lift_endpoint("abAB") == (0, 0)` 对应环面普适覆盖的格点终点；它不说明同一个交换子在楔圆自由群中为单位。
- `rp2_class(k)` 只实现关系 $a^2=1$ 后的奇偶化；关系来自二维圆盘的次数2附着，不是代码发现的。
