cmake_minimum_required(VERSION 3.20)
project(WWLOwnedReportCoroutine LANGUAGES CXX)
include(CTest)
find_package(WWLMeasurement 1.2 CONFIG REQUIRED)
foreach(target IN ITEMS coroutine_consumer coroutine_tests)
  if(target STREQUAL "coroutine_consumer")
    add_executable(${target} consumer.cpp)
  else()
    add_executable(${target} tests.cpp)
  endif()
  target_link_libraries(${target} PRIVATE WWL::measurement)
  target_compile_features(${target} PRIVATE cxx_std_20)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
  endif()
endforeach()
if(BUILD_TESTING)
  add_test(NAME owned_report_consumer COMMAND coroutine_consumer)
  add_test(NAME owned_report_coroutine_contracts COMMAND coroutine_tests)
  set_tests_properties(owned_report_consumer owned_report_coroutine_contracts PROPERTIES TIMEOUT 20)
endif()
