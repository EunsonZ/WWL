# 公开数据建模复现实验

本实验读取随课程保存的官方年度快照，计算训练/选择/保留指标，并重新生成两幅图。它不联网，不安装依赖，也不访问个人数据。数值计算需要 Python≥3.10 和 NumPy；生成图额外需要 Matplotlib。环境缺少依赖时如实记录未运行，不自动安装。

从库根运行：

```bash
python3 -B -m unittest discover -s 数学/70-数学建模 -v
MPLCONFIGDIR=/private/tmp/wwl-m4-mpl python3 -B 数学/70-数学建模/modeling_lab.py --output /private/tmp/wwl-m4-run
```

第二条命令只向指定输出目录写 `results.json` 与 `图/` 下 SVG/PNG。不要省略输出位置后意外覆盖自己已经编辑的课程图；默认输出是源码所在目录。设置绘图缓存目录是为了在只读主目录环境也能运行。两图内年份、单位、图例和训练/保留区间必须一起看；英文图题的中文解释在 [[数学/70-数学建模/04-真实数据保留验证与误差诊断|验证课]]。

## 输入输出与数值合同

`design(years, degree, center, scale)` 接受非空有限一维年份向量，返回 $n\times(d+1)$ 设计矩阵；`degree` 仅接受整数 1 或 2，尺度有限且正。`fit` 的 values 与年份等长，观察数至少 $d+1$，数值秩必须满列；返回长度 $d+1$ 的 beta 与该表示的二范数条件数。年份在纯数值函数里可连续取值；读取官方年度表时才要求整数年份。

`predict` 接受长度 2 或 3 的有限系数向量；`metrics` 接受相同形状的两个非空有限向量，返回 n、RMSE、MAE、bias。遇到不足观察、重复年份的秩亏、NaN/Inf、溢出、字段错、负 mean/unc 或缺年度等情况，抛 ValueError。`interval` 针对已通过 `load_data` 的三列表使用，端点必须为有序整数。只有 NumPy 的标准双精度范围被支持，不宣称任意大时间或参数都稳定可算。

独立 oracle 有两层：三点手解 $(332,20,4)$ 与直线 $(1004/3,20)$；真实对称训练窗另从原 CSV 用 `Fraction` 精确有理数求矩方程，核对 NumPy 系数及保留 MSE。系数绝对容限 $10^{-10}$ ppm，基底转换后的预测容限 $10^{-9}$ ppm，MSE 核对到九位小数；容限用于双精度舍入，不包含实测不确定性。14 个测试方法涵盖这些 oracle、解析敏感性、基底变换、指标手解、无效输入、CSV 首行及后续行的多余字段、缺字段、空值、重复表头与区间边界、选模平手规则。作者在保存快照上运行通过，不代表人的自主项目已完成。

```python
"""Reproducible retrospective NOAA annual CO2 comparison, no network access.
Python >=3.10, NumPy. Plotting only in run() additionally needs Matplotlib.
All ppm errors are prediction minus observation; data uncertainty is not model error.
"""
from pathlib import Path
import csv
import hashlib
import json
import math
import numpy as np

BASE = Path(__file__).resolve().parent
CENTER = 1979.0
SCALE = 20.0

def vector(value, name):
    a = np.asarray(value, dtype=float)
    if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
        raise ValueError(f'{name} must be a nonempty finite vector')
    return a

def design(years, degree, center=CENTER, scale=SCALE):
    y = vector(years, 'years')
    if type(degree) is not int or degree not in (1, 2):
        raise ValueError('degree must be integer 1 or 2')
    if not math.isfinite(center) or not math.isfinite(scale) or scale <= 0:
        raise ValueError('finite center and positive scale required')
    with np.errstate(over='ignore', invalid='ignore'):
        x = (y-center)/scale
        a = np.column_stack([x**i for i in range(degree+1)])
    if not np.isfinite(a).all():
        raise ValueError('design overflow')
    return a

def fit(years, values, degree, center=CENTER, scale=SCALE):
    a = design(years, degree, center, scale)
    b = vector(values, 'values')
    if len(b) != len(a) or len(b) < degree+1:
        raise ValueError('matching lengths and sufficient observations required')
    beta, _, rank, singular = np.linalg.lstsq(a, b, rcond=None)
    if rank != degree+1:
        raise ValueError('rank deficient design: coefficients not identifiable')
    if not np.isfinite(beta).all():
        raise ValueError('nonfinite fitted coefficients')
    return beta, float(singular[0]/singular[-1])

def predict(years, beta, center=CENTER, scale=SCALE):
    b = vector(beta, 'beta')
    if len(b) not in (2, 3):
        raise ValueError('beta must have length 2 or 3')
    with np.errstate(over='ignore', invalid='ignore'):
        p = design(years, len(b)-1, center, scale) @ b
    if not np.isfinite(p).all():
        raise ValueError('prediction overflow')
    return p

def metrics(observed, predicted):
    y, p = vector(observed, 'observed'), vector(predicted, 'predicted')
    if y.shape != p.shape:
        raise ValueError('matching lengths required')
    with np.errstate(over='ignore', invalid='ignore'):
        e = p-y
        rmse = np.sqrt(np.mean(e*e))
    if not np.isfinite(e).all() or not np.isfinite(rmse):
        raise ValueError('metric overflow')
    return {'n':len(y), 'rmse':float(rmse), 'mae':float(np.mean(abs(e))),
            'bias':float(np.mean(e))}

def load_data(path):
    with Path(path).open(encoding='utf-8') as f:
        reader = csv.DictReader(line for line in f if not line.startswith('#'))
        rows = list(reader)
        fields = reader.fieldnames
    expected = {'year', 'mean', 'unc'}
    if not rows or fields is None or len(fields) != 3 or set(fields) != expected:
        raise ValueError('expected exactly NOAA year,mean,unc columns')
    for row in rows:
        if set(row) != expected or any(row[k] is None or not row[k].strip() for k in expected):
            raise ValueError('every row must contain exactly three nonempty fields')
    try:
        a = np.array([[float(r[k]) for k in ('year','mean','unc')] for r in rows])
    except (TypeError, ValueError) as exc:
        raise ValueError('invalid numeric data') from exc
    if not np.isfinite(a).all() or (np.diff(a[:,0]) <= 0).any():
        raise ValueError('finite, strictly increasing years required')
    if (a[:,0] != np.floor(a[:,0])).any() or (a[:,1] <= 0).any() or (a[:,2] < 0).any():
        raise ValueError('integer years, positive mole fractions and nonnegative uncertainties required')
    return a

def interval(data, first, last):
    if type(first) is not int or type(last) is not int or first > last:
        raise ValueError('ordered integer endpoints required')
    part = data[(data[:,0] >= first) & (data[:,0] <= last)]
    if len(part) != last-first+1 or not np.array_equal(part[:,0], np.arange(first,last+1)):
        raise ValueError('incomplete requested year interval')
    return part

def choose_degree(scores):
    if set(scores) != {1,2} or not all(math.isfinite(s) and s >= 0 for s in scores.values()):
        raise ValueError('finite nonnegative scores for degrees 1 and 2 required')
    return 1 if scores[1] <= scores[2]+1e-9 else 2

def save_svg(fig, path):
    """Remove trailing spaces/tabs from SVG lines without changing plot objects."""
    path = Path(path)
    fig.savefig(path, metadata={'Date': None})
    path.write_bytes(b'\n'.join(line.rstrip(b' \t') for line in path.read_bytes().split(b'\n')))

def run(output=BASE):
    """Write deterministic JSON and figures only to caller-selected output directory."""
    import matplotlib
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt
    matplotlib.rcParams.update({'svg.hashsalt':'wwl-co2-m4','font.family':'DejaVu Sans'})
    out = Path(output)
    (out/'图').mkdir(parents=True, exist_ok=True)
    path = BASE/'数据/co2_annmean_mlo.csv'
    data = load_data(path)
    train = interval(data,1959,1999)
    selection = interval(data,2000,2009)
    holdout = interval(data,2010,2025)
    result = {'data_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
              'center':CENTER,'scale':SCALE,'models':{},'refit':False}
    fits = {}
    for degree in (1,2):
        beta, cond = fit(train[:,0],train[:,1],degree)
        fits[degree] = beta
        result['models'][str(degree)] = {'beta':beta.tolist(),'condition':cond,
           'train':metrics(train[:,1],predict(train[:,0],beta)),
           'selection':metrics(selection[:,1],predict(selection[:,0],beta))}
    selected = choose_degree({d:result['models'][str(d)]['selection']['rmse'] for d in fits})
    result['selected_degree'] = selected
    # Holdout metrics are computed only after the selection rule has chosen a model.
    for d,beta in fits.items():
        result['models'][str(d)]['holdout'] = metrics(holdout[:,1],predict(holdout[:,0],beta))
        result['models'][str(d)]['prediction_2025'] = float(predict([2025],beta)[0])
    beta = fits[selected]
    clean = holdout[~np.isin(holdout[:,0],[2022,2023])]
    recent = interval(data,1979,1999)
    rb, rc = fit(recent[:,0],recent[:,1],selected)
    plus = beta.copy(); plus[-1] += 0.1
    result['posthoc_diagnostics'] = {
      'holdout_excluding_2022_2023':metrics(clean[:,1],predict(clean[:,0],beta)),
      'train_1979_1999_beta':rb.tolist(), 'train_1979_1999_condition':rc,
      'train_1979_1999_holdout':metrics(holdout[:,1],predict(holdout[:,0],rb)),
      'beta_last_plus_0_1_change_2025':float(predict([2025],plus)[0]-predict([2025],beta)[0])}
    result['observed_2025'] = float(holdout[-1,1])
    # Both error and trajectory figures use original train-only fits.
    fig,ax = plt.subplots(figsize=(9,4.8),layout='constrained')
    ax.scatter(data[:,0],data[:,1],s=13,color='black',label='NOAA annual observations')
    for d,b in fits.items(): ax.plot(data[:,0],predict(data[:,0],b),label=f'Degree {d}, fit 1959-1999')
    ax.axvspan(1999.5,2009.5,color='gold',alpha=.13,label='Model selection')
    ax.axvspan(2009.5,2025.5,color='green',alpha=.08,label='Holdout')
    ax.set(xlabel='Year',ylabel='Dry-air CO2 mole fraction (ppm)',title='Retrospective prediction: no refit after model selection')
    ax.legend(fontsize=8)
    save_svg(fig, out/'图/co2_models.svg')
    fig.savefig(out/'图/co2_models.png',dpi=150)
    plt.close(fig)
    fig,ax = plt.subplots(figsize=(9,4.8),layout='constrained')
    for d,b in fits.items():
        ax.plot(holdout[:,0],predict(holdout[:,0],b)-holdout[:,1],marker='o',label=f'Degree {d}')
    ax.axhline(0,color='black',lw=.8)
    ax.axvspan(2021.7,2023.3,color='grey',alpha=.14,label='Station transition note')
    ax.set(xlabel='Held-out year',ylabel='Prediction - observed (ppm)',title='Temporal residual structure is part of the result')
    ax.legend(fontsize=8)
    save_svg(fig, out/'图/co2_errors.svg')
    fig.savefig(out/'图/co2_errors.png',dpi=150)
    plt.close(fig)
    (out/'results.json').write_text(json.dumps(result,indent=2,ensure_ascii=False)+'\n')
    return result

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--output', type=Path, default=BASE)
    print(json.dumps(run(parser.parse_args().output),indent=2,ensure_ascii=False))
```

```python
"""Independent exact-rational oracles and input-contract tests."""
import csv
from fractions import Fraction as F
from pathlib import Path
import tempfile
import unittest
import numpy as np
from modeling_lab import (design, fit, predict, metrics, load_data, interval, choose_degree)

class ModelingTests(unittest.TestCase):
    def test_toy_quadratic_exact_oracle(self):
        # Three equations: a-b+c=316, a=332, a+b+c=356.
        beta,_ = fit([1959,1979,1999],[316,332,356],2)
        np.testing.assert_allclose(beta,[332,20,4],rtol=0,atol=1e-10)
        self.assertAlmostEqual(predict([2019],beta)[0],388,places=9)

    def test_toy_linear_exact_oracle(self):
        beta,_ = fit([1959,1979,1999],[316,332,356],1)
        np.testing.assert_allclose(beta,[float(F(1004,3)),20],rtol=0,atol=1e-10)
        m=metrics([316,332,356],predict([1959,1979,1999],beta))
        self.assertAlmostEqual(m['rmse']**2,float(F(32,9)),places=9)

    def test_real_training_fraction_oracle(self):
        # Direct CSV parsing with Fraction, separate from loader and NumPy solver.
        path=Path(__file__).parent/'数据/co2_annmean_mlo.csv'
        with path.open() as f:
            rows=list(csv.DictReader(line for line in f if not line.startswith('#')))
        points=[(F(int(r['year'])-1979,20),F(r['mean'])) for r in rows if 1959 <= int(r['year']) <= 1999]
        n=F(len(points)); s2=sum(x*x for x,y in points); s4=sum(x**4 for x,y in points)
        sy=sum(y for x,y in points); sxy=sum(x*y for x,y in points); sx2y=sum(x*x*y for x,y in points)
        # Symmetry makes sum(x)=sum(x^3)=0. Solve two by two exactly.
        det=n*s4-s2*s2
        exact=[(sy*s4-sx2y*s2)/det,sxy/s2,(n*sx2y-s2*sy)/det]
        data=load_data(path); train=interval(data,1959,1999)
        beta,_=fit(train[:,0],train[:,1],2)
        np.testing.assert_allclose(beta,[float(b) for b in exact],rtol=0,atol=1e-10)
        held=[r for r in rows if 2010 <= int(r['year']) <= 2025]
        errors=[sum(exact[i]*F(int(r['year'])-1979,20)**i for i in range(3))-F(r['mean']) for r in held]
        exact_mse=sum(e*e for e in errors)/len(errors)
        h=interval(data,2010,2025)
        self.assertAlmostEqual(metrics(h[:,1],predict(h[:,0],beta))['rmse']**2,float(exact_mse),places=9)

    def test_sensitivity_analytic(self):
        self.assertAlmostEqual(predict([2025],[332,20,4.1])[0]-predict([2025],[332,20,4])[0],.529,places=10)

    def test_metrics_hand_example(self):
        m=metrics([1,3],[2,1])
        self.assertEqual(m['n'],2)
        self.assertEqual(m['mae'],1.5)
        self.assertEqual(m['bias'],-.5)
        self.assertAlmostEqual(m['rmse'],2.5**.5)

    def test_basis_change_keeps_predictions(self):
        years=[1959,1979,1999,2001]
        values=[316,332,356,359]
        a,_=fit(years,values,2)
        b,_=fit(years,values,2,center=1999,scale=10)
        np.testing.assert_allclose(predict([2005,2010],a),predict([2005,2010],b,1999,10),rtol=0,atol=1e-9)

    def test_fit_rank_and_lengths(self):
        for years,values,degree in [([1,1,1],[2,3,4],2),([1,2],[1],1),([1,2],[1,2],2)]:
            with self.assertRaises(ValueError): fit(years,values,degree)

    def test_design_invalid(self):
        for years,degree,center,scale in [([],1,1979,20),([float('nan')],1,1979,20),([[1]],1,1979,20),([1],3,1979,20),([1],True,1979,20),([1],1,1979,0),([1],1,float('inf'),20),([1],2,0,1e-320)]:
            with self.assertRaises(ValueError): design(years,degree,center,scale)

    def test_predict_invalid(self):
        for beta in [[],[1],[1,2,3,4],[1,float('inf')]]:
            with self.assertRaises(ValueError): predict([2000],beta)
        with self.assertRaises(ValueError): predict([1e308],[1,1e308],0,1)

    def test_metric_invalid(self):
        for y,p in [([],[]),([1],[1,2]),([float('nan')],[1]),([1e308],[-1e308])]:
            with self.assertRaises(ValueError): metrics(y,p)

    def test_loader_contracts(self):
        invalid=['x,y,z\n1,2,3\n','year,mean,unc\n1,2,0\n1,3,0\n','year,mean,unc\n1,-999,0\n',
                 'year,mean,unc\n1,2,-1\n','year,mean,unc\n1.5,2,0\n','year,mean,unc\n1,nan,0\n',
                 'year,mean,unc\n1,foo,0\n','year,mean,unc\n']
        with tempfile.TemporaryDirectory() as temp:
            path=Path(temp)/'bad.csv'
            for text in invalid:
                path.write_text(text)
                with self.assertRaises(ValueError): load_data(path)

    def test_nonfirst_row_field_errors(self):
        # First row is valid: errors in later rows must never be ignored.
        prefix='year,mean,unc\n1959,315.98,.12\n'
        invalid=[prefix+'1960,316.91,.12,unexpected\n',
                 prefix+'1960,316.91\n', prefix+'1960,316.91,\n',
                 prefix+'1960, , .12\n',
                 'year,mean,unc,unc\n1959,315.98,.12,.12\n']
        with tempfile.TemporaryDirectory() as temp:
            path=Path(temp)/'malformed.csv'
            for text in invalid:
                path.write_text(text)
                with self.assertRaises(ValueError): load_data(path)

    def test_intervals_and_official_endpoints(self):
        data=load_data(Path(__file__).parent/'数据/co2_annmean_mlo.csv')
        self.assertEqual(data.shape,(67,3))
        np.testing.assert_array_equal(data[0],[1959,315.98,.12])
        np.testing.assert_array_equal(data[-1],[2025,427.35,.12])
        self.assertEqual(interval(data,1959,1999).shape,(41,3))
        for first,last in [(1958,1999),(2000,2026),(2001,2000),(1959.5,1999)]:
            with self.assertRaises(ValueError): interval(data,first,last)

    def test_selection_predefined_tie(self):
        self.assertEqual(choose_degree({1:1,2:1}),1)
        self.assertEqual(choose_degree({1:2,2:1}),2)
        self.assertEqual(choose_degree({1:1+5e-10,2:1}),1)
        for scores in [{1:1},{1:-1,2:1},{1:float('nan'),2:1}]:
            with self.assertRaises(ValueError): choose_degree(scores)

if __name__ == '__main__': unittest.main()
```
