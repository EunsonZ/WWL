package org.wwl.intentdemo

import android.content.pm.ActivityInfo
import android.content.res.Configuration
import androidx.test.core.app.ActivityScenario
import androidx.test.ext.junit.runners.AndroidJUnit4
import androidx.test.platform.app.InstrumentationRegistry
import org.junit.Assert.*
import org.junit.Test
import org.junit.runner.RunWith
import java.util.UUID

@RunWith(AndroidJUnit4::class)
class ActivityStateTest {
    private val repo get()=(InstrumentationRegistry.getInstrumentation().targetContext.applicationContext as IntentApp).repository
    private fun waitFor(condition:()->Boolean) { val until=System.nanoTime()+8_000_000_000;while(System.nanoTime()<until){if(condition())return;Thread.sleep(40)};error("bounded state wait failed") }
    @Test fun immediateSendWhilePersistingDoesNotSendOldDisplayedIntent() {
        val oldId="old-${UUID.randomUUID()}";val newId="new-${UUID.randomUUID()}"
        ActivityScenario.launch(MainActivity::class.java).use { scenario ->
            scenario.onActivity { a -> a.startInput.setText("690");a.endInput.setText("720");a.queueDraft(oldId) }
            waitFor { repo.blocking { client.current()?.id==oldId } }
            // Let the old identity reach the actual screen, then make two same-main-thread calls.
            waitFor { var ok=false;scenario.onActivity { a -> ok=a.lastDisplayedMessage.contains("本地待发送") };ok }
            val before=repo.blocking { authority.snapshot().toString() }
            scenario.onActivity { a ->
                a.startInput.setText("810");a.endInput.setText("840");a.queueDraft(newId)
                assertFalse(a.startInput.isEnabled)
                a.attempt("normal") // Must not send oldId or invalidate the new queue's callback.
            }
            waitFor { repo.blocking { client.current()?.id==newId } }
            waitFor { var ok=false;scenario.onActivity { a -> ok=a.startInput.isEnabled&&a.lastDisplayedMessage.contains("本地待发送") };ok }
            assertEquals(before,repo.blocking { authority.snapshot().toString() })
            assertEquals("queued",repo.blocking { client.get("D",oldId)!!.phase })
            assertEquals("queued",repo.blocking { client.get("D",newId)!!.phase })
        }
    }
    @Test fun recreateAndRealOrientationKeepDurableIdentityAndBundleDraft() {
        val id="activity-${UUID.randomUUID()}"
        ActivityScenario.launch(MainActivity::class.java).use { scenario ->
            waitFor { repo.blocking { client.current()!=null || client.draft().first=="690" } }
            scenario.onActivity { a -> a.startInput.setText("690");a.endInput.setText("720");a.queueDraft(id) }
            waitFor { repo.blocking { client.current()?.id==id } }
            val fp=repo.blocking { client.current()!!.fp }
            scenario.onActivity { a -> a.startInput.setText("bad") }
            scenario.recreate()
            scenario.onActivity { a -> assertEquals("bad",a.startInput.text.toString()) }
            assertEquals(id,repo.blocking { client.current()!!.id });assertEquals(fp,repo.blocking { client.current()!!.fp })
            var wanted=Configuration.ORIENTATION_LANDSCAPE
            scenario.onActivity { a ->
                wanted=if(a.resources.configuration.orientation==Configuration.ORIENTATION_LANDSCAPE)Configuration.ORIENTATION_PORTRAIT else Configuration.ORIENTATION_LANDSCAPE
                a.requestedOrientation=if(wanted==Configuration.ORIENTATION_LANDSCAPE)ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE else ActivityInfo.SCREEN_ORIENTATION_PORTRAIT
            }
            waitFor { var ok=false;scenario.onActivity { a -> ok=a.resources.configuration.orientation==wanted };ok }
            scenario.onActivity { a -> assertEquals(wanted,a.resources.configuration.orientation);assertEquals("bad",a.startInput.text.toString()) }
            assertEquals(id,repo.blocking { client.current()!!.id });assertEquals(fp,repo.blocking { client.current()!!.fp })
            assertEquals("queued",repo.blocking { client.current()!!.phase })
        }
    }
}

import androidx.test.ext.junit.runners.AndroidJUnit4
import androidx.test.platform.app.InstrumentationRegistry
import org.json.JSONObject
import org.junit.After

class StoreInstrumentedTest {

    private val clientName="test-client-${UUID.randomUUID()}.db"
    private val authorityName="test-authority-${UUID.randomUUID()}.db"
    private lateinit var client:ClientStore
    private lateinit var authority:MockAuthority
    @Before fun open() { client=ClientStore(context,clientName);authority=MockAuthority(context,authorityName) }
    @After fun close() { client.close();authority.close();context.deleteDatabase(clientName);context.deleteDatabase(authorityName) }

    @Test fun queuedBeforeAnySendSurvivesReopen() {
        val p=client.queue("D","persist-only",690,720);reopen();val q=client.current()!!
        assertEquals(p.id,q.id);assertEquals(p.fp,q.fp);assertEquals("queued",q.phase)
        assertEquals(3,authority.snapshot().getJSONArray("bookings").length())
        assertEquals("unknown",authority.query("D",p.id,p.fp).getString("status"))
    }
    @Test fun committedLostReplyReopensAndReplaysOriginalDecision() {
        val p=client.queue("D","lost-reply",690,720);client.phase(p,"sending")
        val original=authority.reserve(p) // transport deliberately does not deliver this result
        reopen();val q=client.current()!!;assertEquals("unknown",q.phase);assertEquals(p.id,q.id);assertEquals(p.fp,q.fp)
        val queried=authority.query(q.actor,q.id,q.fp);assertEquals(original.toString(),queried.toString())
        client.accept(q,queried);val replay=authority.reserve(q,writerEpoch=1)
        assertEquals(original.toString(),replay.toString());assertEquals(4L,replay.getJSONObject("booking").getLong("id"))
        assertEquals(4,authority.snapshot().getJSONArray("bookings").length());assertEquals(1,authority.snapshot().getJSONArray("outbox").length())
        assertEquals("accepted",client.current()!!.phase)
    }
    @Test fun rollbackIsUnknownNotFinalRejection() {
        val p=client.queue("D","before-commit",690,720)
        try { authority.reserve(p,beforeCommit=true);fail("expected fault") } catch(_:java.io.IOException) {}
        reopen();val query=authority.query(p.actor,p.id,p.fp)
        assertEquals("unknown",query.getString("status"));assertEquals("not_registered",query.getString("reason"));assertFalse(query.has("fingerprint"))
        assertEquals(3,authority.snapshot().getJSONArray("bookings").length())
        val accepted=authority.reserve(p);client.accept(p,accepted);assertEquals("accepted",client.current()!!.phase)
    }
    @Test fun bufferAndInvalidIntervalAreDurableBusinessRejections() {
        val p=client.queue("D","buffer",630,660);val r=authority.reserve(p);client.accept(p,r)
        assertEquals("conflict_with_buffer",r.getString("reason"));assertEquals("rejected",client.current()!!.phase)
        val invalid=client.queue("D","invalid",690,719);assertEquals("invalid_interval",authority.reserve(invalid).getString("reason"))
        assertEquals("690" to "719",client.draft());assertEquals(3,authority.snapshot().getJSONArray("bookings").length())
    }
    @Test fun sameKeyDifferentParametersCannotReplaceOriginal() {
        val p=client.queue("D","same-key",690,720);val saved=authority.reserve(p);client.accept(p,saved)
        try { client.queue("D",p.id,700,730);fail("expected conflict") } catch(_:IllegalArgumentException) {}
        val changed=p.copy(start=700,end=730,fp=Protocol.fingerprint("D",700,730))
        try { authority.reserve(changed);fail("expected authority conflict") } catch(_:IllegalArgumentException) {}
        assertEquals(p.fp,client.current()!!.fp);assertEquals(saved.toString(),authority.query("D",p.id,p.fp).toString())
        assertEquals("690" to "720",client.draft())
    }
    @Test fun malformedResultsAreNotDisplayedAsAccepted() {
        val p=client.queue("D","shape",690,720);val r=authority.reserve(p)
        for(bad in listOf(JSONObject(r.toString()).put("fingerprint","bad"),JSONObject(r.toString()).put("policy_version",2.0),JSONObject(r.toString()).put("intent_id","other"))) {
            try { client.accept(p,bad);fail("expected contract failure") } catch(_:IllegalArgumentException) {}
            assertEquals("queued",client.current()!!.phase)
        }
        client.accept(p,r)
        val reordered=JSONObject();for(k in r.keys().asSequence().toList().reversed())reordered.put(k,r.get(k))
        client.accept(p,reordered);assertEquals("accepted",client.current()!!.phase)
    }
}

package org.wwl.intentdemo

import android.app.Application
import android.os.Handler
import android.os.Looper
import android.os.Process
import org.json.JSONObject
import java.io.File
import java.util.concurrent.Callable
import java.util.concurrent.ScheduledThreadPoolExecutor
import java.util.concurrent.TimeUnit

class IntentApp:Application() {
    lateinit var repository:Repository
        private set
    override fun onCreate() { super.onCreate();repository=Repository(this) }
}

class Repository(private val app:Application) {
    val worker=ScheduledThreadPoolExecutor(1)
    val client=ClientStore(app)
    val authority=MockAuthority(app)
    private val main=Handler(Looper.getMainLooper())
    init { worker.execute { client.recoverInflight();authority.readableDatabase;exportState() } }
    fun <T> blocking(block:Repository.()->T):T { check(Looper.myLooper()!=Looper.getMainLooper());return worker.submit(Callable { block(this) }).get(15,TimeUnit.SECONDS) }
    fun audit(kind:String,detail:String) { worker.execute { client.audit(kind,detail);exportState() } }
    fun exportState():JSONObject {
        val state=JSONObject().put("pid",Process.myPid()).put("client",client.snapshot()).put("mock_authority",authority.snapshot())
        File(app.filesDir,"store-state.json").writeText(state.toString(2));return state
    }
    fun recordUi(value:JSONObject) { worker.execute { File(app.filesDir,"ui-state.json").writeText(value.toString(2));exportState() } }
    fun refresh(callback:(JSONObject)->Unit) { worker.execute { val s=exportState();main.post { callback(s) } } }
    fun queue(start:Int,end:Int,id:String,callback:(Result<JSONObject>)->Unit) { worker.execute {
        val r=runCatching { client.queue("D",id,start,end);exportState() };main.post { callback(r) }
    } }
    fun saveDraft(start:String,end:String,callback:(Result<JSONObject>)->Unit) { worker.execute {
        val r=runCatching { client.saveDraft(start,end);client.audit("draft_saved","explicit action");exportState() };main.post { callback(r) }
    } }
    fun attempt(actor:String,id:String,mode:String,callback:(String,Result<JSONObject>)->Unit) {
        require(mode in setOf("normal","drop","before","delay","recover"))
        worker.execute {
            val p=client.get(actor,id) ?: run { main.post { callback("$actor\u0000$id",Result.failure(IllegalStateException("missing persisted intent"))) };return@execute }
            var response:JSONObject?=null
            val error=runCatching {
                client.phase(p,"sending");client.audit("send_same_identity",p.id)
                if(mode=="recover") {
                    response=authority.query(p.actor,p.id,p.fp)
                    if(response!!.getString("status")=="unknown")response=authority.reserve(p)
                } else response=authority.reserve(p,beforeCommit=mode=="before")
                if(mode=="drop")throw java.io.IOException("lab committed reply lost")
            }.exceptionOrNull()
            if(error!=null) {
                client.phase(p,"unknown");client.audit("attempt_uncertain",p.id);val state=exportState()
                main.post { callback(p.key,Result.success(state)) };return@execute
            }
            val deliver={
                val result=runCatching { client.accept(p,response!!);exportState() }
                if(result.isFailure) { client.phase(p,"protocol_error");exportState() }
                main.post { callback(p.key,result) }
            }
            if(mode=="delay") {
                client.audit("reply_held",p.id);exportState()
                worker.schedule({ deliver() },3,TimeUnit.SECONDS)
            } else deliver()
        }
    }
}

import android.os.Build

import java.util.UUID

/** No configChanges shortcut: normal Android recreation is part of this lesson. */
class MainActivity:Activity() {
    private val repo:Repository get()=(application as IntentApp).repository
    lateinit var startInput:EditText
        private set
    lateinit var endInput:EditText
        private set
    lateinit var statusView:TextView
        private set
    private lateinit var detail:TextView
    private var generation=0L
    private var foreground=false
    private var savingIntent=false
    private val actions=mutableListOf<Button>()
    private var initializedDraft=false
    private var shownActor:String?=null
    private var shownId:String?=null
    private var lastState:JSONObject?=null
    private val instance=UUID.randomUUID().toString()
    private var pendingCommand:Intent?=null
    var lastDisplayedMessage:String="正在读取持久记录，尚未确认新预约。"
        private set
    private fun ticket():Long { generation++;return generation }
    private fun allowed(t:Long,key:String?=null):Boolean = foreground&&!isDestroyed&&t==generation&&
        (key==null||key=="$shownActor\u0000$shownId")

    override fun onCreate(savedInstanceState:Bundle?) {
        super.onCreate(savedInstanceState)
        val body=LinearLayout(this).apply { orientation=LinearLayout.VERTICAL;setPadding(24,16,24,24) }
        body.setOnApplyWindowInsetsListener { view,insets ->
            if(Build.VERSION.SDK_INT>=30) { val bars=insets.getInsets(WindowInsets.Type.systemBars());view.setPadding(24+bars.left,16+bars.top,24+bars.right,24+bars.bottom) }

            insets
        }
        fun text(value:String)=TextView(this).apply { text=value;textSize=18f;body.addView(this) }
        text("本地假服务：意图恢复教学")
        text("仅合成D的一天预约，无HTTP/真实身份。离线/丢回复是故障开关，不是实际网络切换。")
        text("开始分钟")
        startInput=EditText(this).apply { id=1001;contentDescription="开始分钟";inputType=InputType.TYPE_CLASS_NUMBER;filters=arrayOf(InputFilter.LengthFilter(32));setText(savedInstanceState?.getString("start_raw")?:"690");body.addView(this) }
        text("结束分钟")

        initializedDraft=savedInstanceState!=null
        statusView=text(lastDisplayedMessage).apply { id=1003;accessibilityLiveRegion=View.ACCESSIBILITY_LIVE_REGION_POLITE }
        fun button(label:String,action:()->Unit) { body.addView(Button(this).apply { text=label;setOnClickListener { action() };actions.add(this) }) }
        button("保存草稿（落盘）") { saveDraft() }
        button("新意图：仅本地排队") { queueDraft() }
        button("发送：正常回复") { attempt("normal") }
        button("发送：提交后丢回复") { attempt("drop") }
        button("发送：提交前回滚") { attempt("before") }
        button("发送：延迟回复3秒") { attempt("delay") }
        button("恢复：查询/同意图重放") { attempt("recover") }
        button("刷新本地记录") { refresh() }
        detail=text("读取中")
        setContentView(ScrollView(this).apply { addView(body) })
        if(BuildConfig.DEBUG&&savedInstanceState==null&&intent.hasExtra("lab_command"))pendingCommand=Intent(intent)
        // Test-control extras are not a durable business command and must not repeat on recreation.
        setIntent(Intent(intent).apply { for(key in listOf("lab_command","start_raw","end_raw","intent_id"))removeExtra(key) })
        repo.audit("activity_create","pid=${Process.myPid()} instance=$instance")
    }
    override fun onResume() { super.onResume();foreground=true;repo.audit("activity_resume",instance);refresh() }
    override fun onPause() { foreground=false;generation++;repo.audit("activity_pause",instance);super.onPause() }
    override fun onStop() { repo.audit("activity_stop",instance);super.onStop() }
    override fun onDestroy() { foreground=false;generation++;repo.audit("activity_destroy",instance);super.onDestroy() }
    override fun onSaveInstanceState(outState:Bundle) {
        outState.putString("start_raw",startInput.text.toString());outState.putString("end_raw",endInput.text.toString())
        super.onSaveInstanceState(outState)
    }
    override fun onNewIntent(intent:Intent) { super.onNewIntent(intent)
        if(BuildConfig.DEBUG&&intent.hasExtra("lab_command")) { pendingCommand=Intent(intent);refresh() }
    }
    private fun message(value:String) { lastDisplayedMessage=value;statusView.text=value }
    fun refresh() {
        val t=ticket()
        repo.refresh loaded@{ state ->
            if(!allowed(t)) { repo.audit("ui_ignored","refresh:$instance:$t");return@loaded }
            render(state)
            pendingCommand?.let { cmd -> pendingCommand=null;runLabCommand(cmd) }
        }
    }
    private fun render(state:JSONObject) {
        lastState=state
        val c=state.getJSONObject("client")
        if(!initializedDraft) { val d=c.getJSONObject("draft");startInput.setText(d.getString("start_raw"));endInput.setText(d.getString("end_raw"));initializedDraft=true }
        val current=c.optJSONObject("current");shownActor=current?.getString("actor");shownId=current?.getString("intent_id")
        val phase=current?.getString("phase")
        message(when(phase) {
            "queued" -> "本地待发送：尚无服务承诺。"
            "sending" -> "正在等待：结果尚未确认；离开界面不等于撤销。"
            "unknown" -> "结果未知：保留原意图和输入，请查询/同意图重放。"
            "accepted" -> "已确认：假权威保存了原意图的预约结果。"
            "rejected" -> "已拒绝：${current.getJSONObject("result").getString("reason")}；输入保留。"
            "protocol_error" -> "协议错误：保留原意图，不猜测最终结果。"
            else -> "没有待显示意图；草稿不是已预约。"
        })
        detail.text="actor=D\nintent_id=${shownId?:"无"}\nfp=${current?.getString("fingerprint")?:"无"}\n本地状态=${phase?:"无"}\n假权威预约数=${state.getJSONObject("mock_authority").getJSONArray("bookings").length()}\n当前结果=${current?.opt("result")?:"无"}"
        recordUi()
    }
    private fun recordUi() {
        repo.recordUi(JSONObject().put("pid",Process.myPid()).put("activity_instance",instance).put("generation",generation)
            .put("foreground",foreground).put("actor",shownActor?:JSONObject.NULL).put("intent_id",shownId?:JSONObject.NULL)
            .put("start_raw",startInput.text.toString()).put("end_raw",endInput.text.toString()).put("message",lastDisplayedMessage)
            .put("phase",lastState?.getJSONObject("client")?.optJSONObject("current")?.optString("phase")?:JSONObject.NULL))
    }
    private fun saving(value:Boolean) {
        savingIntent=value
        actions.forEach { it.isEnabled=!value };startInput.isEnabled=!value;endInput.isEnabled=!value
    }
    private fun rejectWhileSaving():Boolean {
        if(!savingIntent)return false
        message("新意图正在持久保存，请等待完成；没有发送旧意图。");recordUi();return true
    }
    fun queueDraft(id:String=UUID.randomUUID().toString()) {
        if(rejectWhileSaving())return
        val start=startInput.text.toString().toIntOrNull();val end=endInput.text.toString().toIntOrNull()
        if(start==null||end==null) { message("输入错误：需要整数分钟；原输入和已有意图保留。");recordUi();return }
        // Invalid intervals remain a service business rejection; no local success is invented.
        try { Protocol.text(id) } catch(_:IllegalArgumentException) { message("输入错误：意图id无效；原记录保留。");recordUi();return }
        val t=ticket();saving(true);message("正在持久保存新意图；保存失败则不发送。")
        repo.queue(start,end,id) { r ->
            saving(false)
            if(!allowed(t)) { repo.audit("ui_ignored","queue:$instance:$t");return@queue }
            r.onSuccess { render(it) }.onFailure { message("保存失败：未发送，保留输入及原记录。");recordUi() }
        }
    }
    fun saveDraft() {
        if(rejectWhileSaving())return
        val t=ticket();val s=startInput.text.toString();val e=endInput.text.toString()

            r.onSuccess { render(it) }.onFailure { message("协议/存储异常：原意图保留，需恢复查询。");recordUi() }
        }
    }

package org.wwl.intentdemo

import java.security.MessageDigest

/** Same canonical fields/UTF8 as SE82. This is a parameter checksum, not authentication. */
object Protocol {
    const val NAME = "wwl-reservation-v2/1"
    fun text(value: String): String {
        val count = value.codePointCount(0, value.length)
        // Python str.isspace additionally includes NEL and the four record separators.
        require(count in 1..128 && value.codePoints().anyMatch { cp ->
            !(Character.isWhitespace(cp) || Character.isSpaceChar(cp) || cp == 0x85 || cp in 0x1c..0x1f)
        }) { "nonblank text up to 128 Unicode codepoints required" }
        require(value.indices.none { i -> Character.isSurrogate(value[i]) &&
            !(Character.isHighSurrogate(value[i]) && i+1 < value.length && Character.isLowSurrogate(value[i+1])) &&
            !(Character.isLowSurrogate(value[i]) && i>0 && Character.isHighSurrogate(value[i-1])) }) { "valid Unicode scalar sequence required" }
        return value
    }
    fun quote(value: String): String = buildString {
        append('"')
        for (c in value) when(c) {
            '"' -> append("\\\""); '\\' -> append("\\\\")
            '\b' -> append("\\b"); '\u000c' -> append("\\f"); '\n' -> append("\\n"); '\r' -> append("\\r"); '\t' -> append("\\t")
            else -> if(c.code < 32) append("\\u%04x".format(c.code)) else append(c)
        }
        append('"')
    }
    fun canonical(actor: String, start: Int, end: Int, policy: Int = 2): String {
        text(actor)
        return "{\"actor\":" + quote(actor) + ",\"end\":$end,\"op\":\"reserve\",\"policy_version\":$policy,\"start\":$start}"
    }
    fun fingerprint(actor: String, start: Int, end: Int, policy: Int = 2): String =
        MessageDigest.getInstance("SHA-256").digest(canonical(actor,start,end,policy).toByteArray(Charsets.UTF_8))
            .joinToString("") { "%02x".format(it.toInt() and 255) }
    fun validInterval(start: Int, end: Int): Boolean = start >= 540 && end <= 1020 && end > start && end.toLong()-start in 30L..120L
    fun compatible(start: Int, end: Int, oldStart: Int, oldEnd: Int): Boolean =
        end.toLong()+10 <= oldStart || oldEnd.toLong()+10 <= start
}

                check("same intent parameters/fingerprint and draft survived rotation", (current(rotated)["actor"], current(rotated)["intent_id"], current(rotated)["start"], current(rotated)["end"], current(rotated)["fingerprint"], rotated_ui["start_raw"], rotated_ui["end_raw"]) == ("D", FIRST, 690, 720, FP, "690", "720"))

                (output / "rotation-before.txt").write_text(before_display + "\n")
                (output / "rotation-after.txt").write_text(after_display + "\n")
                rotation = {"original_mode": original_mode, "pid_unchanged": rotated["pid"], "before_degrees": old_rotation, "after_degrees": actual_rotation,

                time.sleep(.25)  # Let the display's transition frame settle; not a timing-performance assertion.
                screen("01b-same-intent-rotated")
            finally:
                call("shell", "wm", "user-rotation", *original_mode.split(), allow_fail=True)
            restored, restored_ui = snap("01c-same-intent-orientation-restored", lambda s, u: u["foreground"] and u["activity_instance"] != rotated_ui["activity_instance"] and current(s)["intent_id"] == u["intent_id"] == FIRST and u["phase"] == "queued")
            check("restored only this emulator's prior rotation mode", call("shell", "wm", "user-rotation") == original_mode)

        wait(lambda s, u: s["client"]["draft"]["start_raw"] == "bad" and "草稿已落盘" in u["message"])

        call("shell", "am", "force-stop", PACKAGE)
        check("explicit force-stop removed process", not call("shell", "pidof", PACKAGE, allow_fail=True))
        command()
        durable, ui = snap("08-force-stop-durable-draft", lambda s, u: s["pid"] != before_stop_pid and s["pid"] == u["pid"] and u["start_raw"] == "bad" and u["intent_id"] == NEXT)
        check("explicit saved raw draft survives force-stop", durable["client"]["draft"] == {"start_raw": "bad", "end_raw": "840"} and ui["start_raw"] == "bad")
        check("editing draft did not mutate pending normalized intent", (current(durable)["start"], current(durable)["end"], current(durable)["phase"]) == (810, 840, "queued"))
        check("all original effects and old accepted outcome still present", len(durable["mock_authority"]["bookings"]) == 4 and next(x for x in durable["client"]["intents"] if x["intent_id"] == FIRST)["result"] == original)
        screen("08-new-pending-saved-error")
        # Read database files only after stopping this synthetic package. Verify the exported contents separately.

        for name in ["client.db", "mock-authority.db"]:

        report = {"scope": "synthetic single-device function/JSON mock; no actual network or human task", "versions": versions,

        (output / "results.json").write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n")
        print(f"PASS {len(checks)} lifecycle/identity assertions; process {old_pid} -> {restarted['pid']}")
    finally:
        (output / "commands.json").write_text(json.dumps(commands, ensure_ascii=False, indent=2) + "\n")
        (output / "assertions.json").write_text(json.dumps(checks, ensure_ascii=False, indent=2) + "\n")
