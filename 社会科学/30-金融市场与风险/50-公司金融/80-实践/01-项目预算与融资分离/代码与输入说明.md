#!/usr/bin/env python3
"""Finite synthetic capital budgeting, cash counterparties and claim pricing.
Python >=3.9, standard library; exact Fraction arithmetic; no real-data inference.
"""
import argparse
import copy
from fractions import Fraction as F
from html import escape
import itertools
import json
from math import isqrt
from pathlib import Path


def rat(x):
    if isinstance(x, bool) or not isinstance(x, (str,int)):
        raise ValueError('rational inputs must be strings or integers')
    return F(x)


def npv(cf,r):
    if r <= -1: raise ValueError('gross discount factor must be positive')
    return sum(v/(1+r)**t for t,v in enumerate(cf))


def financing(raw,loan,fee):
    I,X,r=(rat(raw[k]) for k in ('investment','payoff','rate'))
    if I <= 0 or X < 0 or r <= -1 or loan < 0 or fee < 0:
        raise ValueError('invalid deterministic investment/contract')
    H=(1+r)*loan
    if H > X: raise ValueError('project cannot fully repay riskfree debt without a guarantee')
    equity=[-I-fee+loan,X-H]; debt=[-loan,H]
    counterparties=dict(supplier=[I,F(0)],customer=[F(0),-X],
                        equity=equity,debt=debt,service_provider=[fee,F(0)])
    balances=[sum(c[t] for c in counterparties.values()) for t in range(2)]
    assert balances == [0,0]
    project=npv([-I,X],r); equity_npv=npv(equity,r)
    assert npv(debt,r) == 0 and equity_npv == project-fee
    return dict(loan=loan,fee=fee,repayment=H,project_cf=[-I,X],
                project_value=X/(1+r),project_npv=project,equity_cf=equity,
                equity_claim_value=(X-H)/(1+r),equity_npv=equity_npv,
                debt_cf=debt,debt_npv=npv(debt,r),counterparties=counterparties,
                date_balances=balances,
                equity_cash_return=(equity[1]/(-equity[0])-1 if equity[0] < 0 else None))


def operating(raw,utilization1=None,receivable=None):
    capex,tax,opp,sunk,S,r=(rat(raw[k]) for k in
                         ('capex','tax_rate','opportunity_cash0','sunk_already_paid','salvage','rate'))
    keys=('capacity','utilization','price','unit_cash_cost','fixed_cash_cost','depreciation')
    series={k:[rat(v) for v in raw[k]] for k in keys}
    if any(len(v)!=2 for v in series.values()):raise ValueError('two operating years required')
    if utilization1 is not None:series['utilization'][0]=utilization1
    other=[rat(v) for v in raw['other_nwc']]
    ar=rat(raw['delayed_receivable']) if receivable is None else receivable
    if (capex <= 0 or not 0 <= tax <= 1 or min(opp,sunk,S) < 0 or r <= -1
        or len(other)!=3 or min(other) < 0 or other[-1]!=0
        or min(series['depreciation']) < 0 or sum(series['depreciation']) > capex):
        raise ValueError('invalid tax/assets/NWC/discount contract')
    if (min(series['capacity']+series['price']+series['unit_cash_cost']+series['fixed_cash_cost']) < 0
        or any(not 0 <= u <= 1 for u in series['utilization'])):
        raise ValueError('nonnegative capacity/costs and utilization in [0,1] required')
    tasks=[a*b for a,b in zip(series['capacity'],series['utilization'])]
    revenue=[a*b for a,b in zip(tasks,series['price'])]
    if ar < 0 or ar > revenue[0]:raise ValueError('receivable must be part of year-1 revenue')
    costs=[n*v+f for n,v,f in zip(tasks,series['unit_cash_cost'],series['fixed_cash_cost'])]
    book=capex-sum(series['depreciation']);sale_tax=tax*max(S-book,F(0));net_sale=S-sale_tax
    nwc=[other[0],other[1]+ar,F(0)]
    receipts=[revenue[0]-ar,revenue[1]+ar]
    cf=[-capex-other[0]-opp];direct=[cf[0]];rows=[]
    for k in range(2):
        ebit=revenue[k]-costs[k]-series['depreciation'][k]
        T=tax*max(ebit,F(0));ocf=ebit-T+series['depreciation'][k]
        dnwc=nwc[k+1]-nwc[k];dother=other[k+1]-other[k]
        sale=net_sale if k==1 else F(0)
        accrual=ocf-dnwc+sale
        cash=receipts[k]-costs[k]-T-dother+sale
        assert accrual == cash
        cf.append(accrual);direct.append(cash)
        rows.append(dict(t=k+1,tasks=tasks[k],revenue=revenue[k],cash_receipts=receipts[k],
                         cash_cost=costs[k],depreciation=series['depreciation'][k],EBIT=ebit,
                         operating_tax=T,operating_cash=ocf,delta_total_NWC=dnwc,
                         delta_other_NWC=dother,net_salvage=sale,FCF=accrual,direct_FCF=cash))
    # A decision counterfactual: doing nothing pays the rent; historic research is in both histories.
    with_cash=[-capex-other[0]]+[v['direct_FCF'] for v in rows]
    without_cash=[opp,F(0),F(0)]
    difference=[a-b for a,b in zip(with_cash,without_cash)]
    assert difference == cf == direct
    terminal_wealth=sum(v*(1+r)**(2-t) for t,v in enumerate(cf))
    assert terminal_wealth == npv(cf,r)*(1+r)**2
    return dict(tasks=tasks,receivable=ar,total_NWC=nwc,other_NWC=other,
                capex=capex,opportunity_cash0=opp,sunk_excluded=sunk,tax_book_end=book,
                sale_tax=sale_tax,net_salvage=net_sale,rows=rows,CF=cf,direct_CF=direct,
                with_project_cash=with_cash,without_project_cash=without_cash,
                incremental_difference=difference,actual_cash_needed0=capex+other[0],
                NPV=npv(cf,r),terminal_wealth=terminal_wealth,rate=r)


def ai_case(raw,u):
    Q,p,c,f,I,S,r=(rat(raw[k]) for k in
                 ('reference_capacity','price','unit_cash_cost','fixed_cash_cost','investment','salvage','rate'))
    a=[rat(v) for v in raw['closed_loop_fraction']]
    if len(a)!=2 or any(not 0 <= x <= 1 for x in a+[u]) or min(Q,p,c,f,I,S) < 0 or r <= -1:
        raise ValueError('invalid fixed AI application input')
    n=[Q*x*u for x in a];cash=[(p-c)*v-f for v in n]
    cf=[-I,cash[0],cash[1]+S]
    return dict(utilization=u,tasks=n,operating_cash=cash,salvage=S,CF=cf,NPV=npv(cf,r))


def project_table(raws,r):
    result=[]
    for q in raws:
        cost,pay=rat(q['cost']),rat(q['payoff'])
        if cost <= 0 or pay < 0:raise ValueError('project cost positive and payoff nonnegative')
        pv=pay/(1+r)
        result.append(dict(name=q['name'],cost=cost,payoff=pay,PV=pv,NPV=pv-cost,
                           IRR=(pay/cost-1 if pay else None),PI=pv/cost))
    if len(set(q['name'] for q in result)) != len(result):raise ValueError('duplicate project names')
    return result


def exclusive_choices(projects,caps):
    choices=[]
    for cap in caps:
        if cap < 0:raise ValueError('cash cap nonnegative')
        feasible=[dict(names=[],cost=F(0),NPV=F(0))]+[
            dict(names=[p['name']],cost=p['cost'],NPV=p['NPV']) for p in projects if p['cost'] <= cap]
        best=max(z['NPV'] for z in feasible)
        choices.append(dict(cap=cap,best_value=best,winners=[z for z in feasible if z['NPV']==best]))
    return choices


def budgets(projects,caps):
    result=[]
    for cap in caps:
        if cap < 0:raise ValueError('cash budget nonnegative')
        rows=[]
        for bits in itertools.product((0,1),repeat=len(projects)):
            cost=sum(p['cost']*b for p,b in zip(projects,bits))
            value=sum(p['NPV']*b for p,b in zip(projects,bits))
            rows.append(dict(names=[p['name'] for p,b in zip(projects,bits) if b],
                             cost=cost,NPV=value,feasible=cost<=cap))
        best=max(z['NPV'] for z in rows if z['feasible'])
        chosen=[];remaining=cap
        for p in sorted(projects,key=lambda x:(-x['PI'],x['name'])):
            if p['NPV'] > 0 and p['cost'] <= remaining:
                chosen.append(p);remaining-=p['cost']
        result.append(dict(cap=cap,all_subsets=rows,best_value=best,
                           winners=[z for z in rows if z['feasible'] and z['NPV']==best],
                           ratio_greedy_names=[p['name'] for p in chosen],
                           ratio_greedy_value=sum(p['NPV'] for p in chosen)))
    return result


def quadratic_irr(cf,rates):
    if len(cf)!=3 or cf[0]==0:raise ValueError('exact IRR example needs three flows with nonzero CF0')
    a,b,c=cf;disc=b*b-4*a*c
    if disc < 0:return dict(CF=cf,discriminant=disc,IRRs=[],tests=[dict(rate=x,NPV=npv(cf,x)) for x in rates])
    n,d=isqrt(disc.numerator),isqrt(disc.denominator)
    if n*n!=disc.numerator or d*d!=disc.denominator:
        raise ValueError('this exact finite IRR checker requires a rational-square discriminant')
    root=F(n,d);zs=sorted(set([(-b-root)/(2*a),(-b+root)/(2*a)]))
    rs=[z-1 for z in zs if z>0]
    for r in rs:assert npv(cf,r)==0
    assert all(a*z*z+b*z+c==0 for z in zs)
    return dict(CF=cf,discriminant=disc,IRRs=rs,
                tests=[dict(rate=x,NPV=npv(cf,x)) for x in rates])


def risk_case(raw,loan):
    P,pi,X=([rat(v) for v in raw[k]] for k in ('probabilities','state_prices','payoffs'))
    I,df=rat(raw['investment']),rat(raw['bank_price'])
    if (len(P)!=len(pi) or len(P)!=len(X) or not P or min(P+pi)<=0
        or sum(P)!=1 or min(X)<0 or I<=0 or df<=0 or sum(pi)!=df or loan<0):
        raise ValueError('positive probabilities/state prices and correctly priced bank required')
    rf=1/df-1;H=loan/df
    if H>min(X):raise ValueError('riskfree debt exceeds the low-state repayment capacity')
    d=[H for _ in X];e=[x-H for x in X]
    price=lambda ys:sum(q*y for q,y in zip(pi,ys))
    mean=lambda ys:sum(p*y for p,y in zip(P,ys))
    V,D,E=price(X),price(d),price(e)
    if V<=0:raise ValueError('positive operating asset value required')
    assert D==loan and D+E==V and all(a+b==x for a,b,x in zip(d,e,X))
    e_npv=E-(I-loan)
    assert e_npv==V-I
    rA=mean(X)/V-1;rE=(mean(e)/E-1 if E else None)
    if E:
        assert rE==rA+D/E*(rA-rf)
        assert mean(e)/(1+rE)==E
    var=mean([(x-mean(X))**2 for x in X])
    return dict(loan=loan,probabilities=P,state_prices=pi,project_payoffs=X,
                debt_payoffs=d,equity_payoffs=e,V=V,D=D,E=E,investment=I,
                equity_cash_contribution=I-loan,project_NPV=V-I,equity_NPV=e_npv,
                bank_rate=rf,asset_expected_return=rA,equity_expected_return=rE,
                debt_expected_return=(mean(d)/D-1 if D else None),
                wrong_equity_value_at_asset_rate=mean(e)/(1+rA),
                project_return_variance=var/V**2,
                equity_return_variance=(var/E**2 if E else None),
                max_riskfree_loan=min(X)*df)


def compute(inp):
    d=inp['deterministic'];loans=[rat(v) for v in d['loans']]
    finance=[financing(d,L,F(0)) for L in loans]
    fee_case=financing(d,rat(d['fee_case_loan']),rat(d['fee']))
    migration_finance=financing(d,rat(d['migration_loan']),rat(d['migration_fee']))
    boundaries=[]
    for L in [rat(v) for v in d['boundary_loans']]:
        try:v=financing(d,L,rat(d['migration_fee']));boundaries.append(dict(loan=L,accepted=True,result=v))
        except ValueError as e:boundaries.append(dict(loan=L,accepted=False,reason=str(e)))
    o=inp['operating'];base=operating(o)
    delayed=operating(o,receivable=rat(o['migration_receivable']))
    migration_o=operating(o,rat(o['migration_utilization1']),rat(o['migration_receivable']))
    # Research spent before either decision history cancels, even if its amount changes.
    altered=copy.deepcopy(o);altered['sunk_already_paid']=str(rat(o['sunk_already_paid'])+100)
    assert operating(altered)['NPV']==base['NPV']
    ai=inp['existing_ai']
    se=inp['selection'];r=rat(se['rate'])
    if r<=-1:raise ValueError('selection rate must exceed -1')
    ex=project_table(se['exclusive'],r);exm=project_table(se['migration_exclusive'],r)
    if len(ex)!=2:raise ValueError('two mutually exclusive projects required by this diagram contract')
    if len(se['projects'])>12:raise ValueError('finite classroom enumeration limited to 12 projects')
    ps=project_table(se['projects'],r);caps=[rat(v) for v in se['exclusive_cash_caps']]
    curves=[]
    for raw in se['plot_rates']:
        rate=rat(raw)
        if rate<=-1:raise ValueError('plot rate must exceed -1')
        curves.append(dict(rate=rate,projects=[dict(name=p['name'],NPV=-p['cost']+p['payoff']/(1+rate)) for p in ex]))
    if not curves:raise ValueError('nonempty plot rate grid required')
    rk=inp['risk'];risk=[risk_case(rk,rat(v)) for v in rk['loans']]
    if not risk:raise ValueError('nonempty risk loan grid required')
    return dict(identity=inp['identity'],deterministic=finance,fee_case=fee_case,
                migration_finance=migration_finance,boundary_finance=boundaries,
                operating=base,delayed_operating=delayed,migration_operating=migration_o,
                existing_ai=dict(base=ai_case(ai,rat(ai['utilization'])),
                                 migration=ai_case(ai,rat(ai['migration_utilization']))),
                exclusive_projects=ex,exclusive_choices=exclusive_choices(ex,caps),
                migration_exclusive_projects=exm,migration_exclusive_choices=exclusive_choices(exm,caps),
                budget_projects=ps,budgets=budgets(ps,[rat(v) for v in se['budgets']]),
                NPV_curves=curves,nonconventional=quadratic_irr([rat(v) for v in se['nonconventional_cf']],
                                                            [rat(v) for v in se['root_test_rates']]),
                risk_grid=risk,risk_main=risk_case(rk,rat(rk['main_loan'])),
                risk_migration=risk_case(rk,rat(rk['migration_loan'])),
                checks=dict(cash_counterparties=True,accrual_vs_direct_receipts=True,
                            with_minus_without=True,sunk_cancels=True,
                            finite_subsets='complete enumeration',state_claim_additivity=True))


def serial(x):
    if isinstance(x,F):return str(x)
    if isinstance(x,dict):return {k:serial(v) for k,v in x.items()}
    if isinstance(x,list):return [serial(v) for v in x]
    return x


def label(x):return str(x) if abs(float(x))<10000 else '%.1f'%float(x)


class SVG:
    def __init__(self,title,desc):
        self.s=['<svg xmlns="http://www.w3.org/2000/svg" width="1100" height="520" viewBox="0 0 1100 520">',
                '<title>'+escape(title)+'</title><desc>'+escape(desc)+'</desc>',
                '<rect width="1100" height="520" fill="white"/>',
                '<style>text{font:15px Arial,sans-serif;fill:#183044}.small{font-size:13px}.title{font-size:20px;font-weight:bold}</style>']
        self.text(25,30,title,'title')
    def text(self,x,y,t,cls=''):
        self.s.append('<text x="%g" y="%g" class="%s">%s</text>'%(x,y,cls,escape(str(t))))
    def line(self,x1,y1,x2,y2,color='#cdd6de',width=1):
        self.s.append('<line x1="%g" y1="%g" x2="%g" y2="%g" stroke="%s" stroke-width="%g"/>'%(x1,y1,x2,y2,color,width))
    def rect(self,x,y,w,h,color):
        self.s.append('<rect x="%g" y="%g" width="%g" height="%g" fill="%s"/>'%(x,y,w,h,color))
    def dot(self,x,y,color):self.s.append('<circle cx="%g" cy="%g" r="4" fill="%s"/>'%(x,y,color))
    def finish(self):return '\n'.join(self.s+['</svg>'])+'\n'

def choice_svg(r):
    s=SVG('Ranking and leverage: keep the valuation object fixed',
          'Left: NPV curves of deterministic mutually exclusive projects. Right: required expected returns implied by supplied physical probabilities and positive state prices, as nondefault debt changes. Separate models; no market forecast.')
    cs=sorted(r['NPV_curves'],key=lambda q:q['rate']);rates=[q['rate'] for q in cs]
    xlo,xhi=min(rates),max(rates)
    if xlo==xhi:xhi=xlo+1
    vals=[p['NPV'] for q in cs for p in q['projects']]
    vlo,vhi=min(F(0),*vals),max(F(1),*vals)
    xx=lambda x:75+400*float((x-xlo)/(xhi-xlo));yy=lambda v:380-260*float((v-vlo)/(vhi-vlo))
    s.text(35,65,'NPV (currency), same opportunity rate r')
    projects=r['exclusive_projects'];colors=['#2b7d9a','#d58a32','#8c6cab']
    for j,p in enumerate(projects):
        color=colors[j%len(colors)]
        points=[(q['rate'],next(z['NPV'] for z in q['projects'] if z['name']==p['name'])) for q in cs]
        for a,b in zip(points,points[1:]):s.line(xx(a[0]),yy(a[1]),xx(b[0]),yy(b[1]),color,3)
        for a in points:s.dot(xx(a[0]),yy(a[1]),color)
        if p['IRR'] is not None and xlo<=p['IRR']<=xhi:s.dot(xx(p['IRR']),yy(F(0)),color)
        irr_label=('%g%%'%(100*float(p['IRR']))) if p['IRR'] is not None else 'none'
        s.line(65,460+20*j,92,460+20*j,color,3)
        s.text(102,465+20*j,p['name']+' IRR='+irr_label,'small')
    s.line(65,yy(F(0)),490,yy(F(0)),'#687d8c')
    ticks=sorted(set([vlo,F(0),vhi,vhi/2]))
    for v in ticks:s.text(25,yy(v)+5,'%g'%float(v),'small')
    for i in range(5):
        rate=xlo+(xhi-xlo)*F(i,4);s.text(xx(rate)-8,415,'%g%%'%(100*float(rate)),'small')
    s.text(435,440,'rate r','small')
    grid=sorted(r['risk_grid'],key=lambda q:q['loan']);dl=min(q['loan'] for q in grid);dh=max(q['loan'] for q in grid)
    if dh==dl:dh=dl+1
    values=[q[k] for q in grid for k in ('asset_expected_return','equity_expected_return','bank_rate') if q[k] is not None]
    rl,rh=min(F(0),*values),max(F(1,10),*values)
    xxx=lambda d:655+370*float((d-dl)/(dh-dl));yyy=lambda v:380-260*float((v-rl)/(rh-rl))
    s.text(600,65,'Expected returns on market values (P measure)')
    for k,color,name in [('asset_expected_return','#d58a32','asset rA'),('equity_expected_return','#2b7d9a','equity rE'),('bank_rate','#8c6cab','bank rf')]:
        pts=[(q['loan'],q[k]) for q in grid if q[k] is not None]
        for a,b in zip(pts,pts[1:]):s.line(xxx(a[0]),yyy(a[1]),xxx(b[0]),yyy(b[1]),color,3)
        for a in pts:s.dot(xxx(a[0]),yyy(a[1]),color)
        j=['asset_expected_return','equity_expected_return','bank_rate'].index(k)
        s.line(620,460+20*j,648,460+20*j,color,3)
        s.text(658,465+20*j,name,'small')
    for i in range(5):
        v=rl+(rh-rl)*F(i,4);s.text(590,yyy(v)+5,'%.0f%%'%(100*float(v)),'small')
    for d in sorted(set(q['loan'] for q in grid)):s.text(xxx(d)-8,415,label(d),'small')
    s.text(960,440,'debt D','small')
    return s.finish()


def main():
    p=argparse.ArgumentParser(description=__doc__)
    p.add_argument('--input',type=Path,default=Path(__file__).with_name('inputs.json'))
    p.add_argument('--output-dir',type=Path,required=True)
    args=p.parse_args();data=compute(json.loads(args.input.read_text()))
