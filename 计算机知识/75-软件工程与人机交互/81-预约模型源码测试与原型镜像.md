## 顺序状态模型

单日、整数分钟、单进程内存，无认证、持久化、并发或v2缓冲。运行 `python3 -B reservation.py`；`actor`仅是教学身份标签。

```python
"""Sequential, single-day teaching model. No persistence, auth or networking."""
from dataclasses import dataclass, replace


class Invalid(ValueError):
    pass


class Conflict(ValueError):
    pass


class Forbidden(PermissionError):
    pass


@dataclass(frozen=True)
class Booking:
    id: int
    owner: str
    start: int
    end: int
    status: str = "active"


def validate(actor: str, start: int, end: int) -> None:
    if type(actor) is not str or not actor.strip():
        raise Invalid("actor must be a nonempty string")
    if type(start) is not int or type(end) is not int:
        raise Invalid("minutes must be integers, excluding bool")
    if not 540 <= start < end <= 1020:
        raise Invalid("outside the single-day opening interval")
    if not 30 <= end - start <= 120:
        raise Invalid("duration must be 30 through 120 minutes")


class Schedule:
    def __init__(self) -> None:
        self._bookings: tuple[Booking, ...] = ()
        self._next_id = 1

    def snapshot(self) -> tuple[Booking, ...]:
        return self._bookings

    def reserve(self, actor: str, start: int, end: int) -> Booking:
        validate(actor, start, end)
        if any(b.status == "active" and start < b.end and b.start < end
               for b in self._bookings):
            raise Conflict("requested interval overlaps an active booking")
        booking = Booking(self._next_id, actor, start, end)
        self._bookings += (booking,)
        self._next_id += 1
        return booking

    def cancel(self, actor: str, booking_id: int) -> Booking:
        if type(actor) is not str or not actor.strip():
            raise Invalid("actor must be a nonempty string")
        if type(booking_id) is not int or booking_id <= 0:
            raise Invalid("booking id must be a positive integer, excluding bool")
        for index, booking in enumerate(self._bookings):
            if booking.id != booking_id:
                continue
            if booking.owner != actor:
                raise Forbidden("only the creating actor may cancel")
            if booking.status == "cancelled":
                return booking
            cancelled = replace(booking, status="cancelled")
            self._bookings = (self._bookings[:index] + (cancelled,)
                              + self._bookings[index + 1:])
            return cancelled
        raise KeyError("unknown booking id")


def main() -> None:
    schedule = Schedule()
    first = schedule.reserve("demo", 540, 600)
    print("created", first)
    try:
        schedule.reserve("other", 570, 630)
    except Conflict:
        print("conflict: not created")
    print("adjacent", schedule.reserve("other", 600, 630))
    print("cancelled", schedule.cancel("demo", first.id))
    print("released", schedule.reserve("demo", 540, 600))


if __name__ == "__main__":
    main()
```

## 规格边界与有限oracle测试

```python
import unittest
from dataclasses import FrozenInstanceError
from itertools import product

from reservation import Conflict, Forbidden, Invalid, Schedule


class ReservationTests(unittest.TestCase):
    def test_invalid_inputs_leave_state_and_id_unchanged(self):
        cases = (("a", True, 600), ("a", 540, False), ("a", 540.0, 600),
                 ("a", "540", 600), ("a", 539, 600), ("a", 990, 1021),
                 ("a", 600, 600), ("a", 601, 600), ("a", 540, 569),
                 ("a", 540, 661), ("", 540, 600), ("  ", 540, 600))
        for args in cases:
            with self.subTest(args=args):
                s = Schedule()
                with self.assertRaises(Invalid):
                    s.reserve(*args)
                self.assertEqual(s.snapshot(), ())
                self.assertEqual(s.reserve("ok", 540, 570).id, 1)

    def test_duration_and_opening_boundaries(self):
        for start, end in ((540, 570), (540, 660), (990, 1020)):
            with self.subTest(interval=(start, end)):
                self.assertEqual(Schedule().reserve("a", start, end).end, end)

    def test_adjacent_allowed_overlap_rejected_without_change(self):
        s = Schedule()
        s.reserve("a", 540, 600)
        before = s.snapshot()
        with self.assertRaises(Conflict):
            s.reserve("b", 599, 629)
        self.assertEqual(s.snapshot(), before)
        self.assertEqual(s.reserve("b", 600, 630).id, 2)

    def test_cancel_authority_release_and_idempotence(self):
        s = Schedule()
        b = s.reserve("a", 540, 600)
        before = s.snapshot()
        with self.assertRaises(Forbidden):
            s.cancel("b", b.id)
        self.assertEqual(s.snapshot(), before)
        cancelled = s.cancel("a", b.id)
        once = s.snapshot()
        self.assertEqual(s.cancel("a", b.id), cancelled)
        self.assertEqual(s.snapshot(), once)
        self.assertEqual(s.reserve("b", 540, 600).id, 2)

    def test_invalid_and_unknown_cancel_do_not_change_state(self):
        s = Schedule()
        b = s.reserve("a", 540, 600)
        before = s.snapshot()
        for actor, key in (("a", True), ("a", 0), ("a", "1"), ("", b.id)):
            with self.subTest(key=key, actor=actor):
                with self.assertRaises(Invalid):
                    s.cancel(actor, key)
        with self.assertRaises(KeyError):
            s.cancel("a", 999)
        self.assertEqual(s.snapshot(), before)

    def test_snapshot_and_records_are_immutable(self):
        s = Schedule()
        b = s.reserve("a", 540, 600)
        old = s.snapshot()
        with self.assertRaises(FrozenInstanceError):
            b.status = "cancelled"
        s.cancel("a", b.id)
        self.assertEqual(old[0].status, "active")
        self.assertEqual(s.snapshot()[0].status, "cancelled")

    def test_finite_minute_occupancy_oracle(self):
        # Four valid intervals, 16 ordered pairs. Independent of endpoint formula.
        intervals = ((540, 570), (540, 600), (570, 600), (570, 630))
        for left, right in product(intervals, repeat=2):
            with self.subTest(left=left, right=right):
                shared_minutes = set(range(*left)) & set(range(*right))
                s = Schedule()
                s.reserve("a", *left)
                before = s.snapshot()
                if shared_minutes:
                    with self.assertRaises(Conflict):
                        s.reserve("b", *right)
                    self.assertEqual(s.snapshot(), before)
                else:
                    s.reserve("b", *right)
                    self.assertEqual(len(s.snapshot()), 2)


if __name__ == "__main__":
    unittest.main()
```

"use strict";
const form = document.getElementById("form");
const startInput = document.getElementById("start");
const endInput = document.getElementById("end");
const errors = document.getElementById("errors");
const result = document.getElementById("result");
const list = document.getElementById("bookings");
let nextId = 1;
const bookings = [];
function reportError(message, field) {
  errors.textContent = message;
  field.setAttribute("aria-invalid", "true");
  result.textContent = "未创建：" + message;
  field.focus();
}
function render() {
  list.replaceChildren();
  for (const b of bookings.filter(b => b.active)) {
    const item = document.createElement("li");
    item.append(document.createTextNode(`预约${b.id}：${b.start}–${b.end}分钟 `));
    const cancel = document.createElement("button");
    cancel.type = "button";
    cancel.textContent = `取消预约${b.id}`;
    cancel.addEventListener("click", () => {
      b.active = false;
      render();
      result.textContent = `已取消预约${b.id}，其区间已释放。`;
      startInput.focus();
    });
    item.append(cancel);
    list.append(item);
  }
}
form.addEventListener("submit", event => {
  event.preventDefault();
  errors.textContent = "";
  startInput.removeAttribute("aria-invalid");
  endInput.removeAttribute("aria-invalid");
  const start = startInput.valueAsNumber;
  const end = endInput.valueAsNumber;
  if (!Number.isInteger(start) || start < 540 || start > 990) {
    reportError("开始必须为540–990的整数分钟。", startInput); return;
  }
  if (!Number.isInteger(end) || end > 1020 || end <= start) {
    reportError("结束必须为大于开始且不超过1020的整数分钟。", endInput); return;
  }
  if (end - start < 30 || end - start > 120) {
    reportError("时长必须为30–120分钟，请修改结束。", endInput); return;
  }
  if (bookings.some(b => b.active && start < b.end && b.start < end)) {
    reportError("该时间与活动预约重叠，请修改开始或结束。", startInput); return;
  }
  const booking = {id: nextId++, start, end, active: true};
  bookings.push(booking);
  render();
  result.textContent = `已创建预约${booking.id}：${start}–${end}分钟。`;
});
</script>
