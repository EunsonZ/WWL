package org.wwl.networkdemo

import android.app.job.JobParameters
import android.app.job.JobService
import android.os.Build
import android.os.Handler
import android.os.Looper
import org.json.JSONObject
import org.wwl.intentdemo.Pending
import org.wwl.intentdemo.Protocol
import java.util.concurrent.atomic.AtomicBoolean

/** Persisted JobScheduler metadata points to a durable intent; it never invents one. */
class IntentJobService:JobService() {
    private class Run(val params:JobParameters) {
        val stopped=AtomicBoolean(false)
        @Volatile var transport:HttpsTransport?=null
    }
    private var active:Run?=null
    private val app:ResourceApp get()=application as ResourceApp
    private val main=Handler(Looper.getMainLooper())
    override fun onStartJob(params:JobParameters):Boolean {
        val run=Run(params);active=run
        app.repository.worker.execute {
            var retry=false
            try {
                if(run.stopped.get())return@execute
                val e=params.extras;val actor=e.getString("actor")?:error("missing actor")
                val id=e.getString("intent_id")?:error("missing id");val fp=e.getString("fingerprint")?:error("missing fp")
                val p=app.repository.client.get(actor,id)?:error("no durable intent")
                require(actor==BuildConfig.ACTOR&&p.fp==fp)
                if(p.result!=null) { app.resourceStore.audit("job_already_final",id) }
                else if(app.resourceStore.attempts(p)>=ResourceStore.MAX_ATTEMPTS) {
                    app.resourceStore.audit("budget_exhausted",id+": unknown is not a business rejection")
                } else if(!run.stopped.get()) {
                    val attempt=app.resourceStore.takeAttempt(p)
                    val transport=HttpsTransport(actor,if(Build.VERSION.SDK_INT>=28)params.network else null)
                    run.transport=transport
                    if(run.stopped.get())transport.cancel()
                    app.repository.client.phase(p,"sending")
                    app.resourceStore.audit("job_started",id+":attempt="+attempt)
                    try {
                        val q=transport.query(p,"self")
                        require(q.get("protocol")==Protocol.NAME&&q.get("actor")==actor&&q.get("intent_id")==id)
                        val reply=if(q.optString("status")=="unknown") {
                            require(q.keys().asSequence().toSet()==setOf("protocol","actor","intent_id","status","reason")&&q.get("reason")=="not_registered")
                            if(run.stopped.get())throw java.io.InterruptedIOException("job stopped before replay")
                            transport.reserve(p,"self")
                        } else q
                        app.repository.client.accept(p,reply)
                        app.resourceStore.outcome(p,"durable_"+reply.getString("status"))
                    } catch(ex:Exception) {
                        app.repository.client.phase(p,if(ex is HttpFailure&&ex.code==409)"protocol_error" else "unknown")
                        app.resourceStore.outcome(p,if(ex is HttpFailure)"HTTP "+ex.code else ex.javaClass.simpleName)
                        // Do not spin on credentials/protocol errors. I/O unknown may retry
                        // same id, subject to both system constraints and durable budget.
                        retry=ex is java.io.IOException&&ex !is HttpFailure&&attempt<ResourceStore.MAX_ATTEMPTS
                    }
                }
                app.exportResources()
            } catch(ex:Exception) {
                // Migration/storage failure: keep old files; no destructive recovery/send.
                java.io.File(filesDir,"job-error.txt").writeText(ex.javaClass.simpleName+": "+ex.message)
            } finally {
                main.post {
                    if(active===run&&!run.stopped.get()) { active=null;jobFinished(params,retry) }
                }
            }
        }
        return true // Work is asynchronous; onStartJob itself does no SQLite/network I/O.
    }
    override fun onStopJob(params:JobParameters):Boolean {
        val run=active
        if(run!=null&&run.params.jobId==params.jobId) {
            run.stopped.set(true);active=null
            val reason=if(Build.VERSION.SDK_INT>=31)params.stopReason.toString() else "API<31"
            app.repository.worker.execute {
                runCatching { app.resourceStore.audit("job_stopped","job="+params.jobId+":reason="+reason);app.exportResources() }
            }
            // Never block the main callback while a socket is being disconnected.
            Thread({ run.transport?.cancel() },"wwl175-cancel").apply { isDaemon=true;start() }
        }
        // A later bind rechecks persisted id/result/budget before any new send.
        // Cancellation cannot retract a transaction that already committed.
        return true
    }
}

    fun queueDraft(id:String=UUID.randomUUID().toString()) {
        if(rejectWhileSaving())return

    }

/** Resource accounting is separate from client.db's immutable business intentions. */
class ResourceStore(context:Context,name:String="resources.db",private val requestedVersion:Int=BuildConfig.COURSE_DB_VERSION,
    private val failMigration:Boolean=BuildConfig.COURSE_FAIL_MIGRATION):SQLiteOpenHelper(context,name,null,requestedVersion) {
    init { require(requestedVersion in 1..2) }
    companion object { const val MAX_ATTEMPTS=3 }
    override fun onCreate(db:SQLiteDatabase) {
        require(requestedVersion in 1..2)
        db.execSQL("CREATE TABLE work(actor TEXT NOT NULL,intent_id TEXT NOT NULL,fp TEXT NOT NULL,attempts INTEGER NOT NULL CHECK(attempts BETWEEN 0 AND 3),PRIMARY KEY(actor,intent_id))")
        db.execSQL("CREATE TABLE events(seq INTEGER PRIMARY KEY AUTOINCREMENT,kind TEXT NOT NULL,detail TEXT NOT NULL)")
        if(requestedVersion==2)db.execSQL("ALTER TABLE work ADD COLUMN last_http TEXT NOT NULL DEFAULT 'none'")
    }
    override fun onUpgrade(db:SQLiteDatabase,old:Int,new:Int) {
        require(old==1&&new==2) { "explicit resource migration required; keep database" }
        db.execSQL("ALTER TABLE work ADD COLUMN last_http TEXT NOT NULL DEFAULT 'none'")
        if(failMigration)throw android.database.sqlite.SQLiteException("PUBLIC LAB: fail after ALTER; transaction must roll back")
        db.execSQL("INSERT INTO events(kind,detail) VALUES('migration','1->2 retained work/id/fp/attempts')")
    }
    override fun onDowngrade(db:SQLiteDatabase,old:Int,new:Int) { error("resource downgrade rejected; no destructive reset") }
    fun audit(kind:String,detail:String) { writableDatabase.execSQL("INSERT INTO events(kind,detail) VALUES(?,?)",arrayOf(kind,detail)) }
    fun attempts(p:Pending):Int = readableDatabase.rawQuery("SELECT fp,attempts FROM work WHERE actor=? AND intent_id=?",arrayOf(p.actor,p.id)).use {
        if(!it.moveToFirst())0 else { require(it.getString(0)==p.fp);it.getInt(1) }
    }
    fun takeAttempt(p:Pending):Int {
        val db=writableDatabase;db.beginTransaction()
        try {
            db.execSQL("INSERT OR IGNORE INTO work(actor,intent_id,fp,attempts) VALUES(?,?,?,0)",arrayOf(p.actor,p.id,p.fp))
            val old=attempts(p);require(old<MAX_ATTEMPTS) { "logical attempt budget exhausted; business remains unknown" }
            db.execSQL("UPDATE work SET attempts=? WHERE actor=? AND intent_id=?",arrayOf<Any>(old+1,p.actor,p.id))
            audit("budget_taken",p.id+":"+(old+1));db.setTransactionSuccessful();return old+1
        } finally { db.endTransaction() }
    }
    fun outcome(p:Pending,detail:String) {
        if(readableDatabase.version>=2)writableDatabase.execSQL("UPDATE work SET last_http=? WHERE actor=? AND intent_id=?",arrayOf(detail,p.actor,p.id))
        audit("job_outcome",p.id+":"+detail)
    }
    fun snapshot():JSONObject {
        val db=readableDatabase;val rows=JSONArray();val log=JSONArray()
        db.rawQuery("SELECT * FROM work ORDER BY actor,intent_id",null).use { c -> while(c.moveToNext()) {
            val v=JSONObject();for(i in 0 until c.columnCount)v.put(c.getColumnName(i),if(c.getColumnName(i)=="attempts")c.getInt(i) else c.getString(i));rows.put(v)
        } }
        db.rawQuery("SELECT seq,kind,detail FROM events ORDER BY seq",null).use { c -> while(c.moveToNext())log.put(JSONObject().put("seq",c.getInt(0)).put("kind",c.getString(1)).put("detail",c.getString(2))) }
        return JSONObject().put("schema_version",db.version).put("work",rows).put("events",log)
    }
}
