```asm
    .section .text.start, "ax"
    .align  4
    .globl  _start
_start:
    csrr    t0, mhartid
    bnez    t0, .Lpark

    la      t0, __bss_start
    la      t1, __bss_end
.Lzero_bss:
    bgeu    t0, t1, .Lbss_done
    sd      zero, 0(t0)
    addi    t0, t0, 8
    j       .Lzero_bss
.Lbss_done:
    la      sp, m_stack_top
    la      t0, machine_trap_entry
    csrw    mtvec, t0
    call    machine_banner

    # A single broad PMP TOR region allows the teaching S/U payload to access
    # RAM and virt MMIO. Sv39 remains the U/S permission boundary under test.
    li      t0, -1
    csrw    pmpaddr0, t0
    li      t0, 0x0f
    csrw    pmpcfg0, t0

    # U ecall plus instruction/load/store page faults are handled in S mode.
    li      t0, ((1 << 8) | (1 << 12) | (1 << 13) | (1 << 15))
    csrw    medeleg, t0
    csrw    mideleg, zero

    la      t0, supervisor_entry
    csrw    mepc, t0
    csrr    t0, mstatus
    li      t1, (3 << 11)
    not     t1, t1
    and     t0, t0, t1
    li      t1, (1 << 11)
    or      t0, t0, t1
    csrw    mstatus, t0
    mret

supervisor_entry:
    la      sp, s_stack_top
    call    supervisor_main
    li      a0, 0x7f
    call    qemu_fail

    .globl  machine_trap_entry
machine_trap_entry:
    la      sp, m_stack_top
    call    machine_trap

.Lpark:
    wfi
    j       .Lpark
```

## `user.S`

<!-- code-sync: ./user.S -->
```asm
    .section .user.text, "ax"
    .align  4
    .globl  user_start
user_start:
    la      sp, user_stack_top
    li      s0, 0x1122334455667788
    li      s1, 0x8877665544332211

    # A real call/return proves that the U=1, X=1 alias is fetching code.
    call    user_exec_probe
    li      t0, 0x123
    bne     a2, t0, user_fail

    # Both addresses below are PC-relative. Because this code executes through
    # a fixed-offset alias, the linked physical symbols become low user VAs.
    la      t1, user_data_backing
    ld      t2, 0(t1)
    li      t3, 0x1111222233334444
    bne     t2, t3, user_fail
    li      t2, 0x5555666677778888
    sd      t2, 0(t1)
    ld      a0, 0(t1)
    li      a7, 1
    ecall
    bnez    a0, user_fail

    # S has now cleared W and fenced this VA. The store itself must trap.
    li      t2, 0xa5a5a5a5a5a5a5a5
    .globl  user_store_fault_insn
user_store_fault_insn:
    sd      t2, 0(t1)
    j       user_fail
    .globl  user_after_store_fault
user_after_store_fault:
    li      t0, -15
    bne     a0, t0, user_fail
    ld      t2, 0(t1)
    li      t3, 0x5555666677778888
    bne     t2, t3, user_fail
    li      a7, 2
    ecall
    bnez    a0, user_fail

    # This address is identity mapped only as an S page (PTE.U=0).
    li      t1, 0x80013000
    .globl  user_load_fault_insn
user_load_fault_insn:
    ld      a0, 0(t1)
    j       user_fail
    .globl  user_after_load_fault
user_after_load_fault:
    li      t0, -13
    bne     a0, t0, user_fail
    li      a7, 3
    ecall
    bnez    a0, user_fail

    # PTE.R=1 but PTE.X=0. The JALR executes; fetching its target must fault.
    la      t1, user_nx_target
    jalr    ra, t1, 0
    j       user_fail
    .globl  user_after_exec_fault
user_after_exec_fault:
    li      t0, -12
    bne     a0, t0, user_fail
    li      a7, 4
    ecall
    j       user_fail

user_exec_probe:
    li      a2, 0x123
    ret

user_fail:
    li      a7, 7
    ecall
    j       user_fail

    .section .user.data, "aw"
    .align  3
    .globl  user_data_backing
user_data_backing:
    .dword  0x1111222233334444

    .section .user.nx, "ax"
    .align  4
    .globl  user_nx_target
user_nx_target:
    li      a0, 99
    ret

    .section .kernel.secret, "aw"
    .align  3
    .globl  kernel_secret
kernel_secret:
    .dword  0xcafebabedeadbeef
```

```c
#include "trapframe.h"

typedef unsigned int u32;
typedef unsigned char u8;

extern char supervisor_trap_entry[];
extern char user_start[];
extern char user_store_fault_insn[];
extern char user_after_store_fault[];
extern char user_load_fault_insn[];
extern char user_after_load_fault[];
extern char user_after_exec_fault[];
extern char user_nx_target[];
extern char user_data_backing[];
extern char kernel_secret[];
extern char root_page_table[];
extern char user_l1_page_table[];
extern char user_l0_page_table[];
extern char kernel_l1_page_table[];
extern char kernel_l0_page_table[];
extern char trapframe_area[];
extern char kernel_end[];

enum {
    PTE_V = 1UL << 0,
    PTE_R = 1UL << 1,
    PTE_W = 1UL << 2,
    PTE_X = 1UL << 3,
    PTE_U = 1UL << 4,
    PTE_G = 1UL << 5,
    PTE_A = 1UL << 6,
    PTE_D = 1UL << 7,
};

enum stage {
    STAGE_ALIAS = 0,
    STAGE_EXPECT_STORE,
    STAGE_STORE_RETURNED,
    STAGE_EXPECT_LOAD,
    STAGE_LOAD_RETURNED,
    STAGE_EXPECT_EXEC,
    STAGE_EXEC_RETURNED,
};

#define UART0 0x10000000UL
#define TEST_FINISHER 0x00100000UL
#define USER_VA_BASE 0x40000000UL
#define USER_PA_BASE 0x80010000UL
#define USER_ALIAS_DELTA (USER_PA_BASE - USER_VA_BASE)
#define USER_DATA_VA 0x40001000UL
#define USER_NX_VA 0x40002000UL
#define USER_STACK_VA 0x40005000UL
#define KERNEL_SECRET_VA 0x80013000UL
#define DATA_INITIAL 0x1111222233334444UL
#define DATA_WRITTEN 0x5555666677778888UL
#define SECRET_VALUE 0xcafebabedeadbeefUL
#define CANARY_S0 0x1122334455667788UL
#define CANARY_S1 0x8877665544332211UL

static volatile u8 *const uart = (volatile u8 *)UART0;
static volatile u32 *const finisher = (volatile u32 *)TEST_FINISHER;
static enum stage current_stage = STAGE_ALIAS;

_Static_assert(sizeof(struct trapframe) == 272, "trap frame assembly/C mismatch");

static inline u64 csr_read_satp(void) {
    u64 value;
    __asm__ volatile("csrr %0, satp" : "=r"(value));
    return value;
}

static inline u64 csr_read_sstatus(void) {
    u64 value;
    __asm__ volatile("csrr %0, sstatus" : "=r"(value));
    return value;
}

static inline u64 csr_read_mcause(void) {
    u64 value;
    __asm__ volatile("csrr %0, mcause" : "=r"(value));
    return value;
}

static inline u64 csr_read_mepc(void) {
    u64 value;
    __asm__ volatile("csrr %0, mepc" : "=r"(value));
    return value;
}

static inline u64 csr_read_mtval(void) {
    u64 value;
    __asm__ volatile("csrr %0, mtval" : "=r"(value));
    return value;
}

static void putc(char character) {
    *uart = (u8)character;
}

static void puts(const char *text) {
    while (*text != '\0') {
        putc(*text++);
    }
}

static void put_hex(u64 value) {
    static const char digits[] = "0123456789abcdef";
    puts("0x");
    for (int shift = 60; shift >= 0; shift -= 4) {
        putc(digits[(value >> shift) & 0xfUL]);
    }
}

static void put_uint(u64 value) {
    char buffer[24];
    int index = 0;
    if (value == 0) {
        putc('0');
        return;
    }
    while (value != 0) {
        buffer[index++] = (char)('0' + value % 10);
        value /= 10;
    }
    while (index != 0) {
        putc(buffer[--index]);
    }
}

__attribute__((noreturn)) void qemu_fail(u64 code) {
    puts("RESULT FAIL code=");
    put_hex(code);
    puts("\n");
    *finisher = (u32)(((code & 0xffffUL) << 16) | 0x3333UL);
    for (;;) {
        __asm__ volatile("wfi");
    }
}

__attribute__((noreturn)) static void qemu_pass(void) {
    puts("RESULT PASS\n");
    *finisher = 0x5555U;
    for (;;) {
        __asm__ volatile("wfi");
    }
}

void machine_banner(void) {
    puts("MODE M: boot hart=0 delegate=0x000000000000b100\n");
}

__attribute__((noreturn)) void machine_trap(void) {
    puts("MACHINE TRAP mcause=");
    put_hex(csr_read_mcause());
    puts(" mepc=");
    put_hex(csr_read_mepc());
    puts(" mtval=");
    put_hex(csr_read_mtval());
    puts("\n");
    qemu_fail(0xe0);
}

static u64 pa_to_pte(u64 pa) {
    return (pa >> 12) << 10;
}

static u64 pte_to_pa(u64 pte) {
    return (pte >> 10) << 12;
}

static u64 pointer_pte(const volatile void *page) {
    return pa_to_pte((u64)page) | PTE_V;
}

static u64 leaf_pte(u64 pa, u64 flags) {
    if ((flags & PTE_W) != 0 && (flags & PTE_R) == 0) {
        qemu_fail(0xe1);
    }
    return pa_to_pte(pa) | flags | PTE_V;
}

static void zero_page(volatile u64 *page) {
    for (u64 index = 0; index < 512; ++index) {
        page[index] = 0;
    }
}

static u64 user_va(const void *physical_symbol) {
    return (u64)physical_symbol - USER_ALIAS_DELTA;
}

static void map_kernel_identity(volatile u64 *l0) {
    for (u64 pa = 0x80000000UL; pa < (u64)kernel_end; pa += 4096) {
        u64 index = (pa >> 12) & 0x1ffUL;
        l0[index] = leaf_pte(pa, PTE_R | PTE_W | PTE_X | PTE_A | PTE_D);
    }
}

static void build_page_tables(void) {
    volatile u64 *root = (volatile u64 *)root_page_table;
    volatile u64 *user_l1 = (volatile u64 *)user_l1_page_table;
    volatile u64 *user_l0 = (volatile u64 *)user_l0_page_table;
    volatile u64 *kernel_l1 = (volatile u64 *)kernel_l1_page_table;
    volatile u64 *kernel_l0 = (volatile u64 *)kernel_l0_page_table;

    zero_page(root);
    zero_page(user_l1);
    zero_page(user_l0);
    zero_page(kernel_l1);
    zero_page(kernel_l0);

    #define VPN2(va) (((u64)(va) >> 30) & 0x1ffUL)
    #define VPN1(va) (((u64)(va) >> 21) & 0x1ffUL)
    #define VPN0(va) (((u64)(va) >> 12) & 0x1ffUL)

    /* Root entry 0 is a naturally aligned 1 GiB U=0 MMIO leaf. */
    root[0] = leaf_pte(0, PTE_R | PTE_W | PTE_A | PTE_D);

    root[VPN2(USER_VA_BASE)] = pointer_pte(user_l1);
    user_l1[VPN1(USER_VA_BASE)] = pointer_pte(user_l0);
    user_l0[VPN0(USER_VA_BASE)] =
        leaf_pte(USER_PA_BASE, PTE_R | PTE_X | PTE_U | PTE_A);
    user_l0[VPN0(USER_DATA_VA)] =
        leaf_pte((u64)user_data_backing,
                 PTE_R | PTE_W | PTE_U | PTE_A | PTE_D);
    user_l0[VPN0(USER_NX_VA)] =
        leaf_pte((u64)user_nx_target, PTE_R | PTE_U | PTE_A);
    user_l0[VPN0(USER_STACK_VA)] =
        leaf_pte(0x80015000UL, PTE_R | PTE_W | PTE_U | PTE_A | PTE_D);

    root[VPN2(0x80000000UL)] = pointer_pte(kernel_l1);
    kernel_l1[VPN1(0x80000000UL)] = pointer_pte(kernel_l0);
    map_kernel_identity(kernel_l0);

    #undef VPN2
    #undef VPN1
    #undef VPN0
}

struct walk_result {
    u64 l2;
    u64 l1;
    u64 l0;
    u64 pa;
};

static struct walk_result walk(u64 va) {
    volatile u64 *level = (volatile u64 *)root_page_table;
    struct walk_result result;
    result.l2 = level[(va >> 30) & 0x1ffUL];
    level = (volatile u64 *)pte_to_pa(result.l2);
    result.l1 = level[(va >> 21) & 0x1ffUL];
    level = (volatile u64 *)pte_to_pa(result.l1);
    result.l0 = level[(va >> 12) & 0x1ffUL];
    result.pa = pte_to_pa(result.l0) | (va & 0xfffUL);
    return result;
}

static void print_walk(void) {
    struct walk_result value = walk(USER_DATA_VA);
    puts("WALK va=");
    put_hex(USER_DATA_VA);
    puts(" vpn2=1 vpn1=0 vpn0=1 offset=0 l2=");
    put_hex(value.l2);
    puts(" l1=");
    put_hex(value.l1);
    puts(" l0=");
    put_hex(value.l0);
    puts(" pa=");
    put_hex(value.pa);
    puts(" flags=VRWUAD\n");
}

static void require_canaries(const struct trapframe *tf, u64 code) {
    if (tf->s0 != CANARY_S0 || tf->s1 != CANARY_S1) {
        qemu_fail(code);
    }
}

static void print_fault(const char *kind, const struct trapframe *tf,
                        const char *action) {
    puts("FAULT kind=");
    puts(kind);
    puts(" scause=");
    put_uint(tf->scause);
    puts(" sepc=");
    put_hex(tf->sepc);
    puts(" stval=");
    put_hex(tf->stval);
    puts(" action=");
    puts(action);
    puts("\n");
}

static struct trapframe *handle_ecall(struct trapframe *tf) {
    if (tf->scause != 8 || (tf->sepc & 3UL) != 0) {
        qemu_fail(0x20);
    }
    require_canaries(tf, 0x21);
    puts("TRAP kind=ecall stage=");
    put_uint((u64)current_stage);
    puts(" scause=8 sepc=");
    put_hex(tf->sepc);
    puts("\n");

    if (tf->a7 == 1 && current_stage == STAGE_ALIAS) {
        volatile u64 *data = (volatile u64 *)user_data_backing;
        if (tf->a2 != 0x123 || *data != DATA_WRITTEN ||
            tf->sepc < USER_VA_BASE || tf->sepc >= USER_DATA_VA) {
            qemu_fail(0x22);
        }
        puts("ALIAS execute_va=");
        put_hex(USER_VA_BASE);
        puts(" data_va=");
        put_hex(USER_DATA_VA);
        puts(" data_pa=");
        put_hex((u64)user_data_backing);
        puts(" backing=");
        put_hex(*data);
        puts(" status=PASS\n");

        volatile u64 *user_l0 = (volatile u64 *)user_l0_page_table;
        u64 index = (USER_DATA_VA >> 12) & 0x1ffUL;
        u64 before = user_l0[index];
        user_l0[index] = before & ~PTE_W;
        __asm__ volatile("sfence.vma %0, zero" : : "r"(USER_DATA_VA) : "memory");
        puts("PTE UPDATE va=");
        put_hex(USER_DATA_VA);
        puts(" before=");
        put_hex(before);
        puts(" after=");
        put_hex(user_l0[index]);
        puts(" sfence.vma=address-asid-all\n");
        current_stage = STAGE_EXPECT_STORE;
        tf->a0 = 0;
        tf->sepc += 4;
        return tf;
    }

    if (tf->a7 == 2 && current_stage == STAGE_STORE_RETURNED) {
        if (*(volatile u64 *)user_data_backing != DATA_WRITTEN) {
            qemu_fail(0x23);
        }
        puts("CHECK store-denied=PASS backing=");
        put_hex(*(volatile u64 *)user_data_backing);
        puts("\n");
        current_stage = STAGE_EXPECT_LOAD;
        tf->a0 = 0;
        tf->sepc += 4;
        return tf;
    }

    if (tf->a7 == 3 && current_stage == STAGE_LOAD_RETURNED) {
        if (*(volatile u64 *)kernel_secret != SECRET_VALUE) {
            qemu_fail(0x24);
        }
        puts("CHECK load-u0-denied=PASS secret=");
        put_hex(*(volatile u64 *)kernel_secret);
        puts("\n");
        current_stage = STAGE_EXPECT_EXEC;
        tf->a0 = 0;
        tf->sepc += 4;
        return tf;
    }

    if (tf->a7 == 4 && current_stage == STAGE_EXEC_RETURNED) {
        puts("CHECK execute-nx-denied=PASS\n");
        qemu_pass();
    }

    qemu_fail(tf->a7 == 7 ? 0x27 : 0x26);
}

struct trapframe *supervisor_trap(struct trapframe *tf) {
    if (tf->scause == 8) {
        return handle_ecall(tf);
    }

    require_canaries(tf, 0x30);
    if (current_stage == STAGE_EXPECT_STORE && tf->scause == 15 &&
        tf->stval == USER_DATA_VA &&
        tf->sepc == user_va(user_store_fault_insn) &&
        *(volatile u64 *)user_data_backing == DATA_WRITTEN) {
        print_fault("store", tf, "redirect-explicit-label");
        tf->a0 = (u64)(i64)-15;
        tf->sepc = user_va(user_after_store_fault);
        current_stage = STAGE_STORE_RETURNED;
        return tf;
    }

    if (current_stage == STAGE_EXPECT_LOAD && tf->scause == 13 &&
        tf->stval == KERNEL_SECRET_VA &&
        tf->sepc == user_va(user_load_fault_insn) &&
        *(volatile u64 *)kernel_secret == SECRET_VALUE) {
        print_fault("load", tf, "redirect-explicit-label");
        tf->a0 = (u64)(i64)-13;
        tf->sepc = user_va(user_after_load_fault);
        current_stage = STAGE_LOAD_RETURNED;
        return tf;
    }

    if (current_stage == STAGE_EXPECT_EXEC && tf->scause == 12 &&
        tf->stval == USER_NX_VA && tf->sepc == USER_NX_VA) {
        print_fault("instruction", tf, "redirect-explicit-label");
        tf->a0 = (u64)(i64)-12;
        tf->sepc = user_va(user_after_exec_fault);
        current_stage = STAGE_EXEC_RETURNED;
        return tf;
    }

    puts("UNEXPECTED stage=");
    put_uint((u64)current_stage);
    puts(" scause=");
    put_uint(tf->scause);
    puts(" sepc=");
    put_hex(tf->sepc);
    puts(" stval=");
    put_hex(tf->stval);
    puts("\n");
    qemu_fail(0x3f);
}

__attribute__((noreturn)) void supervisor_main(void) {
    puts("MODE S: build Sv39 tables\n");
    build_page_tables();

    puts("PAGE_TABLE root_pa=");
    put_hex((u64)root_page_table);
    puts(" entries=512 bytes=4096 levels=3\n");
    print_walk();

    u64 root_ppn = (u64)root_page_table >> 12;
    u64 requested_satp = (8UL << 60) | root_ppn;
    __asm__ volatile("fence rw, rw" : : : "memory");
    __asm__ volatile("csrw satp, %0" : : "r"(requested_satp) : "memory");
    __asm__ volatile("sfence.vma zero, zero" : : : "memory");
    u64 readback = csr_read_satp();
    if ((readback >> 60) != 8 || (readback & ((1UL << 44) - 1)) != root_ppn) {
        qemu_fail(0x40);
    }
    puts("SATP readback=");
    put_hex(readback);
    puts(" mode=8 root_ppn=");
    put_hex(root_ppn);
    puts(" sfence.vma=global\n");

    u64 status = csr_read_sstatus();
    status &= ~((1UL << 8) | (1UL << 18) | (1UL << 19) | (1UL << 1));
    status |= 1UL << 5;
    __asm__ volatile("csrw sstatus, %0" : : "r"(status));
    __asm__ volatile("csrw stvec, %0" : : "r"((u64)supervisor_trap_entry));
    __asm__ volatile("csrw sscratch, %0" : : "r"((u64)trapframe_area));
    __asm__ volatile("csrw sepc, %0" : : "r"(USER_VA_BASE));
    puts("POLICY SUM=0 MXR=0 A=pre-set D=pre-set\n");
    puts("MODE U: entry_va=");
    put_hex(USER_VA_BASE);
    puts(" entry_pa=");
    put_hex(USER_PA_BASE);
    puts(" stack_va=0x0000000040005000\n");
    __asm__ volatile("sret");
    __builtin_unreachable();
}
```

```python
#!/usr/bin/env python3
"""Build the RV64 Sv39 lab, run QEMU virt, and verify its complete trace."""

from __future__ import annotations

import argparse
import contextlib
import os
from pathlib import Path
import re
import shlex
import shutil
import signal
import struct
import subprocess
import tempfile


class LabError(RuntimeError):
    pass


USER_ALIAS_DELTA = 0x40010000


def resolve_tool(value: str | None, fallback: str) -> Path:
    candidate = value or shutil.which(fallback)
    if not candidate:
        raise LabError(f"tool not found: pass --{fallback.split('-')[0]} explicitly")
    path = Path(candidate).expanduser().resolve()
    if not path.is_file() or not os.access(path, os.X_OK):
        raise LabError(f"tool is not executable: {path}")
    return path


def run_checked(command: list[str], *, cwd: Path) -> str:
    print("+", shlex.join(command))
    result = subprocess.run(
        command,
        cwd=cwd,
        text=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        check=False,
    )
    if result.returncode != 0:
        raise LabError(f"command exited {result.returncode}:\n{result.stdout}")
    return result.stdout


def first_version_line(tool: Path) -> str:
    result = subprocess.run(
        [str(tool), "--version"],
        text=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        check=False,
    )
    if result.returncode != 0 or not result.stdout.strip():
        raise LabError(f"cannot read version from {tool}")
    return result.stdout.splitlines()[0]


def check_riscv_elf(path: Path) -> None:
    header = path.read_bytes()[:64]
    if len(header) < 64 or header[:4] != b"\x7fELF":
        raise LabError("linker output is not an ELF file")
    if header[4] != 2 or header[5] != 1:
        raise LabError("ELF must be 64-bit little-endian")
    machine = struct.unpack_from("<H", header, 18)[0]
    if machine != 243:
        raise LabError(f"ELF e_machine is {machine}, expected RISC-V (243)")


def elf_symbols(path: Path) -> dict[str, int]:
    data = path.read_bytes()
    section_offset = struct.unpack_from("<Q", data, 40)[0]
    section_size = struct.unpack_from("<H", data, 58)[0]
    section_count = struct.unpack_from("<H", data, 60)[0]
    sections = []
    for index in range(section_count):
        offset = section_offset + index * section_size
        sections.append(struct.unpack_from("<IIQQQQIIQQ", data, offset))
    symbols: dict[str, int] = {}
    for section in sections:
        if section[1] != 2:  # SHT_SYMTAB
            continue
        sym_offset, sym_size, link, entry_size = (
            section[4],
            section[5],
            section[6],
            section[9],
        )
        string_section = sections[link]
        strings = data[string_section[4] : string_section[4] + string_section[5]]
        for offset in range(sym_offset, sym_offset + sym_size, entry_size):
            name_offset, _, _, _, value, _ = struct.unpack_from(
                "<IBBHQQ", data, offset
            )
            end = strings.find(b"\0", name_offset)
            name = strings[name_offset:end].decode("utf-8", errors="replace")
            if name:
                symbols[name] = value
    return symbols


def build(source_dir: Path, out_dir: Path, clang: Path, linker: Path) -> Path:
    common = [
        str(clang),
        "--target=riscv64-unknown-elf",
        "-march=rv64im_zicsr",
        "-mabi=lp64",
        "-mcmodel=medany",
        "-mno-relax",
        "-msmall-data-limit=0",
        "-ffreestanding",
        "-fno-builtin",
        "-fno-pic",
        "-fno-stack-protector",
        "-ffunction-sections",
        "-fdata-sections",
        "-O2",
        "-Wall",
        "-Wextra",
        "-Werror",
        f"-I{source_dir}",
    ]
    objects: list[Path] = []
    for source_name in ("boot.S", "trap.S", "user.S", "kernel.c"):
        source = source_dir / source_name
        obj = out_dir / f"{source.stem}.o"
        run_checked(common + ["-c", str(source), "-o", str(obj)], cwd=source_dir)
        objects.append(obj)

    elf = out_dir / "kernel.elf"
    map_file = out_dir / "kernel.map"
    run_checked(
        [
            str(linker),
            "-m",
            "elf64lriscv",
            "--gc-sections",
            "--no-relax",
            "-z",
            "max-page-size=4096",
            "-T",
            str(source_dir / "linker.ld"),
            "-Map",
            str(map_file),
            "-o",
            str(elf),
            *(str(obj) for obj in objects),
        ],
        cwd=source_dir,
    )
    check_riscv_elf(elf)
    required = {
        "root_page_table": 0x80016000,
        "user_l1_page_table": 0x80017000,
        "user_l0_page_table": 0x80018000,
        "kernel_l1_page_table": 0x80019000,
        "kernel_l0_page_table": 0x8001A000,
    }
    symbols = elf_symbols(elf)
    for name, expected in required.items():
        if symbols.get(name) != expected:
            raise LabError(
                f"ELF symbol {name}={symbols.get(name)!r}, expected 0x{expected:x}"
            )
    print(f"ELF CHECK PASS: {elf} is ELF64 little-endian RISC-V")
    return elf


def run_qemu(qemu: Path, elf: Path, timeout_seconds: float) -> tuple[int, str]:
    command = [
        str(qemu),
        "-machine",
        "virt",
        "-accel",
        "tcg",
        "-smp",
        "1",
        "-m",
        "128M",
        "-bios",
        "none",
        "-kernel",
        str(elf),
        "-nographic",
        "-serial",
        "stdio",
        "-monitor",
        "none",
        "-net",
        "none",
        "-no-reboot",
    ]
    print("+", shlex.join(command))
    process = subprocess.Popen(
        command,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        start_new_session=True,
    )
    try:
        raw_output, _ = process.communicate(timeout=timeout_seconds)
    except subprocess.TimeoutExpired as error:
        os.killpg(process.pid, signal.SIGKILL)
        raw_output, _ = process.communicate()
        partial = raw_output.decode("utf-8", errors="replace")
        raise LabError(
            f"QEMU exceeded {timeout_seconds:.1f}s; process group killed:\n{partial}"
        ) from error
    finally:
        if process.poll() is None:
            os.killpg(process.pid, signal.SIGKILL)
            process.wait()
    return process.returncode, raw_output.decode("utf-8", errors="replace")


def _hx(value: int) -> str:
    return f"0x{value:016x}"

def verify_output(returncode: int, output: str, symbols: dict[str, int]) -> None:
    if returncode != 0:
        raise LabError(f"QEMU exited with {returncode}, expected 0\n{output}")
    forbidden = ("RESULT FAIL", "UNEXPECTED", "MACHINE TRAP")
    for marker in forbidden:
        if marker in output:
            raise LabError(f"guest emitted forbidden marker {marker!r}\n{output}")
    lines = output.splitlines()
    patterns = expected_patterns(symbols)
    if len(lines) != len(patterns):
        raise LabError(
            f"guest emitted {len(lines)} lines, expected {len(patterns)}\n{output}"
        )
    for index, (line, pattern) in enumerate(zip(lines, patterns), 1):
        if re.fullmatch(pattern, line) is None:
            raise LabError(
                f"trace line {index} mismatch\nexpected /{pattern}/\nactual {line!r}"
            )
    if output.count("RESULT PASS") != 1:
        raise LabError("guest must emit exactly one RESULT PASS")


def parser() -> argparse.ArgumentParser:
    result = argparse.ArgumentParser(description=__doc__)
    result.add_argument("--clang", help="path to Clang with a riscv64 target")
    result.add_argument("--ld", help="path to ld.lld")
    result.add_argument("--qemu", help="path to qemu-system-riscv64")
    result.add_argument(
        "--out-dir",
        type=Path,
        help="keep products here; default uses an automatically removed temp dir",
    )
    result.add_argument("--timeout", type=float, default=8.0)
    return result


def checked_output_path(source_dir: Path, requested: Path) -> Path:
    output_path = requested.expanduser().resolve()
    if output_path == source_dir or source_dir in output_path.parents:
        raise LabError("--out-dir must be outside the source directory")
    return output_path


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    if args.timeout <= 0 or args.timeout > 60:
        raise SystemExit("--timeout must be in (0, 60]")
    source_dir = Path(__file__).resolve().parent
    clang = resolve_tool(args.clang, "clang")
    linker = resolve_tool(args.ld, "ld.lld")
    qemu = resolve_tool(args.qemu, "qemu-system-riscv64")
    print(f"CLANG VERSION: {first_version_line(clang)}")
    print(f"LLD VERSION: {first_version_line(linker)}")
    print(f"QEMU VERSION: {first_version_line(qemu)}")

    if args.out_dir is None:
        directory_context = tempfile.TemporaryDirectory(prefix="sv39-lab-")
        cleanup = True
    else:
        output_path = checked_output_path(source_dir, args.out_dir)
        output_path.mkdir(parents=True, exist_ok=True)
        directory_context = contextlib.nullcontext(str(output_path))
        cleanup = False

    with directory_context as directory:
        out_dir = Path(directory).resolve()
        print(f"BUILD DIR: {out_dir}")
        elf = build(source_dir, out_dir, clang, linker)
        symbols = elf_symbols(elf)
        returncode, output = run_qemu(qemu, elf, args.timeout)
        print("--- guest stdout begin ---")
        print(output, end="" if output.endswith("\n") else "\n")
        print("--- guest stdout end ---")
        verify_output(returncode, output, symbols)
        print(f"VERIFY PASS: QEMU exit=0; {len(expected_patterns(symbols))} exact ordered lines")

    if cleanup:
        print("CLEANUP PASS: automatic temporary build directory removed")
    else:
        print(f"CLEANUP NOTE: kept caller-selected build directory {args.out_dir.resolve()}")
    return 0


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except LabError as error:
        raise SystemExit(f"ERROR: {error}") from error
```
