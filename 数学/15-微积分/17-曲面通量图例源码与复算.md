"""One dimensionless graph: r(u,v)=(u,v,u*u+v*v), D=[0,1]^2.

Editable SVG, PNG, CSV and JSON are generated locally; no network is used.
Use the existing NumPy/Matplotlib environment, --output DIR, and a writable
MPLCONFIGDIR. The drawn finite patch is illustrative, not an exact area formula.
"""
from __future__ import annotations

import argparse
import csv
import json
from pathlib import Path
import xml.etree.ElementTree as ET

import numpy as np


def points(u, v):
    u, v = np.broadcast_arrays(np.asarray(u, dtype=float), np.asarray(v, dtype=float))
    if not np.all(np.isfinite(u)) or not np.all(np.isfinite(v)):
        raise ValueError("parameters must be finite")
    return np.stack((u, v, u*u + v*v), axis=-1)


def tangents(u, v):
    p = points(u, v)
    one, zero = np.ones_like(p[..., 0]), np.zeros_like(p[..., 0])
    return (np.stack((one, zero, 2*p[..., 0]), axis=-1),
            np.stack((zero, one, 2*p[..., 1]), axis=-1))


def normal_area(u, v):
    return np.cross(*tangents(u, v))


def integrate_square(function, order=32):
    if isinstance(order, bool) or not isinstance(order, int) or order < 1:
        raise ValueError("quadrature order must be a positive integer")
    q, weights = np.polynomial.legendre.leggauss(order)
    u, v = np.meshgrid((q+1)/2, (q+1)/2, indexing="ij")
    values = np.asarray(function(u, v), dtype=float)
    if values.shape != u.shape or not np.all(np.isfinite(values)):
        raise ValueError("integrand must give one finite value per point")
    return float(np.sum(np.outer(weights, weights)*values)/4)


def exact_area():
    # Independent closed-form reference for this square, not the disk example.
    return float(1 + 7*np.log(5)/12 - np.arctan(4/3)/12)


def save_figure(fig, target, title, description):
    fig.savefig(target.with_suffix(".svg"), metadata={"Date": None})
    fig.savefig(target.with_suffix(".png"), dpi=100)
    ET.register_namespace("", "http://www.w3.org/2000/svg")
    root = ET.parse(target.with_suffix(".svg")).getroot()
    ns = "{http://www.w3.org/2000/svg}"
    root.set("role", "img")
    root.set("aria-labelledby", "figure-title figure-desc")
    for tag, identifier, text in [("title", "figure-title", title),
                                   ("desc", "figure-desc", description)]:
        node = ET.Element(ns+tag, {"id": identifier})
        node.text = text
        root.insert(0, node)
    ET.ElementTree(root).write(target.with_suffix(".svg"), encoding="utf-8", xml_declaration=True)

                         "svg.fonttype": "none", "svg.hashsalt": "wwl-surface-flux-v1",
                         "figure.facecolor": "#ffffff", "axes.facecolor": "#ffffff"})
    p = points(.5, .5)
    a, b = tangents(.5, .5)
    n = normal_area(.5, .5)
    delta = .2
    blue, orange, green = "#155eb0", "#b55212", "#207148"
    fig = plt.figure(figsize=(4.25, 5.1))
    ax = fig.add_axes([.16, .32, .78, .56])
    fig.text(.06, .955, "二维域：小矩形从哪里来？", fontsize=15)
    ax.plot([0, 1, 1, 0, 0], [0, 0, 1, 1, 0], color="#536170")
    ax.fill([.5, .7, .7, .5], [.5, .5, .7, .7], color="#dcecf6")
    ax.plot([.5, .7, .7, .5, .5], [.5, .5, .7, .7, .5], color="#263238")
    ax.annotate("", (.7, .5), (.5, .5), arrowprops={"arrowstyle": "->", "color": blue, "lw": 2})
    ax.annotate("", (.5, .7), (.5, .5), arrowprops={"arrowstyle": "->", "color": orange, "lw": 2, "linestyle": "--"})
    ax.text(.6, .43, "Δu", ha="center", color=blue)

    fig.text(.06, .025, "画面为局部；坐标无量纲；曲面片 ≠ 切平面片。", fontsize=10)
    save_figure(fig, output/"surface-patch-3d", "同一参数小矩形的三维像与切平面近似",
                "抛物面z=x平方加y平方上，深色曲面片为二维小矩形的像，浅色平行四边形是P点切平面近似。实线和虚线分别为两个切向边，黑箭头是朝上的叉积方向且缩放显示，绿色F竖直朝上。")
    plt.close(fig)
    return font_path


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=True)
    flux = integrate_square(lambda u, v: normal_area(u, v)[..., 2])
    area = integrate_square(lambda u, v: np.linalg.norm(normal_area(u, v), axis=-1))
    np.testing.assert_allclose(flux, 1, atol=1e-14, rtol=0)
    np.testing.assert_allclose(area, exact_area(), atol=2e-13, rtol=0)
    font_path = render(args.output)
    with (args.output/"surface-grid.csv").open("w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(["u", "v", "x", "y", "z", "N_x", "N_y", "N_z", "area_density", "flux_density_up"])
        for u in np.linspace(0, 1, 25):
            for v in np.linspace(0, 1, 25):
                n = normal_area(u, v)
                writer.writerow([u, v, *points(u, v), *n, np.linalg.norm(n), n[2]])

        "N": [-1, -1, 1], "local_area_density": float(np.sqrt(3)),
        "local_unit_normal": (np.array([-1, -1, 1])/np.sqrt(3)).tolist(),
        "flux_up_quadrature": flux, "flux_up_exact": 1, "flux_down_exact": -1,
        "area_quadrature_order_32": area, "area_exact": exact_area(),
        "area_formula": "1+(7*log(5)-atan(4/3))/12",

    print(json.dumps({"flux": flux, "area": area, "area_exact": exact_area(), "font": font_path}))


if __name__ == "__main__":
    main()

"""Independent geometry checks for the graph/flux illustration."""
import unittest
import numpy as np

import surface_flux_figures as model


class SurfaceFluxTests(unittest.TestCase):
    def test_derivatives_against_position_differences(self):
        h = 1e-5
        for u, v in [(0, 0), (.5, .5), (.2, .8), (1, 1)]:
            a, b = model.tangents(u, v)
            np.testing.assert_allclose(a, (model.points(u+h, v)-model.points(u-h, v))/(2*h), atol=2e-11)
            np.testing.assert_allclose(b, (model.points(u, v+h)-model.points(u, v-h))/(2*h), atol=2e-11)

    def test_gram_and_unit_normal_are_different_objects(self):
        a, b = model.tangents(.5, .5)
        n = model.normal_area(.5, .5)
        np.testing.assert_array_equal(n, [-1, -1, 1])
        gram = np.array([[a@a, a@b], [a@b, b@b]])
        self.assertAlmostEqual(np.linalg.det(gram), 3)
        self.assertAlmostEqual(n@n, np.linalg.det(gram))
        self.assertAlmostEqual(n@a, 0); self.assertAlmostEqual(n@b, 0)
        self.assertAlmostEqual(np.linalg.norm(n/np.linalg.norm(n)), 1)
        # Vertical field has projected magnitude 1/sqrt(3), not sqrt(3).
        self.assertAlmostEqual((n/np.linalg.norm(n))[2]*np.sqrt(3), 1)

    def test_flux_area_and_orientation_independent_references(self):
        area = model.integrate_square(lambda u, v: np.sqrt(1+4*u*u+4*v*v), 40)
        self.assertAlmostEqual(area, model.exact_area(), places=12)
        self.assertGreater(area, 1); self.assertLess(area, 3)
        up = model.integrate_square(lambda u, v: model.normal_area(u, v)[..., 2])
        down = model.integrate_square(lambda u, v: -model.normal_area(u, v)[..., 2])
        self.assertAlmostEqual(up, 1); self.assertAlmostEqual(down, -1)

    def test_coordinate_swap_requires_orientation_correction(self):
        # rho(s,t)=r(t,s); differentiate in the new order (s,t).
        a, b = model.tangents(.8, .2)
        swapped = np.cross(b, a)
        np.testing.assert_allclose(swapped, -model.normal_area(.8, .2))
        self.assertLess(swapped[2], 0)
        self.assertAlmostEqual((-swapped)[2], 1)

    def test_rescaling_preserves_surface_area_and_flux(self):
        # rho(s,t)=r(2s,t), 0<=s<=1/2, 0<=t<=1.
        # Map quadrature a in [0,1] to s=a/2: density factor 2 cancels ds=da/2.
        rescaled = model.integrate_square(lambda a, t: .5*np.linalg.norm(
            np.cross(2*model.tangents(a, t)[0], model.tangents(a, t)[1]), axis=-1))
        self.assertAlmostEqual(rescaled, model.exact_area(), places=12)
        flux = model.integrate_square(lambda a, t: .5*np.cross(
            2*model.tangents(a, t)[0], model.tangents(a, t)[1])[..., 2])
        self.assertAlmostEqual(flux, 1)

    def test_finite_patch_is_only_first_order_tangent_approximation(self):
        p = model.points(.5, .5)
        a, _ = model.tangents(.5, .5)
        errors = [np.linalg.norm(model.points(.5+h, .5)-p-h*a) for h in [.2, .1]]
        self.assertAlmostEqual(errors[0], .04)
        self.assertAlmostEqual(errors[1], .01)
        self.assertAlmostEqual(errors[0]/errors[1], 4)

    def test_transfer_nonvertical_field_and_new_domain(self):
        # r(s,t)=(2s,t,8s*s+t*t), downward orientation.
        def density(s, t):
            rs = np.stack((2*np.ones_like(s), np.zeros_like(s), 16*s), axis=-1)
            rt = np.stack((np.zeros_like(t), np.ones_like(t), 2*t), axis=-1)
            field = np.stack((2*s, np.zeros_like(s), np.ones_like(s)), axis=-1)
            return np.sum(field*(-np.cross(rs, rt)), axis=-1)
        self.assertAlmostEqual(model.integrate_square(density), 26/3, places=12)

    def test_invalid_inputs_are_rejected(self):
        for bad in [float("nan"), float("inf")]:
            with self.assertRaises(ValueError): model.points(bad, .5)
        for bad in [0, -1, 2.5, True]:
            with self.assertRaises(ValueError): model.integrate_square(lambda u, v: u, bad)
        with self.assertRaises(ValueError): model.integrate_square(lambda u, v: np.nan*u)


if __name__ == "__main__":
    unittest.main()
