## `boot.S`

<!-- code-sync: ./boot.S -->
```asm
    .section .text.start, "ax"
    .align  4
    .globl  _start
_start:
    # This experiment runs exactly one hart. Park any unexpected extra hart.
    csrr    t0, mhartid
    bnez    t0, .Lpark

    # The ELF loader normally zeroes .bss. Do it explicitly so the contract is
    # visible and does not depend on that convenience.
    la      t0, __bss_start
    la      t1, __bss_end
.Lzero_bss:
    bgeu    t0, t1, .Lbss_done
    sd      zero, 0(t0)
    addi    t0, t0, 8
    j       .Lzero_bss
.Lbss_done:
    la      sp, m_stack_top
    la      t0, machine_trap_entry
    csrw    mtvec, t0
    call    machine_banner

    # Let S/U execute, load, store, and fetch throughout the QEMU virt address
    # space. This is deliberately broad because the lab does not build page
    # tables; README.md states the resulting isolation limit.
    li      t0, -1
    csrw    pmpaddr0, t0
    li      t0, 0x0f              # TOR, read/write/execute
    csrw    pmpcfg0, t0

    # Delegate only the two U-mode synchronous exceptions used by the lab:
    # illegal instruction (2) and environment call from U-mode (8).
    li      t0, ((1 << 2) | (1 << 8))
    csrw    medeleg, t0
    csrw    mideleg, zero

    la      t0, supervisor_entry
    csrw    mepc, t0
    csrr    t0, mstatus
    li      t1, (3 << 11)
    not     t1, t1
    and     t0, t0, t1
    li      t1, (1 << 11)         # MPP = S
    or      t0, t0, t1
    csrw    mstatus, t0
    mret

supervisor_entry:
    la      sp, s_stack_top
    call    supervisor_main
    j       .Lfatal_return

    .globl  machine_trap_entry
machine_trap_entry:
    la      sp, m_stack_top
    call    machine_trap

.Lfatal_return:
    li      a0, 0x7f
    call    qemu_fail

.Lpark:
    wfi
    j       .Lpark
```

## `trap.S`

<!-- code-sync: ./trap.S -->
```asm
#include "trapframe_offsets.h"

    .section .text.trap, "ax"
    .align  4
    .globl  supervisor_trap_entry
supervisor_trap_entry:
    # On entry from U, sscratch holds the trap-frame address and sp is the
    # user's stack. The atomic swap obtains a trusted base without losing sp.
    csrrw   sp, sscratch, sp

    sd      ra, TF_RA(sp)
    sd      gp, TF_GP(sp)
    sd      tp, TF_TP(sp)
    sd      t0, TF_T0(sp)
    sd      t1, TF_T1(sp)
    sd      t2, TF_T2(sp)
    sd      s0, TF_S0(sp)
    sd      s1, TF_S1(sp)
    sd      a0, TF_A0(sp)
    sd      a1, TF_A1(sp)
    sd      a2, TF_A2(sp)
    sd      a3, TF_A3(sp)
    sd      a4, TF_A4(sp)
    sd      a5, TF_A5(sp)
    sd      a6, TF_A6(sp)
    sd      a7, TF_A7(sp)
    sd      s2, TF_S2(sp)
    sd      s3, TF_S3(sp)
    sd      s4, TF_S4(sp)
    sd      s5, TF_S5(sp)
    sd      s6, TF_S6(sp)
    sd      s7, TF_S7(sp)
    sd      s8, TF_S8(sp)
    sd      s9, TF_S9(sp)
    sd      s10, TF_S10(sp)
    sd      s11, TF_S11(sp)
    sd      t3, TF_T3(sp)
    sd      t4, TF_T4(sp)
    sd      t5, TF_T5(sp)
    sd      t6, TF_T6(sp)

    # t0 is safe to reuse only after its user value is in the frame.
    csrr    t0, sscratch
    sd      t0, TF_SP(sp)
    csrr    t0, sepc
    sd      t0, TF_SEPC(sp)
    csrr    t0, scause
    sd      t0, TF_SCAUSE(sp)
    csrr    t0, stval
    sd      t0, TF_STVAL(sp)

    mv      a0, sp
    la      sp, s_stack_top
    call    supervisor_trap

    # supervisor_trap returns the frame pointer in a0. It may have changed a0
    # and sepc, but all other saved integer registers must round-trip.
    mv      sp, a0
    ld      t0, TF_SEPC(sp)
    csrw    sepc, t0
    csrw    sscratch, sp

    ld      ra, TF_RA(sp)
    ld      gp, TF_GP(sp)
    ld      tp, TF_TP(sp)
    ld      t0, TF_T0(sp)
    ld      t1, TF_T1(sp)
    ld      t2, TF_T2(sp)
    ld      s0, TF_S0(sp)
    ld      s1, TF_S1(sp)
    ld      a0, TF_A0(sp)
    ld      a1, TF_A1(sp)
    ld      a2, TF_A2(sp)
    ld      a3, TF_A3(sp)
    ld      a4, TF_A4(sp)
    ld      a5, TF_A5(sp)
    ld      a6, TF_A6(sp)
    ld      a7, TF_A7(sp)
    ld      s2, TF_S2(sp)
    ld      s3, TF_S3(sp)
    ld      s4, TF_S4(sp)
    ld      s5, TF_S5(sp)
    ld      s6, TF_S6(sp)
    ld      s7, TF_S7(sp)
    ld      s8, TF_S8(sp)
    ld      s9, TF_S9(sp)
    ld      s10, TF_S10(sp)
    ld      s11, TF_S11(sp)
    ld      t3, TF_T3(sp)
    ld      t4, TF_T4(sp)
    ld      t5, TF_T5(sp)
    ld      t6, TF_T6(sp)
    ld      sp, TF_SP(sp)
    sret
```

## `user.S`

```asm
#include "trapframe_offsets.h"

    .section .text, "ax"
    .align  4
    .globl  user_start
user_start:
    la      sp, user_stack_top

    # Assign every observable integer register (except sp/a0/a7, which have
    # protocol values) a distinct x-register-number sentinel.
    li      ra, 101
    li      gp, 103
    li      tp, 104
    li      t0, 105
    li      t1, 106
    li      t2, 107
    li      s0, 108
    li      s1, 109
    li      a1, 111
    li      a2, 112
    li      a3, 113
    li      a4, 114
    li      a5, 115
    li      a6, 116
    li      s2, 118
    li      s3, 119
    li      s4, 120
    li      s5, 121
    li      s6, 122
    li      s7, 123
    li      s8, 124
    li      s9, 125
    li      s10, 126
    li      s11, 127
    li      t3, 128
    li      t4, 129
    li      t5, 130
    li      t6, 131
    li      a0, 41
    li      a7, 1
    ecall

    # Snapshot the returned register file before using any register for a
    # comparison. Preserve t6 temporarily before reusing it as the base.
    sd      t6, -8(sp)
    la      t6, user_snapshot
    sd      ra, TF_RA(t6)
    sd      sp, TF_SP(t6)
    sd      gp, TF_GP(t6)
    sd      tp, TF_TP(t6)
    sd      t0, TF_T0(t6)
    sd      t1, TF_T1(t6)
    sd      t2, TF_T2(t6)
    sd      s0, TF_S0(t6)
    sd      s1, TF_S1(t6)
    sd      a0, TF_A0(t6)
    sd      a1, TF_A1(t6)
    sd      a2, TF_A2(t6)
    sd      a3, TF_A3(t6)
    sd      a4, TF_A4(t6)
    sd      a5, TF_A5(t6)
    sd      a6, TF_A6(t6)
    sd      a7, TF_A7(t6)
    sd      s2, TF_S2(t6)
    sd      s3, TF_S3(t6)
    sd      s4, TF_S4(t6)
    sd      s5, TF_S5(t6)
    sd      s6, TF_S6(t6)
    sd      s7, TF_S7(t6)
    sd      s8, TF_S8(t6)
    sd      s9, TF_S9(t6)
    sd      s10, TF_S10(t6)
    sd      s11, TF_S11(t6)
    sd      t3, TF_T3(t6)
    sd      t4, TF_T4(t6)
    sd      t5, TF_T5(t6)
    ld      t5, -8(sp)
    sd      t5, TF_T6(t6)

    li      a0, 1
    li      a7, 2
    ecall

    # Unknown number: this is still scause=8 and reaches the syscall
    # dispatcher. Only the dispatch result is an error.
    li      a0, 7
    li      a7, 99
    ecall
    li      t6, -38
    bne     a0, t6, .Lfail_unknown
    li      a0, 1
    li      a7, 3
    ecall

    # sstatus is an S-mode CSR. Reading it in U mode is not an ecall; it raises
    # illegal-instruction scause=2. The handler skips this known 4-byte probe.
    csrr    a0, sstatus
    li      t6, -2
    bne     a0, t6, .Lfail_illegal
    li      a0, 1
    li      a7, 4
    ecall

    li      a0, 0
    li      a7, 5
    ecall
    j       .Lfail_returned

.Lfail_normal:
    li      a0, 1
    j       .Lreport_fail
.Lfail_register:
    li      a0, 2
    j       .Lreport_fail
.Lfail_unknown:
    li      a0, 3
    j       .Lreport_fail
.Lfail_illegal:
    li      a0, 4
    j       .Lreport_fail
.Lfail_returned:
    li      a0, 5
.Lreport_fail:
    li      a7, 6
    ecall
    j       .Lfail_returned

    .section .bss, "aw", @nobits
    .balign 16
    .globl  user_snapshot
user_snapshot:
    .skip   TF_SIZE
```

## `kernel.c`

```c
#include "trapframe.h"

#define UART0_BASE       0x10000000UL
#define UART_THR         0
#define UART_LSR         5
#define UART_LSR_TX_IDLE 0x20
#define SIFIVE_TEST      0x00100000UL
#define FINISHER_FAIL    0x3333U
#define FINISHER_PASS    0x5555U

#define SSTATUS_SPP      (1UL << 8)
#define SSTATUS_SPIE     (1UL << 5)
#define SCAUSE_ILLEGAL   2UL
#define SCAUSE_U_ECALL   8UL
#define INSN_CSRR_A0_SSTATUS 0x10002573U

#define SYS_ADD_ONE          1UL
#define SYS_CHECK_REGISTERS  2UL
#define SYS_CHECK_UNKNOWN    3UL
#define SYS_CHECK_ILLEGAL    4UL
#define SYS_FINISH           5UL
#define SYS_FAIL             6UL

#define OFFSETOF(type, member) __builtin_offsetof(type, member)
#define CHECK_OFFSET(member, expected) \
    _Static_assert(OFFSETOF(struct trapframe, member) == (expected), \
                   "trapframe offset mismatch: " #member)

CHECK_OFFSET(ra, 0);
CHECK_OFFSET(sp, 8);
CHECK_OFFSET(t0, 32);
CHECK_OFFSET(s0, 56);
CHECK_OFFSET(a0, 72);
CHECK_OFFSET(a7, 128);
CHECK_OFFSET(s11, 208);
CHECK_OFFSET(t6, 240);
CHECK_OFFSET(sepc, 248);
CHECK_OFFSET(scause, 256);
CHECK_OFFSET(stval, 264);
_Static_assert(sizeof(struct trapframe) == 272, "trapframe size mismatch");

extern char supervisor_trap_entry[];
extern char trapframe_area[];
extern char user_start[];
extern char user_stack_top[];
extern struct trapframe user_snapshot;

static volatile u64 normal_seen;
static volatile u64 registers_seen;
static volatile u64 unknown_seen;
static volatile u64 unknown_checked;
static volatile u64 illegal_seen;
static volatile u64 illegal_checked;

static inline u64 csr_read_mcause(void) {
    u64 value;
    __asm__ volatile("csrr %0, mcause" : "=r"(value));
    return value;
}

static inline u64 csr_read_mepc(void) {
    u64 value;
    __asm__ volatile("csrr %0, mepc" : "=r"(value));
    return value;
}

static inline u64 csr_read_mtval(void) {
    u64 value;
    __asm__ volatile("csrr %0, mtval" : "=r"(value));
    return value;
}

static inline u64 csr_read_sstatus(void) {
    u64 value;
    __asm__ volatile("csrr %0, sstatus" : "=r"(value));
    return value;
}

static inline void csr_write_satp(u64 value) {
    __asm__ volatile("csrw satp, %0" : : "r"(value) : "memory");
}

static inline void csr_write_stvec(u64 value) {
    __asm__ volatile("csrw stvec, %0" : : "r"(value) : "memory");
}

static inline void csr_write_sscratch(u64 value) {
    __asm__ volatile("csrw sscratch, %0" : : "r"(value) : "memory");
}

static inline void csr_write_sepc(u64 value) {
    __asm__ volatile("csrw sepc, %0" : : "r"(value) : "memory");
}

static inline void csr_write_sstatus(u64 value) {
    __asm__ volatile("csrw sstatus, %0" : : "r"(value) : "memory");
}

static void uart_putc(char ch) {
    volatile unsigned char *uart = (volatile unsigned char *)UART0_BASE;
    while ((uart[UART_LSR] & UART_LSR_TX_IDLE) == 0) {
    }
    uart[UART_THR] = (unsigned char)ch;
}

static void putstr(const char *text) {
    while (*text != '\0') {
        uart_putc(*text++);
    }
}

static void put_u64(u64 value) {
    char digits[21];
    unsigned int count = 0;
    if (value == 0) {
        uart_putc('0');
        return;
    }
    while (value != 0) {
        digits[count++] = (char)('0' + value % 10);
        value /= 10;
    }
    while (count != 0) {
        uart_putc(digits[--count]);
    }
}

static void put_i64(i64 value) {
    if (value < 0) {
        uart_putc('-');
        put_u64((u64)(-value));
    } else {
        put_u64((u64)value);
    }
}

static void put_hex(u64 value) {
    static const char hex[] = "0123456789abcdef";
    int shift;
    putstr("0x");
    for (shift = 60; shift >= 0; shift -= 4) {
        uart_putc(hex[(value >> (unsigned int)shift) & 0xf]);
    }
}

__attribute__((noreturn)) void qemu_fail(u64 code) {
    volatile unsigned int *finisher = (volatile unsigned int *)SIFIVE_TEST;
    *finisher = ((unsigned int)(code & 0xffffUL) << 16) | FINISHER_FAIL;
    for (;;) {
        __asm__ volatile("wfi");
    }
}

static __attribute__((noreturn)) void qemu_pass(void) {
    volatile unsigned int *finisher = (volatile unsigned int *)SIFIVE_TEST;
    *finisher = FINISHER_PASS;
    for (;;) {
        __asm__ volatile("wfi");
    }
}

void machine_banner(void) {
    putstr("MODE M: boot hart=0, delegate causes 2 and 8\n");
}

__attribute__((noreturn)) void machine_trap(void) {
    putstr("RESULT FAIL: unexpected M trap mcause=");
    put_hex(csr_read_mcause());
    putstr(" mepc=");
    put_hex(csr_read_mepc());
    putstr(" mtval=");
    put_hex(csr_read_mtval());
    uart_putc('\n');
    qemu_fail(0x10);
}

__attribute__((noreturn)) void supervisor_main(void) {
    u64 status;
    putstr("MODE S: stvec/sscratch installed, preparing sret\n");
    csr_write_satp(0);
    csr_write_stvec((u64)supervisor_trap_entry);
    csr_write_sscratch((u64)trapframe_area);
    csr_write_sepc((u64)user_start);
    status = csr_read_sstatus();
    status &= ~SSTATUS_SPP;
    status |= SSTATUS_SPIE;
    csr_write_sstatus(status);
    putstr("MODE U: entered by sret\n");
    __asm__ volatile("sret");
    qemu_fail(0x11);
}

static void print_trap(const char *kind, const struct trapframe *tf) {
    putstr("TRAP kind=");
    putstr(kind);
    putstr(" scause=");
    put_u64(tf->scause);
    putstr(" sepc=");
    put_hex(tf->sepc);
    putstr(" stval=");
    put_hex(tf->stval);
    putstr(" origin.SPP=0 saved.SPIE=1");
    uart_putc('\n');
}

static int register_snapshot_valid(void) {
    const struct trapframe *r = &user_snapshot;
    return r->ra == 101 && r->sp == (u64)user_stack_top && r->gp == 103 &&
           r->tp == 104 && r->t0 == 105 && r->t1 == 106 && r->t2 == 107 &&
           r->s0 == 108 && r->s1 == 109 && r->a0 == 42 && r->a1 == 111 &&
           r->a2 == 112 && r->a3 == 113 && r->a4 == 114 && r->a5 == 115 &&
           r->a6 == 116 && r->a7 == 1 && r->s2 == 118 && r->s3 == 119 &&
           r->s4 == 120 && r->s5 == 121 && r->s6 == 122 && r->s7 == 123 &&
           r->s8 == 124 && r->s9 == 125 && r->s10 == 126 && r->s11 == 127 &&
           r->t3 == 128 && r->t4 == 129 && r->t5 == 130 && r->t6 == 131;
}

static void fail_from_user(u64 code) {
    putstr("RESULT FAIL: user-check code=");
    put_u64(code);
    uart_putc('\n');
    qemu_fail(0x20 + code);
}

struct trapframe *supervisor_trap(struct trapframe *tf) {
    u64 number;
    i64 argument;

    if ((csr_read_sstatus() & (SSTATUS_SPP | SSTATUS_SPIE)) != SSTATUS_SPIE) {
        putstr("RESULT FAIL: expected trap entry SPP=0 and SPIE=1\n");
        qemu_fail(0x30);
    }

    if (tf->scause == SCAUSE_U_ECALL) {
        print_trap("ecall", tf);
        tf->sepc += 4;
        number = tf->a7;
        argument = (i64)tf->a0;

        if (number == SYS_ADD_ONE) {
            tf->a0 = (u64)(argument + 1);
            normal_seen = 1;
            putstr("SYSCALL number=1 arg0=");
            put_i64(argument);
            putstr(" return=");
            put_i64((i64)tf->a0);
            uart_putc('\n');
        } else if (number == SYS_CHECK_REGISTERS) {
            if (argument != 1 || normal_seen == 0 || !register_snapshot_valid()) {
                fail_from_user(0x41);
            }
            registers_seen = 1;
            tf->a0 = 0;
            putstr("CHECK registers=PASS\n");
        } else if (number == SYS_CHECK_UNKNOWN) {
            if (argument != 1 || unknown_seen == 0) {
                fail_from_user(0x42);
            }
            unknown_checked = 1;
            tf->a0 = 0;
            putstr("CHECK unknown-return=PASS\n");
        } else if (number == SYS_CHECK_ILLEGAL) {
            if (argument != 1 || illegal_seen == 0) {
                fail_from_user(0x43);
            }
            illegal_checked = 1;
            tf->a0 = 0;
            putstr("CHECK illegal-recovery=PASS\n");
        } else if (number == SYS_FINISH) {
            if (normal_seen && registers_seen && unknown_seen &&
                unknown_checked && illegal_seen && illegal_checked) {
                putstr("RESULT PASS\n");
                qemu_pass();
            }
            fail_from_user(0x44);
        } else if (number == SYS_FAIL) {
            fail_from_user((u64)argument);
        } else {
            unknown_seen = 1;
            tf->a0 = (u64)(i64)-38;
            putstr("SYSCALL number=");
            put_u64(number);
            putstr(" arg0=");
            put_i64(argument);
            putstr(" return=-38 status=unknown\n");
        }
        return tf;
    }

    if (tf->scause == SCAUSE_ILLEGAL) {
        unsigned int instruction = *(const volatile unsigned int *)tf->sepc;
        print_trap("illegal", tf);
        if (instruction != INSN_CSRR_A0_SSTATUS) {
            putstr("RESULT FAIL: illegal instruction was not CSR probe, insn=");
            put_hex(instruction);
            uart_putc('\n');
            qemu_fail(0x50);
        }
        illegal_seen = 1;
        tf->sepc += 4;
        tf->a0 = (u64)(i64)-2;
        putstr("ILLEGAL instruction=csrr-a0-sstatus action=skip return=-2\n");
        return tf;
    }

    putstr("RESULT FAIL: unexpected S trap scause=");
    put_hex(tf->scause);
    putstr(" sepc=");
    put_hex(tf->sepc);
    putstr(" stval=");
    put_hex(tf->stval);
    uart_putc('\n');
    qemu_fail(0x60);
}
```

```python
#!/usr/bin/env python3
"""Build a freestanding RV64 ELF, run it on QEMU virt, and verify the trace."""

from __future__ import annotations

import argparse
import contextlib
import os
from pathlib import Path
import shlex
import shutil
import signal
import struct
import subprocess
import tempfile


class LabError(RuntimeError):
    pass


TRACE_STEPS = (
    "MODE M: boot hart=0, delegate causes 2 and 8",
    "MODE S: stvec/sscratch installed, preparing sret",
    "MODE U: entered by sret",
    "TRAP kind=ecall scause=8",
    "origin.SPP=0 saved.SPIE=1",
    "SYSCALL number=1 arg0=41 return=42",
    "CHECK registers=PASS",
    "SYSCALL number=99 arg0=7 return=-38 status=unknown",
    "CHECK unknown-return=PASS",
    "TRAP kind=illegal scause=2",
    "origin.SPP=0 saved.SPIE=1",
    "ILLEGAL instruction=csrr-a0-sstatus action=skip return=-2",
    "CHECK illegal-recovery=PASS",
    "RESULT PASS",
)


def resolve_tool(value: str | None, fallback: str) -> Path:
    candidate = value or shutil.which(fallback)
    if not candidate:
        raise LabError(f"tool not found: pass --{fallback.split('-')[0]} explicitly")
    path = Path(candidate).expanduser().resolve()
    if not path.is_file() or not os.access(path, os.X_OK):
        raise LabError(f"tool is not an executable file: {path}")
    return path


def run_checked(command: list[str], *, cwd: Path) -> str:
    print("+", shlex.join(command))
    result = subprocess.run(
        command,
        cwd=cwd,
        text=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        check=False,
    )
    if result.returncode != 0:
        raise LabError(
            f"command failed with exit {result.returncode}:\n{result.stdout}"
        )
    return result.stdout


def check_riscv_elf(path: Path) -> None:
    header = path.read_bytes()[:64]
    if len(header) < 20 or header[:4] != b"\x7fELF":
        raise LabError("linker output is not an ELF file")
    if header[4] != 2 or header[5] != 1:
        raise LabError("ELF must be 64-bit little-endian")
    machine = struct.unpack_from("<H", header, 18)[0]
    if machine != 243:
        raise LabError(f"ELF e_machine is {machine}, expected RISC-V (243)")


def build(source_dir: Path, out_dir: Path, clang: Path, linker: Path) -> Path:
    common = [
        str(clang),
        "--target=riscv64-unknown-elf",
        "-march=rv64im_zicsr",
        "-mabi=lp64",
        "-mcmodel=medany",
        "-mno-relax",
        "-msmall-data-limit=0",
        "-ffreestanding",
        "-fno-builtin",
        "-fno-pic",
        "-fno-stack-protector",
        "-ffunction-sections",
        "-fdata-sections",
        "-O2",
        "-Wall",
        "-Wextra",
        "-Werror",
        f"-I{source_dir}",
    ]
    objects: list[Path] = []
    for source_name in ("boot.S", "trap.S", "user.S", "kernel.c"):
        source = source_dir / source_name
        obj = out_dir / (source.stem + ".o")
        run_checked(common + ["-c", str(source), "-o", str(obj)], cwd=source_dir)
        objects.append(obj)

    elf = out_dir / "kernel.elf"
    map_file = out_dir / "kernel.map"
    link_command = [
        str(linker),
        "-m",
        "elf64lriscv",
        "--gc-sections",
        "--no-relax",
        "-z",
        "max-page-size=4096",
        "-T",
        str(source_dir / "linker.ld"),
        "-Map",
        str(map_file),
        "-o",
        str(elf),
        *(str(obj) for obj in objects),
    ]
    run_checked(link_command, cwd=source_dir)
    check_riscv_elf(elf)
    print(f"ELF CHECK PASS: {elf} is ELF64 little-endian RISC-V")
    return elf


def run_qemu(qemu: Path, elf: Path, timeout_seconds: float) -> tuple[int, str]:
    command = [
        str(qemu),
        "-machine",
        "virt",
        "-accel",
        "tcg",
        "-smp",
        "1",
        "-m",
        "128M",
        "-bios",
        "none",
        "-kernel",
        str(elf),
        "-display",
        "none",
        "-serial",
        "stdio",
        "-monitor",
        "none",
        "-net",
        "none",
        "-no-reboot",
    ]
    print("+", shlex.join(command))
    process = subprocess.Popen(
        command,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        start_new_session=True,
    )
    try:
        raw_output, _ = process.communicate(timeout=timeout_seconds)
    except subprocess.TimeoutExpired as error:
        os.killpg(process.pid, signal.SIGKILL)
        raw_output, _ = process.communicate()
        output = raw_output.decode("utf-8", errors="replace")
        raise LabError(
            f"QEMU exceeded {timeout_seconds:.1f}s and was killed; partial output:\n"
            f"{output}"
        ) from error
    finally:
        if process.poll() is None:
            os.killpg(process.pid, signal.SIGKILL)
            process.wait()
    return process.returncode, raw_output.decode("utf-8", errors="replace")


def verify_output(returncode: int, output: str) -> None:
    if returncode != 0:
        raise LabError(f"QEMU exited with {returncode}, expected 0\n{output}")
    if "RESULT FAIL" in output:
        raise LabError(f"guest reported failure:\n{output}")
    position = 0
    for step in TRACE_STEPS:
        found = output.find(step, position)
        if found < 0:
            raise LabError(f"missing or out-of-order trace step: {step!r}\n{output}")
        position = found + len(step)
    if output.count("RESULT PASS") != 1:
        raise LabError("guest must emit exactly one RESULT PASS")


def parser() -> argparse.ArgumentParser:
    result = argparse.ArgumentParser(description=__doc__)
    result.add_argument("--clang", help="path to a Clang with riscv64 target support")
    result.add_argument("--ld", help="path to ld.lld")
    result.add_argument("--qemu", help="path to qemu-system-riscv64")
    result.add_argument(
        "--out-dir",
        type=Path,
        help="keep build products here; default is an automatically removed temp dir",
    )
    result.add_argument("--timeout", type=float, default=8.0, help="QEMU timeout in seconds")
    return result


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    if args.timeout <= 0 or args.timeout > 60:
        raise SystemExit("--timeout must be in (0, 60]")

    source_dir = Path(__file__).resolve().parent
    clang = resolve_tool(args.clang, "clang")
    linker = resolve_tool(args.ld, "ld.lld")
    qemu = resolve_tool(args.qemu, "qemu-system-riscv64")

    if args.out_dir is None:
        directory_context = tempfile.TemporaryDirectory(prefix="c4-riscv-")
        cleanup = True
    else:
        output_path = args.out_dir.expanduser().resolve()
        if output_path == source_dir or source_dir in output_path.parents:
            raise LabError("--out-dir must be outside the source directory")
        output_path.mkdir(parents=True, exist_ok=True)
        directory_context = contextlib.nullcontext(str(output_path))
        cleanup = False

    with directory_context as directory:
        out_dir = Path(directory).resolve()
        print(f"BUILD DIR: {out_dir}")
        elf = build(source_dir, out_dir, clang, linker)
        returncode, output = run_qemu(qemu, elf, args.timeout)
        print("--- guest stdout begin ---")
        print(output, end="" if output.endswith("\n") else "\n")
        print("--- guest stdout end ---")
        verify_output(returncode, output)
        print(f"VERIFY PASS: QEMU exit={returncode}; {len(TRACE_STEPS)} ordered checks")

    if cleanup:
        print("CLEANUP PASS: automatic temporary build directory removed")
    else:
        print(f"CLEANUP NOTE: kept caller-selected build directory {args.out_dir.resolve()}")
    return 0


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except LabError as error:
        raise SystemExit(f"ERROR: {error}") from error
```

```python
from __future__ import annotations

import unittest

import build_run


GOOD_TRACE = "\n".join(
    (
        "MODE M: boot hart=0, delegate causes 2 and 8",
        "MODE S: stvec/sscratch installed, preparing sret",
        "MODE U: entered by sret",
        "TRAP kind=ecall scause=8 sepc=0x80001000 stval=0x0 origin.SPP=0 saved.SPIE=1",

        "TRAP kind=ecall scause=8 sepc=0x80001004 stval=0x0",
        "CHECK registers=PASS",
        "TRAP kind=ecall scause=8 sepc=0x80001008 stval=0x0",
        "SYSCALL number=99 arg0=7 return=-38 status=unknown",
        "TRAP kind=ecall scause=8 sepc=0x8000100c stval=0x0",
        "CHECK unknown-return=PASS",
        "TRAP kind=illegal scause=2 sepc=0x80001010 stval=0x10002573 origin.SPP=0 saved.SPIE=1",
        "ILLEGAL instruction=csrr-a0-sstatus action=skip return=-2",
        "TRAP kind=ecall scause=8 sepc=0x80001014 stval=0x0",
        "CHECK illegal-recovery=PASS",
        "TRAP kind=ecall scause=8 sepc=0x80001018 stval=0x0",
        "RESULT PASS",
    )
) + "\n"


class TraceContractTests(unittest.TestCase):
    def test_accepts_full_ordered_trace_and_zero_exit(self) -> None:
        build_run.verify_output(0, GOOD_TRACE)

    def test_rejects_nonzero_emulator_exit(self) -> None:
        with self.assertRaisesRegex(build_run.LabError, "exited with 7"):
            build_run.verify_output(7, GOOD_TRACE)

    def test_rejects_guest_failure_even_if_pass_text_is_present(self) -> None:
        with self.assertRaisesRegex(build_run.LabError, "guest reported failure"):
            build_run.verify_output(0, GOOD_TRACE + "RESULT FAIL: late failure\n")

    def test_each_semantic_checkpoint_is_required(self) -> None:
        checkpoints = {
            "normal return": "SYSCALL number=1 arg0=41 return=42",
            "register restoration": "CHECK registers=PASS",
            "unknown syscall": "SYSCALL number=99 arg0=7 return=-38 status=unknown",
            "illegal CSR cause": "TRAP kind=illegal scause=2",
            "illegal CSR recovery": "CHECK illegal-recovery=PASS",
        }
        for label, checkpoint in checkpoints.items():
            with self.subTest(label=label):
                damaged = GOOD_TRACE.replace(checkpoint, "REMOVED", 1)
                with self.assertRaises(build_run.LabError):
                    build_run.verify_output(0, damaged)

    def test_rejects_out_of_order_trace(self) -> None:
        first = "CHECK registers=PASS"
        second = "SYSCALL number=99 arg0=7 return=-38 status=unknown"
        damaged = GOOD_TRACE.replace(first, "TEMP", 1).replace(second, first, 1)
        damaged = damaged.replace("TEMP", second, 1)
        with self.assertRaises(build_run.LabError):
            build_run.verify_output(0, damaged)


if __name__ == "__main__":
    unittest.main()
```
