```c
#include "sum.h"

#include <limits.h>

/* Reject a target where uint32_t would promote to a signed int before +=. */
#if UINT32_MAX <= INT_MAX
#error "sum_u32 needs uint32_t not representable by int"
#endif

uint32_t sum_u32(const uint32_t values[], size_t count) {
    uint32_t total = 0;
    for (size_t index = 0; index < count; ++index) {
        total += values[index];
    }
    return total;
}
```

```python
#!/usr/bin/env python3
"""Build a small, bounded C program and record which build stage failed.

The script writes only inside a TemporaryDirectory, times every child process,
and treats tool absence as SKIP.  It is not a replacement for an object-file
format specification or a cross-platform linker test suite.
"""

from __future__ import annotations

import json
import os
import shutil
import subprocess
import sys
import tempfile
from dataclasses import asdict, dataclass
from pathlib import Path


ROOT = Path(__file__).resolve().parent
TIMEOUT_SECONDS = 15
COMMAND_LINE_TOOLS = Path("/Library/Developer/CommandLineTools")


def find_clang() -> str | None:
    """Prefer an independently installed clang over Apple's license-gated shim."""
    for candidate in ("/opt/homebrew/opt/llvm/bin/clang", "clang"):
        resolved = shutil.which(candidate) if candidate == "clang" else candidate
        if resolved and Path(resolved).is_file() and Path(resolved).stat().st_mode & 0o111:
            return resolved
    return None


def find_nm() -> str | None:
    """Prefer Homebrew LLVM's nm when Apple's developer-tool shim is unavailable."""
    for candidate in ("/opt/homebrew/opt/llvm/bin/llvm-nm", "llvm-nm", "nm"):
        resolved = shutil.which(candidate) if "/" not in candidate else candidate
        if resolved and Path(resolved).is_file() and Path(resolved).stat().st_mode & 0o111:
            return resolved
    return None


@dataclass(frozen=True)
class CommandResult:
    stage: str
    argv: list[str]
    returncode: int | None
    stdout: str
    stderr: str
    timed_out: bool


def child_environment() -> dict[str, str]:
    """Select already-installed CLT tools for this child only; change no system setting."""
    environment = dict(os.environ)
    if sys.platform == "darwin" and COMMAND_LINE_TOOLS.is_dir():
        environment["DEVELOPER_DIR"] = str(COMMAND_LINE_TOOLS)
    return environment


def output_text(value: str | bytes | None) -> str:
    """Keep partial diagnostics readable, including an incomplete UTF-8 suffix."""
    if value is None:
        return ""
    return value.decode("utf-8", errors="replace") if isinstance(value, bytes) else value


def run(stage: str, argv: list[str], directory: Path) -> CommandResult:
    try:
        completed = subprocess.run(
            argv,
            cwd=directory,
            text=True,
            encoding="utf-8",
            errors="replace",
            capture_output=True,
            timeout=TIMEOUT_SECONDS,
            check=False,
            env=child_environment(),
        )
        return CommandResult(stage, argv, completed.returncode,
                             completed.stdout, completed.stderr, False)
    except subprocess.TimeoutExpired as error:
        # TimeoutExpired may carry bytes even when run(text=True) was requested.
        # Preserve the timeout and partial output; replacement is diagnostic only.
        return CommandResult(stage, argv, None,
                             output_text(error.stdout), output_text(error.stderr), True)


def copy_sources(directory: Path) -> None:
    for name in ("sum.h", "sum.c", "main.c", "duplicate_sum.c"):
        shutil.copy2(ROOT / name, directory / name)


def compile_unit(clang: str, source: str, output: str, directory: Path) -> CommandResult:
    return run(
        f"compile:{source}",
        [clang, "-std=c17", "-Wall", "-Wextra", "-Werror", "-c", source, "-o", output],
        directory,
    )


def is_platform_tool_gate(result: CommandResult) -> bool:
    """Recognize the local Xcode-license refusal without accepting it as a link test."""
    return "not agreed to the Xcode license agreements" in result.stderr


def trace(clang: str | None = None, nm: str | None = None) -> dict[str, object]:
    clang = clang or find_clang()
    if clang is None or (Path(clang).is_absolute() and not Path(clang).is_file()):
        return {"status": "SKIP", "reason": "clang was not found; no compiler was installed."}

    nm = nm or find_nm()
    with tempfile.TemporaryDirectory(prefix="wwl-link-trace-") as raw_directory:
        directory = Path(raw_directory)
        copy_sources(directory)
        results: list[CommandResult] = []
        for source, output in (("sum.c", "sum.o"), ("main.c", "main.o"),
                               ("duplicate_sum.c", "duplicate_sum.o")):
            results.append(compile_unit(clang, source, output, directory))

        if any(result.returncode != 0 for result in results):
            return {"status": "FAIL", "reason": "a required translation unit did not compile",
                    "commands": [asdict(result) for result in results]}

        if nm is None:
            results.append(CommandResult("inspect-symbols", ["nm"], None, "", "", False))
            symbol_note = "SKIP: nm was not found; object symbols were not inspected."
        else:
            results.append(run("inspect-symbols", [nm, "sum.o", "main.o"], directory))
            symbol_note = "nm output is a platform-specific observation, not an ABI specification."
        file_tool = shutil.which("file")
        if file_tool is None:
            results.append(CommandResult("inspect-object-format", ["file"], None, "", "", False))
        else:
            results.append(run("inspect-object-format", [file_tool, "sum.o"], directory))

        success = run("link:success", [clang, "main.o", "sum.o", "-o", "sum_demo"], directory)
        results.append(success)
        if is_platform_tool_gate(success):
            return {
                "status": "SKIP",
                "reason": "the platform linker is unavailable because Xcode license acceptance is required; no license was changed.",
                "platform": sys.platform,
                "compiler": clang,
                "developer_dir": child_environment().get("DEVELOPER_DIR"),
                "commands": [asdict(result) for result in results],
            }
        if success.returncode == 0:
            results.append(run("run:success", [str(directory / "sum_demo")], directory))

        missing = run("link:missing-definition", [clang, "main.o", "-o", "missing_demo"], directory)
        duplicate = run("link:duplicate-definition",
                        [clang, "main.o", "sum.o", "duplicate_sum.o", "-o", "duplicate_demo"], directory)
        results.extend((missing, duplicate))
        passed = (
            success.returncode == 0
            and results[-3].returncode == 0
            and results[-3].stdout == "sum=100\n"
            and missing.returncode != 0
            and duplicate.returncode != 0
            and not any(result.timed_out for result in results)
        )
        return {
            "status": "PASS" if passed else "FAIL",
            "platform": sys.platform,
            "compiler": clang,
            "developer_dir": child_environment().get("DEVELOPER_DIR"),
            "symbol_note": symbol_note,
            "expectations": {
                "successful_program_stdout": "sum=100\\n",
                "missing_definition_must_fail_at_link": True,
                "duplicate_definition_must_fail_at_link": True,
            },
            "commands": [asdict(result) for result in results],
        }


def main() -> int:
    result = trace()
    print(json.dumps(result, indent=2, sort_keys=True))
    return 0 if result["status"] in {"PASS", "SKIP"} else 1


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
import json
import shutil
import subprocess
import sys
import unittest
from dataclasses import asdict
from pathlib import Path
from unittest.mock import patch

from build_trace import (COMMAND_LINE_TOOLS, CommandResult, ROOT, child_environment,
                         find_clang, find_nm, is_platform_tool_gate, run, trace)


class BuildTraceTests(unittest.TestCase):
    def test_timeout_output_is_serializable_diagnostic_text(self) -> None:
        cases = (
            (b"partial\n", b"error\n", "partial\n", "error\n"),
            (None, None, "", ""),
            (b"partial \xe4\xb8", b"error \xff", "partial \ufffd", "error \ufffd"),
        )
        for stdout, stderr, expected_stdout, expected_stderr in cases:
            with self.subTest(stdout=stdout, stderr=stderr):
                error = subprocess.TimeoutExpired(["fixture"], 0.1, stdout, stderr)
                with patch("build_trace.subprocess.run", side_effect=error):
                    result = run("timeout-fixture", ["fixture"], ROOT)
                restored = json.loads(json.dumps(asdict(result)))
                self.assertTrue(restored["timed_out"])
                self.assertIsNone(restored["returncode"])
                self.assertEqual(restored["stdout"], expected_stdout)
                self.assertEqual(restored["stderr"], expected_stderr)
                self.assertIsInstance(result.stdout, str)
                self.assertIsInstance(result.stderr, str)
                self.assertFalse(is_platform_tool_gate(result))

    def test_real_child_timeout_retains_partial_utf8_output(self) -> None:
        script = (
            "import os, time; "
            "os.write(1, b'partial \\xe4\\xb8'); "
            "os.write(2, b'error\\n'); time.sleep(5)"
        )
        with patch("build_trace.TIMEOUT_SECONDS", 0.2):
            result = run("timeout-child", [sys.executable, "-c", script], ROOT)
        self.assertTrue(result.timed_out)
        self.assertIsNone(result.returncode)
        self.assertEqual(result.stdout, "partial \ufffd")
        self.assertEqual(result.stderr, "error\n")
        self.assertEqual(json.loads(json.dumps(asdict(result)))["stdout"], result.stdout)

    def test_success_and_nonzero_exit_keep_text_and_returncode(self) -> None:
        for code in (0, 3):
            with self.subTest(returncode=code):
                script = f"import sys; print('ok'); print('failure', file=sys.stderr); sys.exit({code})"
                result = run("exit-fixture", [sys.executable, "-c", script], ROOT)
                self.assertEqual((result.returncode, result.stdout, result.stderr),
                                 (code, "ok\n", "failure\n"))
                self.assertFalse(result.timed_out)

    def test_missing_compiler_is_an_explicit_skip(self) -> None:
        self.assertEqual(trace(clang="/definitely/not/a/compiler")["status"], "SKIP")

    def test_llvm_nm_probe_has_no_side_effect(self) -> None:
        found = find_nm()
        self.assertTrue(found is None or Path(found).is_file())

    def test_child_environment_only_selects_existing_command_line_tools(self) -> None:
        environment = child_environment()
        if sys.platform == "darwin" and COMMAND_LINE_TOOLS.is_dir():
            self.assertEqual(environment.get("DEVELOPER_DIR"), str(COMMAND_LINE_TOOLS))

    @unittest.skipUnless(find_clang(), "usable clang is unavailable")
    def test_build_contract_and_failure_stages(self) -> None:
        result = trace()
        self.assertIn(result["status"], {"PASS", "SKIP"})
        commands = {entry["stage"]: entry for entry in result["commands"]}
        self.assertTrue(all(commands[key]["returncode"] == 0 for key in (
            "compile:sum.c", "compile:main.c", "compile:duplicate_sum.c")))
        if result["status"] == "PASS":
            self.assertEqual(commands["run:success"]["stdout"], "sum=100\n")
            self.assertNotEqual(commands["link:missing-definition"]["returncode"], 0)
            self.assertNotEqual(commands["link:duplicate-definition"]["returncode"], 0)
            self.assertFalse(any(entry["timed_out"] for entry in commands.values()))
        else:
            self.assertIn("platform linker", result["reason"])

    @unittest.skipUnless(find_clang(), "usable clang is unavailable")
    def test_cli_returns_machine_readable_trace(self) -> None:
        completed = subprocess.run(
            [sys.executable, str(ROOT / "build_trace.py")],
            text=True,
            capture_output=True,
            timeout=45,
            check=False,
        )
        self.assertEqual(completed.returncode, 0, completed.stderr)
        self.assertIn(json.loads(completed.stdout)["status"], {"PASS", "SKIP"})

    def test_xcode_license_gate_is_not_mistaken_for_symbol_failure(self) -> None:
        gated = CommandResult("link:success", ["clang"], 69, "",
                              "You have not agreed to the Xcode license agreements.", False)
        ordinary_failure = CommandResult("link:success", ["clang"], 1, "", "undefined symbol", False)
        self.assertTrue(is_platform_tool_gate(gated))
        self.assertFalse(is_platform_tool_gate(ordinary_failure))


if __name__ == "__main__":
    unittest.main()
```

```python
import re
import unittest
from pathlib import Path


ROOT = Path(__file__).resolve().parent
MIRROR = ROOT / "源码与测试镜像.md"
SYNCED = ("sum.h", "sum.c", "main.c", "duplicate_sum.c", "build_trace.py", "test_build_trace.py")


class CodeSyncTests(unittest.TestCase):
    def test_every_declared_source_matches_its_markdown_block(self) -> None:
        mirror = MIRROR.read_text(encoding="utf-8")
        for name in SYNCED:
            marker = re.escape(f"<!-- code-sync: ./{name} -->")
            pattern = marker + r"\n```[^\n]*\n(.*?)\n```"
            match = re.search(pattern, mirror, flags=re.DOTALL)
            self.assertIsNotNone(match, name)
            self.assertEqual(match.group(1), (ROOT / name).read_text(encoding="utf-8").rstrip("\n"), name)


if __name__ == "__main__":
    unittest.main()
```
