# 状态实验源码与验证

三个同目录Python文件是规范真源；此页只供阅读与正式code-sync。数学对象、运行和未验层见 [[信号与系统/80-实践/04-状态实现与隐藏模态/README|实验说明]]。修改真源后须同步此镜像，并在库外重新生成受影响数据与图。

数值支持边界：rank_report 在比较前要求奇异值与阈值均有限，否则明确 ValueError 并提示缩放，不报告秩0。例如 `1.7e308*[[1,1],[1,-1]]` 精确秩2，但奇异值幅度超机器范围；缩放后的两奇异值为 sqrt(2)，可独立复核。此实现不承诺任意极端矩阵都支持，也不把浮点数值秩当结构性认证。

"""Finite-dimensional continuous LTI teaching models; no plant identification.

States and outputs are normalized amplitudes, time is seconds. NumPy only.
Results use floating point Taylor scaling/squaring, not certified intervals.
"""
from __future__ import annotations
import argparse
import json
import math
from pathlib import Path
import numpy as np

def real_array(x, ndim):
    # Object dtype hides NumPy complex scalars from np.iscomplexobj(raw).
    # Reject before float conversion, including zero imaginary part/subclasses.
    try:
        raw = np.asarray(x)
        if raw.ndim != ndim or np.iscomplexobj(raw) or (raw.dtype == object and any(
                isinstance(v, (complex, np.complexfloating)) or np.iscomplexobj(v)
                for v in raw.flat)):
            raise ValueError('finite real array with specified dimension required')
        a = np.asarray(raw, dtype=float)
    except (TypeError, ValueError, OverflowError) as e:
        raise ValueError('real numeric array required') from e
    if not np.all(np.isfinite(a)):
        raise ValueError('nonfinite array')
    return a

def square(A):
    A = real_array(A, 2)
    if A.shape[0] == 0 or A.shape[0] != A.shape[1]:
        raise ValueError('nonempty square A required')
    return A

def time_value(T):
    if isinstance(T, (bool, np.bool_)) or not np.isscalar(T) or np.iscomplexobj(T):
        raise ValueError('finite nonnegative time required')
    T = float(T)
    if not math.isfinite(T) or T < 0:
        raise ValueError('finite nonnegative time required')
    return T

def finite_result(a):
    if not np.all(np.isfinite(a)):
        raise ValueError('numerical result is not representable')
    return a

def model(A, B, C, D):
    A, B, C, D = square(A), real_array(B,2), real_array(C,2), real_array(D,2)
    n=A.shape[0]
    if B.shape[0]!=n or C.shape[1]!=n or B.shape[1]<1 or C.shape[0]<1 or D.shape!=(C.shape[0],B.shape[1]):
        raise ValueError('A:n*n B:n*m C:p*n D:p*m with n,m,p >= 1')
    return A,B,C,D

def expm(A):
    """Taylor at ||A/2**s||_inf <= .5, then square; handles Jordan blocks.

    Explicit finite range checks; no arbitrary ill-conditioned accuracy guarantee.
    """
    A=square(A)
    with np.errstate(over='ignore',invalid='ignore',under='ignore'):
        norm=float(np.linalg.norm(A,np.inf))
        if not math.isfinite(norm):
            raise ValueError('matrix norm is not representable')
        s=max(0, int(math.ceil(math.log2(norm)+1))) if norm > .5 else 0
        scaled=np.ldexp(A,-s)
        result=term=np.eye(len(A))
        for k in range(1,81):
            term=term@scaled/k
            result=result+term
            if np.linalg.norm(term,np.inf) <= 2e-16*max(1.,float(np.linalg.norm(result,np.inf))):
                break
        else:
            raise ValueError('matrix series did not converge')
        for _ in range(s):
            result=finite_result(result@result)
    return finite_result(result)

def transition(A,t):
    A=square(A);t=time_value(t)
    with np.errstate(over='ignore',invalid='ignore'):
        At=finite_result(A*t)
    return expm(At)

def constant_response(A,B,C,D,t,x0,u):
    """Exact-form constant-input solution via an augmented matrix exponential.

    No A inverse; singular A including integrators is supported.
    """
    A,B,C,D=model(A,B,C,D);t=time_value(t)
    x0,u=real_array(x0,1),real_array(u,1)
    n,m=B.shape
    if x0.shape!=(n,) or u.shape!=(m,):
        raise ValueError('x0:n and u:m required')
    block=np.zeros((n+m,n+m));block[:n,:n]=A;block[:n,n:]=B
    E=transition(block,t)
    with np.errstate(over='ignore',invalid='ignore'):
        x=finite_result(E[:n,:n]@x0+E[:n,n:]@u)
        y=finite_result(C@x+D@u)
    return x,y

def structure(A,B,C):
    A,B,C=square(A),real_array(B,2),real_array(C,2)
    A,B,C,D=model(A,B,C,np.zeros((C.shape[0],B.shape[1])))
    n=len(A);power=np.eye(n);R=[];O=[]
    with np.errstate(over='ignore',invalid='ignore'):
        for k in range(n):
            R.append(finite_result(power@B));O.append(finite_result(C@power))
            power=finite_result(power@A)
    return np.hstack(R),np.vstack(O)

def rank_report(K, rtol=1e-10, atol=0.):
    K=real_array(K,2)
    if min(K.shape)<1 or not math.isfinite(rtol) or not math.isfinite(atol) or rtol<0 or atol<0:
        raise ValueError('nonempty matrix and finite nonnegative tolerances required')
    sv=np.linalg.svd(K,compute_uv=False)
    if not np.all(np.isfinite(sv)):
        raise ValueError('singular values are outside numerical support; rescale matrix')
    threshold=max(atol,rtol*float(sv[0]))
    if not math.isfinite(threshold):
        raise ValueError('rank threshold is outside numerical support')
    return dict(rank=int(np.count_nonzero(sv>threshold)),singular_values=sv.tolist(),
                threshold=threshold,rtol=rtol,atol=atol,
                evidence='numerical rank at declared coordinates/tolerance, not structural identification')

def gramian(A,B,T):
    """Finite reachability Gramian: exp [[A,BB^T],[0,-A^T]]T.

    Upper-right block times exp(A^T T) equals integral exp(A t)BB^T exp(A^T t)dt.
    """
    A=square(A);B=real_array(B,2);T=time_value(T);n=len(A)
    if B.shape[0]!=n or B.shape[1]<1:
        raise ValueError('B:n*m required')
    block=np.zeros((2*n,2*n))
    with np.errstate(over='ignore',invalid='ignore'):
        block[:n,:n]=A;block[:n,n:]=finite_result(B@B.T);block[n:,n:]=-A.T
        E=transition(block,T)
        W=finite_result(E[:n,n:]@E[:n,:n].T)
    return finite_result(W/2+W.T/2)

def observability_gramian(A,C,T):
    A=square(A);C=real_array(C,2)
    if C.shape[1]!=len(A) or C.shape[0]<1:
        raise ValueError('C:p*n required')
    return gramian(A.T,C.T,T)

def transfer(A,B,C,D,s):
    A,B,C,D=model(A,B,C,D)
    try:s=complex(s)
    except (TypeError,ValueError) as e:raise ValueError('finite complex s required') from e
    if not math.isfinite(s.real) or not math.isfinite(s.imag):raise ValueError('finite complex s required')
    try:
        with np.errstate(over='ignore',invalid='ignore'):
            return finite_result(C@np.linalg.solve(s*np.eye(len(A))-A,B)+D)
    except np.linalg.LinAlgError as e:
        raise ValueError('sI-A singular: use algebraically reduced transfer separately') from e

def markov(A,B,C,count=5):
    A,B,C=square(A),real_array(B,2),real_array(C,2)
    A,B,C,D=model(A,B,C,np.zeros((C.shape[0],B.shape[1])))
    if type(count) is not int or count<0:raise ValueError('nonnegative integer count required')
    terms=[];power=np.eye(len(A))
    with np.errstate(over='ignore',invalid='ignore'):
        for k in range(count):
            terms.append(finite_result(C@power@B))
            if k+1 < count:
                power=finite_result(power@A)
    return terms

def experiment():
    A=np.diag([-1.,1.]);B=np.array([[1.],[1.]]);C=np.array([[1.,0.]]);D=np.zeros((1,1))
    t=np.linspace(0,6,301)
    xs=[];ys=[];visible=[];ic=[]
    for ti in t:
        x,y=constant_response(A,B,C,D,ti,[0,0],[1]);xs.append(x);ys.append(y[0])
        _,v=constant_response(A,B,[[1,1e-3]],D,ti,[0,0],[1]);visible.append(v[0])
        _,z=constant_response(A,[[1],[0]],[[1,1]],D,ti,[0,1],[0]);ic.append(z[0])
    xs=np.array(xs);ys=np.array(ys)
    W=observability_gramian(np.diag([-1.,-2.]),[[1.,1.]],1.)
    W_hidden=observability_gramian(A,C,1.)
    r,o=structure(A,B,C)
    eps=1e-12;_,near=structure(A,B,[[1,eps]])
    _,scaled=structure(A,B,[[1,1]])
    S=np.diag([1.,1e12]);_,scaled=structure(S@A@np.linalg.inv(S),S@B,np.array([[1,1]])@np.linalg.inv(S))
    Q=np.linspace(-1,1,31);q1,q2=np.meshgrid(Q,Q)
    energy=W[0,0]*q1*q1+2*W[0,1]*q1*q2+W[1,1]*q2*q2
    results=dict(schema='wwl-state-v1',normalized_units=True,time_units='s',A=A.tolist(),B=B.tolist(),C=C.tolist(),
        R=rank_report(r),O=rank_report(o),near_O_loose=rank_report(near,rtol=1e-10),near_O_tight=rank_report(near,rtol=1e-14),scaled_O=rank_report(scaled),
        W_observable_T1=W.tolist(),W_hidden_T1=W_hidden.tolist(),
        W_observable_eigenvalues=np.linalg.eigvalsh(W).tolist(),
        hidden_t6=float(xs[-1,1]),output_t6=float(ys[-1]),near_visible_t6=float(visible[-1]),ic_visible_t6=float(ic[-1]),
        numerical_accuracy='floating point teaching checks; no interval enclosure',
        evidence='known-model CPU examples, not identification, plant or controller evidence')
    return dict(t=t,x=xs,y=ys,visible=np.array(visible),ic=np.array(ic),W=W,q1=q1,q2=q2,energy=energy,results=results)

def write_data(output):
    output=Path(output);output.mkdir(parents=True,exist_ok=True);d=experiment()
    np.savetxt(output/'trajectories.csv',np.column_stack([d['t'],d['x'],d['y'],d['visible'],d['ic']]),delimiter=',',header='t_s,x1_normalized,x2_normalized,y_hidden,y_epsilon_0.001,y_IC_visible',comments='',fmt='%.17g')
    np.savetxt(output/'energy.csv',np.column_stack([d['q1'].ravel(),d['q2'].ravel(),d['energy'].ravel()]),delimiter=',',header='x01_normalized,x02_normalized,output_energy_normalized_squared_s',comments='',fmt='%.17g')
    np.savetxt(output/'gramians.csv',np.column_stack([np.arange(4),d['W'].ravel(),np.array(d['results']['W_hidden_T1']).ravel()]),delimiter=',',header='rowmajor_index,Wo_observable_T1,Wo_hidden_T1',comments='',fmt='%.17g')
    (output/'results.json').write_text(json.dumps(d['results'],ensure_ascii=False,sort_keys=True,indent=2)+'\n')
    return d

if __name__=='__main__':
    p=argparse.ArgumentParser(description=__doc__);p.add_argument('--output',type=Path,required=True)
    print(json.dumps(write_data(p.parse_args().output)['results'],sort_keys=True))

"""Independent scalar/closed-form 2x2 and 3x3 oracles; never writes outputs."""
import math
import unittest
import warnings
import numpy as np
from state_lab import (model,expm,transition,constant_response,structure,gramian,
                       observability_gramian,rank_report,transfer,markov)

class StateTests(unittest.TestCase):
    def test_diagonal_2x2_input_and_initial_state(self):
        A=np.diag([-1.,1.]);B=[[1],[1]];C=[[1,0]];D=[[.2]]
        for t in [0,.001,.3,1.,6.]:
            x,y=constant_response(A,B,C,D,t,[.3,-.2],[.7])
            expected=[.3*math.exp(-t)+.7*(1-math.exp(-t)),
                      -.2*math.exp(t)+.7*(math.exp(t)-1)]
            np.testing.assert_allclose(x,expected,atol=1e-12)
            self.assertAlmostEqual(y[0],expected[0]+.14,places=12)

    def test_rotation_and_jordan_2x2(self):
        A=np.array([[0,-2],[2,0.]])
        for t in [0,.2,1,3]:
            np.testing.assert_allclose(transition(A,t),[[math.cos(2*t),-math.sin(2*t)],[math.sin(2*t),math.cos(2*t)]],atol=2e-14)
            np.testing.assert_allclose(transition([[-1,2],[0,-1]],t),math.exp(-t)*np.array([[1,2*t],[0,1]]),atol=2e-14)

    def test_nilpotent_3x3_solution_and_gramian(self):
        J=np.array([[0,1,0],[0,0,1],[0,0,0.]])
        B=np.array([[0],[0],[1.]])
        for t in [0,.01,.5,2]:
            expected=np.eye(3)+t*J+t*t/2*(J@J)
            np.testing.assert_allclose(transition(J,t),expected,atol=2e-14)
            x,y=constant_response(J,B,[[1,0,0]],[[0]],t,[1,2,3],[.4])
            want=[1+2*t+1.5*t*t+.4*t**3/6,2+3*t+.4*t*t/2,3+.4*t]
            np.testing.assert_allclose(x,want,atol=2e-14)
            W=[[t**5/20,t**4/8,t**3/6],[t**4/8,t**3/3,t*t/2],[t**3/6,t*t/2,t]]
            np.testing.assert_allclose(gramian(J,B,t),W,atol=2e-14)
        r,o=structure(J,B,[[1,0,0]])
        self.assertEqual(rank_report(r)['rank'],3);self.assertEqual(rank_report(o)['rank'],3)
        np.testing.assert_allclose(np.array(markov(J,B,[[1,0,0]],4)).ravel(),[0,0,1,0])

    def test_diagonal_gramian_scalar_integrals(self):
        A=np.diag([-1.,1.,0.]);B=np.array([[1.,2],[3,-1],[.5,2]])
        for T in [0,.1,1,2]:
            W=gramian(A,B,T)
            for i,ai in enumerate([-1.,1.,0.]):
                for j,aj in enumerate([-1.,1.,0.]):
                    scalar=T if ai+aj==0 else math.expm1((ai+aj)*T)/(ai+aj)
                    expected=sum(B[i,k]*B[j,k] for k in range(2))*scalar
                    self.assertAlmostEqual(W[i,j],expected,places=10)

    def test_transfer_same_and_singular_implementation(self):
        A=np.diag([-1.,1.])
        for s in [0,2,1j,1+2j]:
            a=transfer(A,[[1],[1]],[[1,0]],[[0]],s)[0,0]
            b=transfer(A,[[1],[0]],[[1,1]],[[0]],s)[0,0]
            self.assertAlmostEqual(abs(a-1/(s+1)),0,places=13)
            self.assertAlmostEqual(abs(a-b),0,places=13)
        with self.assertRaises(ValueError):transfer(A,[[1],[1]],[[1,0]],[[0]],1)
        # Reduced scalar transfer at 1 has value .5; full-state resolvent is singular.
        self.assertEqual(transfer([[-1]],[[1]],[[1]],[[0]],1)[0,0],.5)

    def test_hidden_vs_initially_visible_uncontrolled(self):
        A=np.diag([-1.,1.])
        x,y=constant_response(A,[[1],[1]],[[1,0]],[[0]],3,[0,0],[1])
        self.assertGreater(x[1],19);self.assertLess(y[0],1)
        _,y0=constant_response(A,[[1],[0]],[[1,1]],[[0]],3,[0,0],[1])
        _,yi=constant_response(A,[[1],[0]],[[1,1]],[[0]],3,[0,1],[0])
        self.assertAlmostEqual(y0[0],1-math.exp(-3));self.assertAlmostEqual(yi[0],math.exp(3))

    def test_cascade_exact_cancellation_and_perturbation(self):
        # K=(s-1+eps)/(s+1), P=1/(s-1); known-model algebra, not a controller run.
        for eps in [0.,.001]:
            A=[[-1,0],[-2+eps,1]]
            for t in [0,.1,1,4]:
                _,y=constant_response(A,[[1],[1]],[[0,1]],[[0]],t,[0,0],[1])
                want=1-math.exp(-t)+eps/2*(math.exp(t)+math.exp(-t)-2)
                self.assertAlmostEqual(y[0],want,places=11)
        _,yi=constant_response([[-1,0],[-2,1]],[[1],[1]],[[0,1]],[[0]],2,[0,.1],[0])
        self.assertAlmostEqual(yi[0],.1*math.exp(2),places=13)

    def test_exact_zero_near_zero_and_coordinates(self):
        A=np.diag([-1.,1.]);B=np.ones((2,1))
        _,o0=structure(A,B,[[1,0]]);_,on=structure(A,B,[[1,1e-12]])
        self.assertEqual(rank_report(o0,rtol=0)['rank'],1)
        self.assertEqual(rank_report(on,rtol=1e-10)['rank'],1)
        self.assertEqual(rank_report(on,rtol=1e-14)['rank'],2)
        S=np.diag([1,1e12]);_,o=structure(A,B,[[1,1]])
        _,os=structure(S@A@np.linalg.inv(S),S@B,np.array([[1,1]])@np.linalg.inv(S))
        self.assertEqual(rank_report(o)['rank'],2);self.assertEqual(rank_report(os)['rank'],1)
        # Exact det(O)=+2 epsilon is nonzero independently of a threshold.
        self.assertNotEqual(2e-12,0)

    def test_similarity_and_gramian_congruence(self):
        A=np.diag([-1.,-2.]);B=np.array([[1.],[2.]]);C=np.array([[1.,1.]]);S=np.array([[1.,1],[0,2.]])
        inv=np.linalg.inv(S);Az=S@A@inv;Bz=S@B;Cz=C@inv
        np.testing.assert_allclose(gramian(Az,Bz,1),S@gramian(A,B,1)@S.T,atol=2e-13)
        np.testing.assert_allclose(observability_gramian(Az,Cz,1),inv.T@observability_gramian(A,C,1)@inv,atol=2e-13)
        for t in [0,.2,1]:
            x,y=constant_response(A,B,C,[[0]],t,[.2,.3],[1])
            z,yz=constant_response(Az,Bz,Cz,[[0]],t,S@np.array([.2,.3]),[1])
            np.testing.assert_allclose(z,S@x,atol=2e-13);np.testing.assert_allclose(yz,y,atol=2e-13)

    def test_window_zero_and_observable_energy(self):
        A=np.diag([-1.,-2.]);C=[[1,1]]
        np.testing.assert_array_equal(observability_gramian(A,C,0),np.zeros((2,2)))
        for T in [.01,.1,1]:
            W=observability_gramian(A,C,T)
            self.assertGreater(np.linalg.eigvalsh(W)[0],0)
        hidden=observability_gramian(np.diag([-1.,1.]),[[1,0]],1)
        np.testing.assert_array_equal(hidden[:,1],[0,0])

    def test_object_complex_scalars_all_ports_rejected_without_warning(self):
        class Complex64Child(np.complex64):
            pass
        class Complex128Child(np.complex128):
            pass
        class ComplexChild(complex):
            pass
        scalar_types=(complex,np.complex64,np.complex128,
                      ComplexChild,Complex64Child,Complex128Child)
        with warnings.catch_warnings():
            warnings.simplefilter('error')
            for scalar_type in scalar_types:
                for value in [1+0j,1+2j]:
                    z=scalar_type(value)
                    matrix=np.array([[z]],dtype=object)
                    vector=np.array([z],dtype=object)
                    for slot in range(6):
                        args=[[[0]],[[1]],[[1]],[[0]],1,[0],[1]]
                        args[[0,1,2,3,5,6][slot]]=matrix if slot<4 else vector
                        with self.subTest(type=scalar_type.__name__,value=value,slot=slot):
                            with self.assertRaises(ValueError):constant_response(*args)
                    calls=[lambda:expm(matrix),lambda:transition(matrix,1),
                           lambda:structure([[0]],matrix,[[1]]),
                           lambda:structure([[0]],[[1]],matrix),
                           lambda:gramian(matrix,[[1]],1),
                           lambda:gramian([[0]],matrix,1),
                           lambda:observability_gramian([[0]],matrix,1),
                           lambda:transfer([[0]],[[1]],matrix,[[0]],1),
                           lambda:markov([[0]],matrix,[[1]],2),
                           lambda:rank_report(matrix)]
                    for call in calls:
                        with self.assertRaises(ValueError):call()
        # Object containers of real numbers remain legal at every model/IC/input port.
        args=[np.array(v,dtype=object) for v in [[[0]],[[1]],[[1]],[[0]]]]
        x,y=constant_response(*args,2,np.array([.5],dtype=object),np.array([.25],dtype=object))
        np.testing.assert_array_equal(x,[1]);np.testing.assert_array_equal(y,[1])

    def test_static_transfer_redundant_dynamic_state(self):
        # Theory admits n*=0, y=2u; the numerical model deliberately keeps n>=1.
        A,B,C,D=[[0]],[[0]],[[1]],[[2]]
        for t in [0,.5,2]:
            x,y=constant_response(A,B,C,D,t,[0],[.7])
            np.testing.assert_array_equal(x,[0]);np.testing.assert_allclose(y,[1.4])
        for s in [1,2,1j]:
            np.testing.assert_array_equal(transfer(A,B,C,D,s),[[2]])
        self.assertEqual(rank_report(structure(A,B,C)[0])['rank'],0)
        np.testing.assert_array_equal(np.array(markov(A,B,C,4)).ravel(),np.zeros(4))

    def test_rank_nonfinite_singular_values_or_threshold_rejected(self):
        # K=a*[[1,1],[1,-1]]: determinant=-2a^2 is nonzero, rank=2.
        # Each singular value is a*sqrt(2), beyond binary64 for a=1.7e308.
        with self.assertRaises(ValueError):rank_report([[1.7e308,1.7e308],[1.7e308,-1.7e308]])
        scaled=rank_report([[1.,1.],[1.,-1.]])
        self.assertEqual(scaled['rank'],2)
        np.testing.assert_allclose(scaled['singular_values'],[math.sqrt(2)]*2)
        with self.assertRaises(ValueError):rank_report([[2.]],rtol=1e308)

    def test_numerical_failure_and_invalid_shapes(self):
        with warnings.catch_warnings():
            warnings.simplefilter('error',RuntimeWarning)
            with self.assertRaises(ValueError):transition([[1]],1000)
            with self.assertRaises(ValueError):transition([[1e308]],1e308)
        for A in [[],[1,2],[[1,2]],[[np.nan]],[[np.inf]],[[1j]]]:
            with self.assertRaises(ValueError):expm(A)
        for T in [-1,np.inf,np.nan,1j,True]:
