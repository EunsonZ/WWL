```python
"""Actual grayscale pixel patches -> local translations, not identity certification.

I[v,u] uses row/column indexing; uv arrays store continuous (u,v) pixels.
Synthetic sequence has static frontoparallel textured patches and translating rigs.
The public photo is real texture with an explicitly artificial translation.
"""
from pathlib import Path
import argparse
import json
import numpy as np
from PIL import Image
from scipy.ndimage import gaussian_filter, uniform_filter
import stereo_lab as stereo


def image(value):
    a = np.asarray(value, dtype=float)
    if a.ndim != 2 or min(a.shape) < 3 or not np.isfinite(a).all():
        raise ValueError("image must be finite (H,W), H,W>=3")
    if np.min(a) < 0 or np.max(a) > 1:
        raise ValueError("gray intensities must be in [0,1]")
    return a


def sample(a, uv, gradient=False):
    """Bilinear interior sampler and its exact piecewise derivatives; no padding."""
    a, uv = image(a), stereo.batch(uv, 2)
    x, y = uv.T
    h,w = a.shape
    if np.any(x<0) or np.any(y<0) or np.any(x>=w-1) or np.any(y>=h-1):
        raise ValueError("sample outside bilinear interior")
    u,v = np.floor(x).astype(int), np.floor(y).astype(int)
    dx,dy = x-u,y-v
    p,q,r,s = a[v,u],a[v,u+1],a[v+1,u],a[v+1,u+1]
    values = (1-dy)*((1-dx)*p+dx*q)+dy*((1-dx)*r+dx*s)
    if gradient:
        jac = np.column_stack(((1-dy)*(q-p)+dy*(s-r),
                               (1-dx)*(r-p)+dx*(s-q)))
        return values,jac
    return values


def offsets(half):
    if not isinstance(half,int) or isinstance(half,bool) or half<1:
        raise ValueError("half window must be a positive integer")
    y,x=np.mgrid[-half:half+1,-half:half+1]
    return np.column_stack((x.ravel(),y.ravel()))


def solve_patch(a,b,uv,half=5,search=5,min_eigen=5e-5):
    """Coarse integer SSD then forward-additive Gauss Newton, at most 40 steps."""
    if not isinstance(search,int) or isinstance(search,bool) or search<0:
        raise ValueError("search must be a nonnegative integer")
    a,b=image(a),image(b)
    if a.shape!=b.shape:
        raise ValueError("image shapes differ")
    center=stereo.principal(uv)
    grid=center+offsets(half)
    template,j0=sample(a,grid,True)
    eig=np.linalg.eigvalsh(j0.T@j0/len(grid))
    if eig[0]<min_eigen:
        raise ValueError("flat/edge aperture: insufficient two-direction texture")
    best=None
    for dy in range(-search,search+1):
        for dx in range(-search,search+1):
            d=np.array([dx,dy],float)
            try: residual=sample(b,grid+d)-template
            except ValueError: continue
            score=float(residual@residual)
            if best is None or score<best[0]: best=(score,d)
    if best is None: raise ValueError("no valid search window")
    d=best[1].copy()
    converged=False
    for it in range(40):
        values,jac=sample(b,grid+d,True)
        if np.linalg.eigvalsh(jac.T@jac/len(grid))[0]<min_eigen:
            raise ValueError("target loses two-direction texture")
        step=np.linalg.lstsq(jac,template-values,rcond=None)[0]
        # Backtracking keeps the actual nonlinear SSD from increasing.
        cost=float(np.sum((values-template)**2))
        accepted=False
        for fraction in (1.,.5,.25,.125,.0625,.03125,.015625,.0078125,.00390625,.001953125,.0009765625):
            proposal=d+fraction*step
            if np.linalg.norm(proposal,np.inf)>search+1:
                continue
            try: trial=sample(b,grid+proposal)-template
            except ValueError: continue
            if trial@trial<=cost+1e-15:
                d=proposal; accepted=True; break
        if not accepted:
            converged=True  # declared finite line-search stopping, not global optimality
            break
        if np.linalg.norm(fraction*step)<1e-5:
            converged=True; break
    residual=sample(b,grid+d)-template
    return {"uv_px":(center+d).tolist(),"displacement_px":d.tolist(),
            "rms_intensity":float(np.sqrt(np.mean(residual**2))),
            "source_mean_eigenvalues":eig.tolist(),"iterations":it+1,
            "stopped":converged,"stop_rule":"small step <=1e-5 px or no descent at fractions 2^0..2^-10; not global optimum"}


def track(a,b,points,half=5,search=5,min_eigen=5e-5,max_rms=.035,max_fb=.15):
    rows=[]
    for uv in stereo.batch(points,2):
        try:
            item=solve_patch(a,b,uv,half,search,min_eigen)
            if not item["stopped"]: raise ValueError("local iteration hit limit")
            if item["rms_intensity"]>max_rms: raise ValueError("brightness/occlusion residual exceeds policy")
            reverse=solve_patch(b,a,item["uv_px"],half,search,min_eigen)
            if not reverse["stopped"]: raise ValueError("reverse iteration hit limit")
            fb=float(np.linalg.norm(np.asarray(reverse["uv_px"])-uv))
            if fb>max_fb: raise ValueError("forward/backward disagreement exceeds policy")
            item.update({"accepted":True,"fb_error_px":fb})
        except ValueError as exc:
            item={"accepted":False,"reason":str(exc)}
        rows.append(item)
    return rows


def detect(a,count=12,half=5,min_distance=12):
    """Greedy minimum-eigenvalue patch detector; deterministic tie order v then u."""
    a=image(a)
    gy,gx=np.gradient(a)
    h00=uniform_filter(gx*gx,size=2*half+1)
    h11=uniform_filter(gy*gy,size=2*half+1)
    h01=uniform_filter(gx*gy,size=2*half+1)
    score=(h00+h11-np.sqrt((h00-h11)**2+4*h01*h01))/2
    margin=half+8
    score[:margin]=0; score[-margin:]=0; score[:,:margin]=0; score[:,-margin:]=0
    order=np.argsort(-score.ravel(),kind="stable")
    points=[]
    for idx in order:
        v,u=divmod(int(idx),a.shape[1])
        if score[v,u]<5e-5: break
        if all(np.linalg.norm(np.array([u,v])-p)>=min_distance for p in points):
            points.append(np.array([u,v]))
            if len(points)==count: break
    if not points: raise ValueError("no two-direction texture detected")
    return np.asarray(points,float)


def scene(spec):
    uv=stereo.batch(spec["seed_left_px"],2)
    z=np.asarray(spec["depth_m"],float)
    f,c=spec["f_px"],np.asarray(spec["principal_px"])
    points=np.column_stack(((uv-c)*z[:,None]/f,z))
    centers=stereo.batch(spec["centers_world_m"],3)
    if np.any(centers[:,2]!=0):
        raise ValueError("fixture supports lateral translation only; texture depth scaling is not modeled")
    shape=spec["shape_t_eye_h_w"]
    if shape[:2]!=[len(centers),2] or len(points)!=12: raise ValueError("fixture shape mismatch")
    frames=np.full(shape,.08,float)
    truth=np.empty((len(centers),2,len(points),2))
    yy,xx=np.mgrid[:shape[2],:shape[3]]
    for t,center in enumerate(centers):
        for eye in range(2):
            camera=points-center-np.array([eye*spec["baseline_m"],0,0])
            pixels=stereo.project(camera,f,c)
            truth[t,eye]=pixels
            for j,p in enumerate(pixels):
                dx,dy=xx-p[0],yy-p[1]
                inside=(abs(dx)<=12)&(abs(dy)<=12)
                tex=(.45+.17*np.sin(.39*dx+.17*dy+j*.43)
                     +.16*np.cos(-.23*dx+.41*dy+j*.27)
                     +.09*np.sin(.13*dx-.27*dy+j*.61))
                frames[t,eye][inside]=tex[inside]
    return frames.astype(np.float32),truth,points


def sequence_tracks(frames,spec):
    tracker=spec["tracker"]
    kw=dict(half=tracker["half_window"],min_eigen=tracker["min_eigen_mean"],
            max_rms=tracker["max_rms"],max_fb=tracker["max_fb_px"])
    observations=np.empty((len(frames),2,12,2))
    observations[0,0]=spec["seed_left_px"]  # supplied seeds, not detected identities
    logs=[]
    for t in range(len(frames)):
        if t:
            rows=track(frames[t-1,0],frames[t,0],observations[t-1,0],
                       search=tracker["search_temporal"],**kw)
            if not all(r["accepted"] for r in rows): raise ValueError(f"temporal lost track at {t}: {rows}")
            observations[t,0]=[r["uv_px"] for r in rows]
            logs.append({"time":t,"kind":"temporal","rows":rows})
        rows=track(frames[t,0],frames[t,1],observations[t,0],search=tracker["search_stereo"],**kw)
        if not all(r["accepted"] for r in rows): raise ValueError(f"stereo lost track at {t}: {rows}")
        observations[t,1]=[r["uv_px"] for r in rows]
        logs.append({"time":t,"kind":"stereo","rows":rows})
    return observations,logs


def photo_case(root,spec):
    original=np.asarray(Image.open(root/"inputs/camera.png"),float)/255
    y,x,h,w=spec["photo_crop_yxhw"]
    a=gaussian_filter(original[y:y+h,x:x+w],1.2,mode="reflect")
    yy,xx=np.mgrid[:h,:w]
    delta=np.asarray(spec["photo_translation_px"])
    coords=np.column_stack((xx.ravel()-delta[0],yy.ravel()-delta[1]))
    # Only this declared warp uses clipped border extension; LK samples no border.
    coords[:,0]=np.clip(coords[:,0],0,w-1-1e-9); coords[:,1]=np.clip(coords[:,1],0,h-1-1e-9)
    b=sample(a,coords).reshape(h,w)
    seeds=detect(a)
    rows=track(a,b,seeds,search=3)
    errors=[float(np.linalg.norm(np.asarray(r["displacement_px"])-delta)) if r["accepted"] else None for r in rows]
    return a,b,seeds,{"source":"CC0 real photo, Gaussian-smoothed crop + artificial bilinear translation",
      "translation_px":delta.tolist(),"seeds_px":seeds.tolist(),"rows":rows,"errors_px":errors}


def failures(a,spec):
    yy,xx=np.mgrid[:64,:64]
    flat=np.full((64,64),.5)
    edge=.5+.2*np.sin(.3*xx)
    texture=.5+.15*np.sin(.3*xx)+.15*np.cos(.4*yy)
    bright=np.clip(texture+.18,0,1)
    # One whole-period wrong correspondence is observationally indistinguishable.
    periodic=.5+.15*np.cos(2*np.pi*xx/8)+.15*np.cos(2*np.pi*yy/8)
    periodic_result=track(periodic,periodic,[[24,24]],search=0)
    p=periodic_result[0]
    return {"flat":track(flat,flat,[[24,24]],search=2)[0],
            "edge":track(edge,edge,[[24,24]],search=2)[0],
            "brightness_change":track(texture,bright,[[24,24]],search=2)[0],
            "occluded":track(texture,flat,[[24,24]],search=2)[0],
            "periodic_wrong_identity":{"declared_true_displacement_px":[8,0],
                  "estimated":p,"identity_error_px":8.0,
                  "note":"periodic texture shifted by 8 px equals same raster; zero/FB cannot identify period"}}


def run(spec,root,output):
    frames,truth,points=scene(spec)
    uv,logs=sequence_tracks(frames,spec)
    a,b,seeds,photo=photo_case(root,spec)
    shifted_seeds=np.asarray(spec["seed_left_px"],float)+[1.5,-.75]
    biased=track(frames[0,0],frames[1,0],shifted_seeds,search=5)
    localized={"seed_offset_px":[1.5,-.75],"rows":biased,
       "errors_to_intended_patch_center_px":[float(np.linalg.norm(np.asarray(r["uv_px"])-p))
           if r["accepted"] else None for r,p in zip(biased,truth[1,0])],
       "meaning":"a perturbed supplied localization keeps tracking a nearby patch location; tracking is not a semantic center detector"}
    small_search=track(frames[0,0],frames[1,0],spec["seed_left_px"],search=0)
    output.mkdir(parents=True,exist_ok=True)
    np.save(output/"motion-pixels.npy",frames,allow_pickle=False)
    np.save(output/"photo-pair.npy",np.stack((a,b)),allow_pickle=False)
    result={"model":"synthetic stereo time sequence; public photo has artificial warp",
      "units":{"uv":"px","gray":"0..1","xyz":"m"},"shape":list(frames.shape),
      "truth_points_world_m":points.tolist(),"truth_pixels_px":truth.tolist(),
      "tracked_pixels_px":uv.tolist(),"pixel_errors_px":np.linalg.norm(uv-truth,axis=-1).tolist(),
      "tracking_log":logs,"photo":photo,"seed_localization_perturbation":localized,
      "too_small_search":small_search,"failure_cases":failures(a,spec)}
    (output/"flow-results.json").write_text(json.dumps(result,ensure_ascii=False,indent=2,allow_nan=False)+"\n")
    return result


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    root=Path(__file__).resolve().parent
    parser.add_argument("--input",type=Path,default=root/"inputs/motion-spec.json")
    parser.add_argument("--output",type=Path,default=root/"results")
    args=parser.parse_args()
    result=run(json.loads(args.input.read_text()),root,args.output)
    print(json.dumps({"max_synthetic_track_error_px":np.max(result["pixel_errors_px"]),
      "photo_errors_px":result["photo"]["errors_px"],"failures":result["failure_cases"]},indent=2))

if __name__=="__main__": main()
```

```python
"""A fixed observed stereo graph: localization versus joint metric window BA.

World = first left camera. R_CW P_W+t_CW; right point = left point-(B,0,0).
No online SLAM, descriptor matching, loop detection, or truth-based initialization.
"""
from pathlib import Path
import argparse
import json
import numpy as np
from scipy.optimize import least_squares
from scipy.spatial.transform import Rotation
import stereo_lab as stereo
import multiview_lab as geometry


def unpack(x,times=5,landmarks=12):
    x=np.asarray(x,float)
    count=6*(times-1)
    if x.shape!=(count+3*landmarks,) or not np.isfinite(x).all():
        raise ValueError("invalid finite pose/landmark state")
    poses=np.vstack((np.zeros(6),x[:count].reshape(times-1,6)))
    points=x[count:].reshape(landmarks,3)
    return poses,points


def pack(poses,points):
    poses,points=stereo.batch(poses,6),stereo.batch(points,3)
    if not np.array_equal(poses[0],np.zeros(6)): raise ValueError("pose0 must be anchored I/0")
    return np.r_[poses[1:].ravel(),points.ravel()]


def camera_points(points,pose):
    p=stereo.batch(points,3); pose=np.asarray(pose,float)
    if pose.shape!=(6,) or not np.isfinite(pose).all(): raise ValueError("pose must be finite (6,)")
    return p@Rotation.from_rotvec(pose[:3]).as_matrix().T+pose[3:]


def predict(poses,points,edges,spec,baseline=None):
    b=spec["baseline_m"] if baseline is None else float(baseline)
    if not np.isfinite(b) or b<0: raise ValueError("baseline must be finite >=0")
    output=[]
    for row in edges:
        t,eye,j=row["time"],row["eye"],row["landmark"]
        if not(0<=t<len(poses) and eye in (0,1) and 0<=j<len(points)):
            raise ValueError("edge index outside state")
        pc=camera_points(points[j:j+1],poses[t])-np.array([eye*b,0.,0.])
        output.append(stereo.project(pc,spec["f_px"],spec["principal_px"])[0])
    return stereo.batch(output,2)


def graph(uv,spec,closure="none"):
    if closure not in ("none","true","wrong"): raise ValueError("unknown closure")
    uv=np.asarray(uv,float)
    if uv.shape!=(5,2,12,2) or not np.isfinite(uv).all(): raise ValueError("observations must be finite (5,2,12,2)")
    edges=[]
    for t,ids in enumerate(spec["base_ids_by_frame"]):
        for j in ids:
            for eye in range(2):
                edges.append({"time":t,"eye":eye,"landmark":j,"pixel_source_id":j,
                              "uv_px":uv[t,eye,j].tolist(),"closure":False})
    if closure!="none":
        targets=spec["closure_ids_frame4"] if closure=="true" else [3,4,5,0,1,2]
        for source,target in zip(spec["closure_ids_frame4"],targets):
            for eye in range(2):
                edges.append({"time":4,"eye":eye,"landmark":target,"pixel_source_id":source,
                              "uv_px":uv[4,eye,source].tolist(),"closure":True})
    keys=[(e["time"],e["eye"],e["landmark"]) for e in edges]
    if len(set(keys))!=len(keys): raise ValueError("duplicate image factor")
    return edges


def bounds():
    # |rotvec components|<=.08 rad and |t components|<=.15 m; world box.
    lower=np.r_[np.tile([-.08,-.08,-.08,-.15,-.15,-.15],4),np.tile([-1.,-.8,.8],12)]
    upper=np.r_[np.tile([.08,.08,.08,.15,.15,.15],4),np.tile([1.,.8,3.],12)]
    return lower,upper


def initialize(uv,spec):
    """Sequential stereo points + known-map pose fits, using observations only."""
    uv=np.asarray(uv,float)
    poses=np.zeros((5,6)); points=np.full((12,3),np.nan)
    lo,hi=bounds()
    for t,ids in enumerate(spec["base_ids_by_frame"]):
        old=[j for j in ids if np.isfinite(points[j]).all()]
        if t:
            if len(old)<3: raise ValueError("initialization requires >=3 existing map points")
            local_edges=[{"time":0,"eye":eye,"landmark":j} for j in old for eye in range(2)]
            observed=np.array([uv[t,e["eye"],e["landmark"]] for e in local_edges])
            def residual(pose): return (predict(pose[None,:],points,local_edges,spec)-observed).ravel()
            result=least_squares(residual,poses[t-1],bounds=(lo[:6],hi[:6]),
                                 ftol=1e-11,xtol=1e-11,gtol=1e-11,max_nfev=120)
            if not result.success: raise ValueError("initial known-map pose fit failed")
            poses[t]=result.x
        new=[j for j in ids if not np.isfinite(points[j]).all()]
        if new:
            local=np.asarray(geometry.triangulate(uv[t,0,new],uv[t,1,new],np.eye(3),
                [spec["baseline_m"],0,0],spec["f_px"],spec["principal_px"],
                spec["f_px"],spec["principal_px"])["points_left_m"])
            r=Rotation.from_rotvec(poses[t,:3]).as_matrix()
            points[new]=(local-poses[t,3:])@r
    initial=pack(poses,points)
    if np.any(initial<=lo) or np.any(initial>=hi): raise ValueError("initializer outside declared solver box")
    return initial


def fit(uv,spec,closure="none",loss="linear",initial=None):
    if loss not in ("linear","soft_l1"): raise ValueError("loss policy must be linear or soft_l1")
    edges=graph(uv,spec,closure)
    observed=np.asarray([e["uv_px"] for e in edges])
    x0=initialize(uv,spec) if initial is None else np.asarray(initial,float)
    def residual(x):
        poses,points=unpack(x)
        return (predict(poses,points,edges,spec)-observed).ravel()
    result=least_squares(residual,x0,bounds=bounds(),loss=loss,f_scale=spec["ba"]["loss_scale_px"],
        x_scale="jac",ftol=1e-10,xtol=1e-10,gtol=1e-10,max_nfev=spec["ba"]["max_nfev"])
    if not result.success: raise ValueError(f"window solver failed: {result.message}")
    poses,points=unpack(result.x)
    pred=predict(poses,points,edges,spec)
    return {"closure":closure,"loss":loss,"poses_rotvec_t_CW":poses.tolist(),
      "points_world_m":points.tolist(),"edges":edges,"predicted_px":pred.tolist(),
      "residual_px":(pred-observed).tolist(),"point_rms_px":float(np.sqrt(np.mean(np.sum((pred-observed)**2,axis=1)))),
      "cost":float(result.cost),"nfev":int(result.nfev),"status":int(result.status),
      "message":result.message,"initial_state":x0.tolist(),"active_bound_count":int(np.count_nonzero(result.active_mask))}


def centers(poses):
    return np.asarray([-Rotation.from_rotvec(p[:3]).as_matrix().T@p[3:] for p in poses])


def evaluate(result,true_points,true_centers):
    poses=np.asarray(result["poses_rotvec_t_CW"])
    c=centers(poses)
    result["centers_world_m"]=c.tolist()
    result["center_errors_m"]=np.linalg.norm(c-true_centers,axis=1).tolist()
    result["map_errors_m"]=np.linalg.norm(np.asarray(result["points_world_m"])-true_points,axis=1).tolist()
    result["rotation_errors_rad"]=np.linalg.norm(poses[:,:3],axis=1).tolist()
    result["trajectory_rms_m"]=float(np.sqrt(np.mean(np.sum((c-true_centers)**2,axis=1))))
    result["map_rms_m"]=float(np.sqrt(np.mean(np.sum((np.asarray(result["points_world_m"])-true_points)**2,axis=1))))
    return result


def known_map(uv,spec,true_points):
    # Evaluation comparison: externally surveyed map is known, never joint-fit.
    lo,hi=bounds(); poses=np.zeros((5,6))
    for t in range(1,5):
        edges=[{"time":0,"eye":eye,"landmark":j} for j in spec["base_ids_by_frame"][t] for eye in range(2)]
        obs=np.asarray([uv[t,e["eye"],e["landmark"]] for e in edges])
        def fun(pose): return (predict(pose[None,:],true_points,edges,spec)-obs).ravel()
        result=least_squares(fun,poses[t-1],bounds=(lo[:6],hi[:6]),ftol=1e-12,xtol=1e-12,gtol=1e-12,max_nfev=120)
        if not result.success: raise ValueError("known map localization failed")
        poses[t]=result.x
    return {"poses_rotvec_t_CW":poses.tolist(),"centers_world_m":centers(poses).tolist(),
            "map_status":"truth supplied as externally known map for comparison; not SLAM"}


def jacobian(fun,x,h=1e-6):
    columns=[]
    for k in range(len(x)):
        step=np.zeros(len(x)); step[k]=h
        columns.append((fun(x+step)-fun(x-step))/(2*h))
    return np.column_stack(columns)


def gauges(spec,points,truth_centers,uv):
    poses=np.column_stack((np.zeros((5,3)),-truth_centers))
    edges=graph(uv,spec,"true")
    base=predict(poses,points,edges,spec)
    q=Rotation.from_rotvec([.13,-.08,.06]).as_matrix(); a=np.array([.2,-.1,.15])
    transformed=[]
    for p in poses:
        r=Rotation.from_rotvec(p[:3]).as_matrix()
        rp=r@q.T; tp=p[3:]-rp@a
        transformed.append(np.r_[Rotation.from_matrix(rp).as_rotvec(),tp])
    se3=predict(np.asarray(transformed),points@q.T+a,edges,spec)
    scale=1.7
    scaled=poses.copy(); scaled[:,3:]*=scale
    mono=predict(poses,points,edges,spec,0.)
    mono_scaled=predict(scaled,scale*points,edges,spec,0.)
    metric_scaled=predict(scaled,scale*points,edges,spec)
    x=pack(poses,points)
    def make(b):
        def f(state):
            pp,xx=unpack(state)
            return predict(pp,xx,edges,spec,b).ravel()
        return f
    js=jacobian(make(spec["baseline_m"]),x)
    jm=jacobian(make(0.),x)
    svs,svm=np.linalg.svd(js,compute_uv=False),np.linalg.svd(jm,compute_uv=False)
    direction=np.r_[np.column_stack((np.zeros((4,3)),poses[1:,3:])).ravel(),points.ravel()]
    return {"global_SE3_max_pixel_change_px":float(np.max(np.abs(se3-base))),
      "anchored_monocular_scale_max_pixel_change_px":float(np.max(np.abs(mono_scaled-mono))),
      "fixed_baseline_scaled_state_max_pixel_change_px":float(np.max(np.abs(metric_scaled-base))),
      "metric_jacobian_shape":list(js.shape),"metric_singular_values":svs.tolist(),
      "monocular_singular_values":svm.tolist(),"rank_policy":"sv > max_sv * 1e-7",
      "metric_rank":int(np.count_nonzero(svs>svs[0]*1e-7)),
      "monocular_rank":int(np.count_nonzero(svm>svm[0]*1e-7)),
      "mono_scale_direction_J_norm_px":float(np.linalg.norm(jm@direction)),"scale_factor":scale}


def run(spec,flow,output):
    tracked=np.asarray(flow["tracked_pixels_px"])
    true_uv=np.asarray(flow["truth_pixels_px"])
    true_points=np.asarray(flow["truth_points_world_m"])
    true_centers=np.asarray(spec["centers_world_m"])
    noise=np.random.default_rng(spec["seed"]).normal(0,spec["coordinate_noise_px"],tracked.shape)
    noisy=tracked+noise
    cases={}
    clean=fit(true_uv,spec,"true")
    cases["clean_true_closure"]=evaluate(clean,true_points,true_centers)
    for name,closure,loss in [("no_closure","none","linear"),("true_closure","true","linear"),
                              ("wrong_closure","wrong","linear"),("wrong_closure_robust","wrong","soft_l1")]:
        cases[name]=evaluate(fit(noisy,spec,closure,loss),true_points,true_centers)
    known=known_map(noisy,spec,true_points)
    result={"model":"fixed metric stereo window; supplied association graph; no automatic loop discovery",
      "state":"4 free SE3 poses +12 3D points =60; pose0 anchored; known B",
      "units":{"pose_rotvec":"rad","pose_t/map":"m","uv/residual":"px"},
      "coordinate_noise_px":noise.tolist(),"noisy_tracked_pixels_px":noisy.tolist(),
      "cases":cases,"known_map_localization":known,"gauges":gauges(spec,true_points,true_centers,true_uv)}
    output.mkdir(parents=True,exist_ok=True)
    (output/"window-ba-results.json").write_text(json.dumps(result,ensure_ascii=False,indent=2,allow_nan=False)+"\n")
    return result


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    root=Path(__file__).resolve().parent
    parser.add_argument("--input",type=Path,default=root/"inputs/motion-spec.json")
    parser.add_argument("--flow",type=Path,default=root/"results/flow-results.json")
    parser.add_argument("--output",type=Path,default=root/"results")
    args=parser.parse_args()
    result=run(json.loads(args.input.read_text()),json.loads(args.flow.read_text()),args.output)
    print(json.dumps({name:{k:r[k] for k in ("point_rms_px","trajectory_rms_m","map_rms_m","active_bound_count","nfev")}
                      for name,r in result["cases"].items()},indent=2))
    print(json.dumps(result["gauges"],indent=2))

if __name__=="__main__": main()
```

```python
class MotionChecks(unittest.TestCase):
    def test_bilinear_gradient_and_domain(self):

        a=.1+.01*x+.02*y+.0001*x*y
        uv=np.array([[4.3,6.7],[11.2,8.1]])

        with self.assertRaises(ValueError): flow.sample(a,[[23.,4.]])
        with self.assertRaises(ValueError): flow.image(np.full((4,4),np.nan))

    def test_aperture_and_model_failures(self):
        result=flow.failures(None,SPEC)
        for name in ("flat","edge","brightness_change","occluded"):
            self.assertFalse(result[name]["accepted"],name)
        wrong=result["periodic_wrong_identity"]
        self.assertTrue(wrong["estimated"]["accepted"])
        self.assertEqual(wrong["estimated"]["fb_error_px"],0)
        self.assertEqual(wrong["identity_error_px"],8)
        with self.assertRaises(ValueError): flow.detect(np.full((40,40),.5))

    def test_exact_local_constraint_and_aperture_null(self):
        j=np.array([[1,2],[1,-1],[1,-2]],float)
        it=np.array([0,-.375,-.5])
        d=np.linalg.lstsq(j,-it,rcond=None)[0]
        np.testing.assert_allclose(d,[.25,-.125],atol=1e-15)
        edge=np.array([[1,0],[2,0],[-1,0]],float)
        np.testing.assert_array_equal(edge@np.array([0,7]),np.zeros(3))

    def test_unique_closure_factors(self):
        uv=np.zeros((5,2,12,2))
        base=ba.graph(uv,SPEC,"none"); closed=ba.graph(uv,SPEC,"true"); wrong=ba.graph(uv,SPEC,"wrong")
        self.assertEqual(len(base),66); self.assertEqual(len(closed),78)
        keys=lambda rows:{(r["time"],r["eye"],r["landmark"]) for r in rows}
        self.assertEqual(len(keys(closed)),78)
        self.assertEqual(len(keys(closed)-keys(base)),12)
        self.assertEqual(keys(closed),keys(wrong))
        self.assertEqual(sum(r["landmark"]!=r["pixel_source_id"] for r in wrong),12)
        with self.assertRaises(ValueError): ba.graph(uv[:2],SPEC)

    def test_full_pose_and_measured_baseline(self):
```
